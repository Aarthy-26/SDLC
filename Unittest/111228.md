    @Test
    void resolve_nonAava_preservesHeadersAndOverrideAndDoesNotCallTokenResolver() {
        GenericAPI api = apiWithGroup(groupWithNameOnly("NON_AAVA"));
        when(baseUrlResolverService.resolveUrl(api)).thenReturn("https://example/base");

        Map<String, String> headers = new LinkedHashMap<>();
        headers.put("X-Correlation-Id", "c1");
        headers.put(HttpHeaders.AUTHORIZATION, "Bearer SHOULD_BE_STRIPPED");

        var req = new AavaUserTokenAuthentication.GenericAPIExecuteRequestAdapter(headers, ref(UUID.randomUUID(), "u"), true);

        var ctx = service.resolve(api, req);

        assertThat(ctx.resolvedUrl()).isEqualTo("https://example/base");
        assertThat(ctx.authOverrideAllowed()).isTrue();
        assertThat(ctx.headers()).containsEntry("X-Correlation-Id", "c1");
        assertThat(ctx.headers().keySet()).noneMatch(k -> k.equalsIgnoreCase(HttpHeaders.AUTHORIZATION));
        verifyNoInteractions(userAavaTokenResolver);
    }

    @Test
    void resolve_aavaByGroupType_injectsAuthorizationAndForcesOverrideFalse() {
        ApiBaseUrlGroup group = groupWithTypeAndName("AAVA", "anything");
        GenericAPI api = apiWithGroup(group);

        when(userAavaTokenResolver.resolveValidToken(any())).thenReturn(new AavaUserTokenAuthentication.ResolvedUserToken("tok", Instant.now().plusSeconds(3600)));

        Map<String, String> headers = new LinkedHashMap<>();

            var token = new AavaUserTokenAuthentication.RuntimeToken("tok", "r", Instant.now().plusSeconds(600));

            var t1 = resolver.resolveValidToken(ref);

            assertEquals("tok", t2.accessToken());
            // provider should be called only once because cache hit on second call
            verify(runtimeTokenProvider, times(1)).findToken(any());
            verifyNoInteractions(refreshGateway);

        @Test
        void resolveValidToken_expired_refreshesAndCaches() {
            UUID userId = UUID.randomUUID();
            var ref = new AavaUserTokenAuthentication.RuntimeUserReference(userId, "u");

            var expired = new AavaUserTokenAuthentication.RuntimeToken("old", "r", Instant.now());
            var refreshed = new AavaUserTokenAuthentication.RuntimeToken("new", "r", Instant.now().plusSeconds(600));
            when(runtimeTokenProvider.findToken(any())).thenReturn(expired);
            when(refreshGateway.refresh(any(), any())).thenReturn(refreshed);

            var resolved = resolver.resolveValidToken(ref);
            assertEquals("new", resolved.accessToken());
            verify(refreshGateway, times(1)).refresh(any(), any());

            resolver.resolveValidToken(ref);
            verify(runtimeTokenProvider, times(1)).findToken(any());
