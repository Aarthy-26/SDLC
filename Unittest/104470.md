    private static void setAuth(String principal, boolean authenticated, String... roles) {
        var authorities = new ArrayList<SimpleGrantedAuthority>();
        for (String r : roles) authorities.add(new SimpleGrantedAuthority(r));
        var auth = new TestingAuthenticationToken(principal, "n/a", authorities);
        auth.setAuthenticated(authenticated);
        SecurityContextHolder.getContext().setAuthentication(auth);
    }

    private static AppRow appRow(UUID appId, Integer activeVersion) {
        return new AppRow(appId, "app", "{\"k\":\"v\"}", activeVersion, OffsetDateTime.now());
    }

    private static AppVersionRow verRow(UUID appId, int versionNumber, String configJson, String createdBy) {
        return new AppVersionRow(UUID.randomUUID(), appId, versionNumber, configJson, createdBy, OffsetDateTime.now(), false, true);
    }

    @Test
    void tc01_getVersions_nonEmpty_marksActiveCorrectly() {
        UUID appId = UUID.randomUUID();
        when(appRepository.findById(appId)).thenReturn(Optional.of(appRow(appId, 2)));
        when(userService.getCurrentUser()).thenReturn(Optional.of(mock(UserModel.class)));

        var rows = List.of(
                verRow(appId, 3, "{c:3}", "u1"),
                verRow(appId, 2, "{c:2}", "u2"),
                verRow(appId, 1, "{c:1}", "u3")
        );
        when(appVersionRepository.findValidByAppIdLatestFirst(appId)).thenReturn(rows);

        List<AppVersionMetadata> out = service.getApplicationVersions(appId);

        assertEquals(3, out.size());
        assertFalse(out.get(0).active());
        assertTrue(out.get(1).active());
        assertFalse(out.get(2).active());
        assertEquals(List.of(3,2,1), out.stream().map(AppVersionMetadata::versionNumber).toList());
    }

    @Test
    void tc08_restore_success_adminRole_updatesAndAudits() {
        UUID appId = UUID.randomUUID();
        int versionNumber = 3;

        setAuth("admin", true, "ROLE_ADMIN");
        when(appRepository.findByIdForUpdate(appId)).thenReturn(Optional.of(appRow(appId, 1)));
        when(appVersionRepository.findValidByAppIdAndVersionNumber(appId, versionNumber))
                .thenReturn(Optional.of(verRow(appId, versionNumber, "{\"cfg\":3}", "creator")));

        var user = mock(UserModel.class);
        when(user.getUsername()).thenReturn("alice");
        when(userService.getCurrentUser()).thenReturn(Optional.of(user));

        RestoreResult res = service.restoreApplicationVersion(appId, versionNumber);

        assertEquals(appId, res.appId());
        assertEquals(versionNumber, res.restoredVersionNumber());
        assertEquals("alice", res.restoredBy());
        assertNotNull(res.restoredAt());

        verify(appRepository, times(1)).updateConfigAndActiveVersion(eq(appId), eq("{\"cfg\":3}"), eq(versionNumber));
        verify(auditService, times(1)).logAction(eq("APP_VERSION_RESTORE"), contains("appId=" + appId));
    }
