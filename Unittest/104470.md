 * - Assertions are resilient: validate status codes + key fields (not timestamps).

    private static final HttpClient HTTP = HttpClient.newBuilder()
            .connectTimeout(Duration.ofSeconds(2))
            .build();

    private static String baseUrl;

    static void startServer() throws Exception {
        port = findFreePort();
        baseUrl = "http://localhost:" + port;

        // Start server via reflection (start() is private)
        AppVersionHistoryBackend backend = new AppVersionHistoryBackend();

        Method seed = AppVersionHistoryBackend.class.getDeclaredMethod("seedSampleData");
        seed.setAccessible(true);
        seed.invoke(backend);

        Method start = AppVersionHistoryBackend.class.getDeclaredMethod("start", int.class);
        start.setAccessible(true);
        start.invoke(backend, port);

        // Small delay to allow HttpServer to bind.
        Thread.sleep(150);
    }

    // ----------------------- GET /versions -----------------------

    @Test
    void tc01_getVersions_happyPath_adminToken() throws Exception {
        HttpResponse<String> resp = send("GET", "/api/apps/app-1/versions", "admin-token", null, null);
        assertEquals(200, resp.statusCode());
        assertTrue(resp.body().contains("\"success\":true"));
        assertTrue(resp.body().contains("\"message\":\"Versions retrieved\""));
        assertTrue(resp.body().contains("\"appId\":\"app-1\""));
        assertTrue(resp.body().contains("\"activeVersion\":3"));

        // Order and filtering: should include 3 then 2; should not include 1
        int idx3 = resp.body().indexOf("\"versionNumber\":3");
        int idx2 = resp.body().indexOf("\"versionNumber\":2");
        int idx1 = resp.body().indexOf("\"versionNumber\":1");
        assertTrue(idx3 >= 0 && idx2 >= 0, "Expected versions 3 and 2");
        assertTrue(idx3 < idx2, "Expected version 3 before version 2");
        assertEquals(-1, idx1, "Inaccessible version 1 must not be returned");

        // Active flag: v3 active true
        assertTrue(resp.body().contains("\"versionNumber\":3") && resp.body().contains("\"active\":true"));
    }

    void tc15_restore_forcedFailure_rollsBackActiveVersion() throws Exception {
        // set to version 3 first to create a known baseline
        HttpResponse<String> baseline = send("POST", "/api/apps/app-1/versions/3/restore", "admin-token", null, "");
        assertEquals(200, baseline.statusCode());

        assertEquals(500, forcedFail.statusCode());

        int afterActive = extractActiveVersion(after.body());
        assertEquals(beforeActive, afterActive, "Rollback must restore previous active version");

    @Test
    void tc16_restore_concurrentConflict_oneGets409() throws Exception {
        // Use a barrier to align threads; use X-Force-Fail to keep lock held briefly.
        CyclicBarrier barrier = new CyclicBarrier(2);
        ExecutorService es = Executors.newFixedThreadPool(2);

        Callable<Integer> first = () -> {
            barrier.await(2, TimeUnit.SECONDS);
            HttpResponse<String> r = send("POST", "/api/apps/app-1/versions/2/restore", "admin-token",
                    List.of(new Header("X-Force-Fail", "true")), "");
            return r.statusCode();
        };
        Callable<Integer> second = () -> {
            barrier.await(2, TimeUnit.SECONDS);
            // no force-fail; should typically hit lock contention and return 409
            HttpResponse<String> r = send("POST", "/api/apps/app-1/versions/3/restore", "admin-token", null, "");
            return r.statusCode();
        };

        List<Future<Integer>> futures = es.invokeAll(List.of(first, second));
        es.shutdown();
        es.awaitTermination(3, TimeUnit.SECONDS);

        int s1 = futures.get(0).get();
        int s2 = futures.get(1).get();

        assertTrue((s1 == 409 || s2 == 409), "Expected at least one 409 conflict due to per-app lock");
    }

    private static HttpResponse<String> send(String method,
                                            String path,
                                            String bearerToken,
                                            List<Header> extraHeaders,
                                            String body) throws IOException, InterruptedException {
        HttpRequest.Builder b = HttpRequest.newBuilder(URI.create(baseUrl + path))
                .timeout(Duration.ofSeconds(4));

        if (bearerToken != null) {
            b.header("Authorization", "Bearer " + bearerToken);
        }

        if (extraHeaders != null) {
            for (Header h : extraHeaders) {
                b.header(h.name, h.value);
            }
        }

        if (body != null) {
            b.header("Content-Type", "application/json");
        }

        switch (method) {
            case "GET" -> b.GET();
            case "POST" -> b.POST(HttpRequest.BodyPublishers.ofString(body == null ? "" : body));
            case "PUT" -> b.PUT(HttpRequest.BodyPublishers.ofString(body == null ? "" : body));
            default -> throw new IllegalArgumentException("Unsupported method in test: " + method);
        }

        return HTTP.send(b.build(), HttpResponse.BodyHandlers.ofString());
    }

    private static int extractActiveVersion(String json) {
        // Minimal extraction to avoid adding JSON libraries: find "activeVersion":<int>
        String needle = "\"activeVersion\":";
        int idx = json.indexOf(needle);
        if (idx < 0) return -1;
        int start = idx + needle.length();
        int end = start;
        while (end < json.length() && Character.isDigit(json.charAt(end))) end++;
        return Integer.parseInt(json.substring(start, end));
    }

    private static int findFreePort() throws IOException {
        try (ServerSocket socket = new ServerSocket(0)) {
            socket.setReuseAddress(true);
            return socket.getLocalPort();
        }
    }

    private record Header(String name, String value) {}
}
