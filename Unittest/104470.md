    @Test

        when(appRepository.findById(appId)).thenReturn(Optional.of(app));

        var v2 = new AppVersionHistoryBackend.AppVersionRow(UUID.randomUUID(), appId, 2, "{cfg:2}", "bob", OffsetDateTime.now(), false, true);
