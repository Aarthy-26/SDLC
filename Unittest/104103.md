// File: src/test/java/com/dpai/aava/AavaIntegrationUnitTests104103Test.java
package com.dpai.aava;

import com.dpai.aava.config.AavaDataSourceProperties;
import com.dpai.aava.config.AavaDataSourceProperties.AuthMode;
import com.dpai.aava.config.DataSourceConfiguration;
import com.dpai.aava.integration.AavaDataSourceFactory;
import com.dpai.aava.security.SecretNotFoundException;
import com.dpai.aava.security.SecretProvider;
import com.dpai.api.DbHealthController;
import com.zaxxer.hikari.HikariDataSource;
import jakarta.servlet.http.HttpServletRequest;
import org.junit.jupiter.api.AfterEach;
import org.junit.jupiter.api.Test;
import org.mockito.ArgumentCaptor;
import org.mockito.MockedStatic;

import javax.sql.DataSource;
import jakarta.validation.ConstraintViolation;
import jakarta.validation.Validation;
import jakarta.validation.Validator;
import jakarta.validation.ValidatorFactory;

import java.sql.Connection;
import java.sql.PreparedStatement;

import java.sql.SQLException;
import java.util.Map;

import static org.assertj.core.api.Assertions.*;
import static org.mockito.Mockito.*;

/**
 * Unit test inventory implementation for User Story 104103.
 *
 * Security/enterprise notes:
 * - No real DB connections are made; JDBC interfaces are mocked.
 * - Secrets are never read from or written to logs/test output.
 */
class AavaIntegrationUnitTests104103Test {

    private HikariDataSource toClose;

    @AfterEach
    void tearDown() {
        if (toClose != null) {
            toClose.close();
            toClose = null;
        }
    }

    private static AavaDataSourceProperties props(AuthMode mode) {
        AavaDataSourceProperties p = new AavaDataSourceProperties();
        p.setJdbcUrl("jdbc:h2:mem:testdb;MODE=PostgreSQL;DB_CLOSE_DELAY=-1");
        p.setUsername("user1");
        p.setAuthMode(mode);

        AavaDataSourceProperties.Pool pool = new AavaDataSourceProperties.Pool();
        pool.setMaxPoolSize(7);
        pool.setMinIdle(2);
        pool.setConnectionTimeoutMs(31_000);
        pool.setIdleTimeoutMs(601_000);
        pool.setMaxLifetimeMs(1_801_000);
        p.setPool(pool);
        return p;
    }

    // 1
    @Test
    void createDataSource_basicAuth_usesPasswordEnvVarAndCreatesHikari() {
        AavaDataSourceProperties p = props(AuthMode.BASIC);
        SecretProvider secrets = mock(SecretProvider.class);
        when(secrets.requiredEnv("AAVA_DB_PASSWORD")).thenReturn("secret");

        DataSource ds = AavaDataSourceFactory.create(p, secrets);
        assertThat(ds).isInstanceOf(HikariDataSource.class);

        verify(secrets).requiredEnv("AAVA_DB_PASSWORD");
        verify(secrets, never()).requiredEnv("AAVA_DB_IAM_TOKEN");

        HikariDataSource h = (HikariDataSource) ds;
        toClose = h;
        assertThat(h.getJdbcUrl()).isEqualTo(p.getJdbcUrl());
        assertThat(h.getUsername()).isEqualTo(p.getUsername());
        assertThat(h.getPassword()).isEqualTo("secret");
        assertThat(h.getPoolName()).isEqualTo("dpai-aava-pool");
    }

    // 2
    @Test
    void createDataSource_iamAuth_usesIamTokenEnvVar() {
        AavaDataSourceProperties p = props(AuthMode.IAM);
        SecretProvider secrets = mock(SecretProvider.class);
        when(secrets.requiredEnv("AAVA_DB_IAM_TOKEN")).thenReturn("token");

        DataSource ds = AavaDataSourceFactory.create(p, secrets);
        assertThat(ds).isInstanceOf(HikariDataSource.class);

        verify(secrets).requiredEnv("AAVA_DB_IAM_TOKEN");
        verify(secrets, never()).requiredEnv("AAVA_DB_PASSWORD");

        toClose = (HikariDataSource) ds;
        assertThat(toClose.getPassword()).isEqualTo("token");
    }

    // 3
    @Test
    void createDataSource_rejectsNullProps() {
        SecretProvider secrets = mock(SecretProvider.class);
        assertThatThrownBy(() -> AavaDataSourceFactory.create(null, secrets))
                .isInstanceOf(NullPointerException.class)
                .hasMessageContaining("props");
    }

    // 4
    @Test
    void createDataSource_rejectsNullSecrets() {
        AavaDataSourceProperties p = props(AuthMode.BASIC);
        assertThatThrownBy(() -> AavaDataSourceFactory.create(p, null))
                .isInstanceOf(NullPointerException.class)
                .hasMessageContaining("secrets");
    }

    // 5
    @Test
    void createDataSource_basicAuth_missingSecretFailsFast() {
        AavaDataSourceProperties p = props(AuthMode.BASIC);
        SecretProvider secrets = mock(SecretProvider.class);
        when(secrets.requiredEnv("AAVA_DB_PASSWORD"))
                .thenThrow(new SecretNotFoundException("Missing required environment variable: AAVA_DB_PASSWORD"));

        assertThatThrownBy(() -> AavaDataSourceFactory.create(p, secrets))
                .isInstanceOf(SecretNotFoundException.class)
                .hasMessageContaining("AAVA_DB_PASSWORD")
                .doesNotHaveToString("secret");
    }

    // 6
    @Test
    void createDataSource_iamAuth_missingTokenFailsFast() {
        AavaDataSourceProperties p = props(AuthMode.IAM);
        SecretProvider secrets = mock(SecretProvider.class);
        when(secrets.requiredEnv("AAVA_DB_IAM_TOKEN"))
                .thenThrow(new SecretNotFoundException("Missing required environment variable: AAVA_DB_IAM_TOKEN"));

        assertThatThrownBy(() -> AavaDataSourceFactory.create(p, secrets))
                .isInstanceOf(SecretNotFoundException.class)
                .hasMessageContaining("AAVA_DB_IAM_TOKEN");
    }

    // 7, 8, 9
    @Test
    void createDataSource_propagatesPoolPropertiesAndSetsSecurityDefaults() {
        AavaDataSourceProperties p = props(AuthMode.BASIC);
        SecretProvider secrets = mock(SecretProvider.class);
        when(secrets.requiredEnv("AAVA_DB_PASSWORD")).thenReturn("secret");

        HikariDataSource h = (HikariDataSource) AavaDataSourceFactory.create(p, secrets);
        toClose = h;

        assertThat(h.getMaximumPoolSize()).isEqualTo(7);
        assertThat(h.getMinimumIdle()).isEqualTo(2);
        assertThat(h.getConnectionTimeout()).isEqualTo(31_000);
        assertThat(h.getIdleTimeout()).isEqualTo(601_000);
        assertThat(h.getMaxLifetime()).isEqualTo(1_801_000);

        assertThat(h.isAutoCommit()).isTrue();
        assertThat(h.getPoolName()).isEqualTo("dpai-aava-pool");
        assertThat(h.getHikariConfigMXBean().getInitializationFailTimeout()).isEqualTo(10_000);
    }

    // 10
    @Test
    void envSecretProvider_requiredEnv_rejectsNullName() {
        SecretProvider sp = SecretProvider.env();
        assertThatThrownBy(() -> sp.requiredEnv(null))
                .isInstanceOf(NullPointerException.class)
                .hasMessageContaining("name");
    }

    // 11
    @Test
    void envSecretProvider_requiredEnv_blankNameThrowsSecretNotFound() {
        // Note: implementation treats blank env var name as missing lookup.
        SecretProvider sp = SecretProvider.env();
        assertThatThrownBy(() -> sp.requiredEnv(" "))
                .isInstanceOf(SecretNotFoundException.class);
    }

    // 12
    @Test
    void dbHealth_returns200AndDbConnectedTrue_whenQueryReturnsRow() throws Exception {
        DataSource ds = mock(DataSource.class);
        Connection c = mock(Connection.class);
        PreparedStatement ps = mock(PreparedStatement.class);
        ResultSet rs = mock(ResultSet.class);

        when(ds.getConnection()).thenReturn(c);
        when(c.prepareStatement("SELECT 1")).thenReturn(ps);
        when(ps.executeQuery()).thenReturn(rs);
        when(rs.next()).thenReturn(true);

        HttpServletRequest req = mock(HttpServletRequest.class);
        when(req.getRequestURI()).thenReturn("/api/db/health");

        DbHealthController controller = new DbHealthController(ds);
        var response = controller.health(req);

        assertThat(response.getStatusCodeValue()).isEqualTo(200);
        Map<String, Object> body = response.getBody();
        assertThat(body).containsEntry("path", "/api/db/health");
        assertThat(body).containsKey("timestamp");
        assertThat(body).containsEntry("dbConnected", true);
        assertThat(body).doesNotContainKey("error");

        verify(ps).executeQuery();
        verify(rs).next();
        verify(ps).close();
        verify(c).close();
    }

    // 13
    @Test
    void dbHealth_returns200AndDbConnectedFalse_whenResultSetEmpty() throws Exception {
        DataSource ds = mock(DataSource.class);
        Connection c = mock(Connection.class);
        PreparedStatement ps = mock(PreparedStatement.class);
        ResultSet rs = mock(ResultSet.class);

        when(ds.getConnection()).thenReturn(c);
        when(c.prepareStatement("SELECT 1")).thenReturn(ps);
        when(ps.executeQuery()).thenReturn(rs);
        when(rs.next()).thenReturn(false);

        HttpServletRequest req = mock(HttpServletRequest.class);
        when(req.getRequestURI()).thenReturn("/api/db/health");

        DbHealthController controller = new DbHealthController(ds);
        var response = controller.health(req);

        assertThat(response.getStatusCodeValue()).isEqualTo(200);
        assertThat(response.getBody()).containsEntry("dbConnected", false);
    }

    // 14 + 15
    @Test
    void dbHealth_returns503WithGenericError_andDoesNotLeakDetails_onException() throws Exception {
        DataSource ds = mock(DataSource.class);
        when(ds.getConnection()).thenThrow(new SQLException("password=secret jdbc:postgresql://host/db"));

        HttpServletRequest req = mock(HttpServletRequest.class);
        when(req.getRequestURI()).thenReturn("/api/db/health");

        DbHealthController controller = new DbHealthController(ds);
        var response = controller.health(req);

        assertThat(response.getStatusCodeValue()).isEqualTo(503);
        Map<String, Object> body = response.getBody();
        assertThat(body).containsEntry("dbConnected", false);
        assertThat(body).containsEntry("error", "Database connectivity check failed");
        assertThat(body.toString()).doesNotContain("password=")
                .doesNotContain("jdbc:")
                .doesNotContain("secret");
    }

    @Test

        assertThat(sp).isNotNull();
        assertThat(sp.getClass().getName()).contains("EnvSecretProvider");
    }

    // 17
    @Test
    void dataSourceConfiguration_dataSource_delegatesToFactory() {
        DataSourceConfiguration cfg = new DataSourceConfiguration();
        AavaDataSourceProperties p = props(AuthMode.BASIC);
        SecretProvider sp = mock(SecretProvider.class);

        DataSource expected = mock(DataSource.class);
        try (MockedStatic<AavaDataSourceFactory> mocked = mockStatic(AavaDataSourceFactory.class)) {
            mocked.when(() -> AavaDataSourceFactory.create(p, sp)).thenReturn(expected);

            DataSource actual = cfg.dataSource(p, sp);
            assertThat(actual).isSameAs(expected);

            mocked.verify(() -> AavaDataSourceFactory.create(p, sp), times(1));
        }
    }

    private static Validator validator() {
        ValidatorFactory vf = Validation.buildDefaultValidatorFactory();
        return vf.getValidator();
    }

    // 20
    @Test
    void propertiesValidation_blankJdbcUrl_violatesNotBlank() {

        p.setJdbcUrl(" ");

        Set<ConstraintViolation<AavaDataSourceProperties>> v = validator().validate(p);
        assertThat(v).anyMatch(cv -> cv.getPropertyPath().toString().equals("jdbcUrl"));
    }
