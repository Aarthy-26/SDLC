/**
 * Unit tests for User Story 104103: DpaiAavaDataSourceApp and nested components.
 *
 * Notes (enterprise/security):

 * - No secrets are logged/asserted in test output (we assert redaction instead).

    @Test
    void dbConfigFrom_secretFromPropsPreferred_overEnv() {
        Properties p = baseProps();
        p.setProperty("db.secret", "propSecret");

        Map<String, String> env = new HashMap<>();
        env.put("DPAI_DB_PASSWORD", "envSecret");

        DbConfig cfg = DbConfig.from(p, env);
        assertEquals("propSecret", cfg.secret());
        assertEquals(15, cfg.loginTimeoutSeconds(), "default login timeout");
    }

    @Test
    void dbConfigFrom_secretFromEnv_usedWhenPropsMissing() {
        Properties p = baseProps();
        // db.secret omitted
        Map<String, String> env = Map.of("DPAI_DB_PASSWORD", "envSecret");

        DbConfig cfg = DbConfig.from(p, env);
        assertEquals("envSecret", cfg.secret());
    }

    @Test

    @Test

    @Test

    @Test
    void aavaLibraryBridge_cachesProvider_singleLoad_multipleCalls() {
        Logger log = Logger.getLogger("t");
        AavaLibraryBridge bridge = new AavaLibraryBridge(log);

        // Inject provider via reflection into providerRef so ServiceLoader is not relied upon.
        TestProvider provider = new TestProvider();
        setProviderRef(bridge, provider);

        DbConfig cfg = DbConfig.from(basePropsWithSecret("s"), Map.of());
        DataSource ds1 = bridge.getOrCreate(cfg);
        DataSource ds2 = bridge.getOrCreate(cfg);

        assertNotNull(ds1);
        assertNotNull(ds2);
        assertEquals(2, provider.createCalls.get(), "createDataSource called per getOrCreate");
        assertEquals(1, provider.nameCalls.get(), "provider name queried at least once and cached provider reused");
    }

    @Test
    void aavaLibraryBridge_wrapsProviderRuntimeException() {

        AavaLibraryBridge bridge = new AavaLibraryBridge(log);
        setProviderRef(bridge, new ThrowingProvider());

        DbConfig cfg = DbConfig.from(basePropsWithSecret("s"), Map.of());
        DpaiDataSourceException ex = assertThrows(DpaiDataSourceException.class, () -> bridge.getOrCreate(cfg));
        assertTrue(ex.getMessage().contains("Failed to create DataSource using provider"));

    @Test
    void smokeTester_wrapsSqlException_andDoesNotLeakSecrets() {
        Logger log = Logger.getLogger("t");
        SmokeTester tester = new SmokeTester(log);

        DataSource failing = new DataSource() {
            @Override public Connection getConnection() throws SQLException {
                throw new SQLException("Authentication failed for user u");
            }
            @Override public Connection getConnection(String username, String password) throws SQLException { throw new SQLException("x"); }
            @Override public <T> T unwrap(Class<T> iface) throws SQLException { throw new SQLException("x"); }
            @Override public boolean isWrapperFor(Class<?> iface) { return false; }
            @Override public PrintWriter getLogWriter() { return null; }
            @Override public void setLogWriter(PrintWriter out) { }
            @Override public void setLoginTimeout(int seconds) { }
            @Override public int getLoginTimeout() { return 0; }
            @Override public java.util.logging.Logger getParentLogger() { return Logger.getLogger("p"); }
        };

        DpaiDataSourceException ex = assertThrows(DpaiDataSourceException.class, () -> tester.testConnection(failing));
        assertTrue(ex.getMessage().contains("Smoke test failed. correlationId="));

        // Ensure the wrapper message does not include known secret token/password strings.
        assertFalse(ex.getMessage().contains("SUPER_SECRET"));
        assertFalse(ex.getMessage().toLowerCase(Locale.ROOT).contains("password"));
    }

    private static Properties baseProps() {
        Properties p = new Properties();
        p.setProperty("db.url", "jdbc:test");
        p.setProperty("db.username", "user");
        p.setProperty("db.authMode", "PASSWORD");
        p.setProperty("db.loginTimeoutSeconds", "15");
        // secret set in test variants
        return p;
    }

    private static Properties basePropsWithSecret(String secret) {
        Properties p = baseProps();
        p.setProperty("db.secret", secret);
        return p;
    }

    private static void setProviderRef(AavaLibraryBridge bridge, AavaDataSourceProvider provider) {
        try {
            java.lang.reflect.Field f = AavaLibraryBridge.class.getDeclaredField("providerRef");
            f.setAccessible(true);
            @SuppressWarnings("unchecked")
            java.util.concurrent.atomic.AtomicReference<AavaDataSourceProvider> ref =
                    (java.util.concurrent.atomic.AtomicReference<AavaDataSourceProvider>) f.get(bridge);
            ref.set(provider);
        } catch (Exception e) {
            throw new RuntimeException("Failed to set providerRef for test", e);
        }
    }

    private static final class TestProvider implements AavaDataSourceProvider {
        final AtomicInteger createCalls = new AtomicInteger();
        final AtomicInteger nameCalls = new AtomicInteger();

        @Override
        public DataSource createDataSource(DbConfig config, Logger log) {
            createCalls.incrementAndGet();
            return new DummyDataSource();
        }

        @Override
        public String name() {
            nameCalls.incrementAndGet();
            return "test-provider";
        }
    }

    private static final class ThrowingProvider implements AavaDataSourceProvider {
        @Override
        public DataSource createDataSource(DbConfig config, Logger log) {
            throw new IllegalStateException("boom");
        }

        @Override
        public String name() {
            return "throwing-provider";
        }
    }

    private static final class DummyDataSource implements DataSource {
        @Override public Connection getConnection() throws SQLException { throw new SQLException("not implemented"); }
        @Override public Connection getConnection(String username, String password) throws SQLException { throw new SQLException("not implemented"); }
        @Override public <T> T unwrap(Class<T> iface) throws SQLException { throw new SQLException("not a wrapper"); }
        @Override public boolean isWrapperFor(Class<?> iface) { return false; }
        @Override public PrintWriter getLogWriter() { return null; }
        @Override public void setLogWriter(PrintWriter out) { }
        @Override public void setLoginTimeout(int seconds) { }
        @Override public int getLoginTimeout() { return 0; }
        @Override public java.util.logging.Logger getParentLogger() throws SQLFeatureNotSupportedException {
            throw new SQLFeatureNotSupportedException("No parent logger");
        }
    }
