    @Test
    void tc14_bridge_usesInjectedProvider_andCaches() {
        Logger log = Logger.getLogger("test");
        AavaLibraryBridge bridge = new AavaLibraryBridge(log);

        AtomicInteger createCalls = new AtomicInteger();
        AavaDataSourceProvider provider = new AavaDataSourceProvider() {
            @Override
            public DataSource createDataSource(DbConfig config, Logger log) {
                createCalls.incrementAndGet();
                return new StubDataSource((Connection) java.lang.reflect.Proxy.newProxyInstance(
                        getClass().getClassLoader(), new Class[]{Connection.class}, (p, m, a) -> {
                            if (m.getName().equals("close")) return null;
                            if (m.getName().equals("getAutoCommit")) return true;
                            if (m.getName().equals("prepareStatement")) {
                                return java.lang.reflect.Proxy.newProxyInstance(
                                        getClass().getClassLoader(), new Class[]{PreparedStatement.class}, (pp, mm, aa) -> {
                                            if (mm.getName().equals("executeQuery")) {
                                                return java.lang.reflect.Proxy.newProxyInstance(
                                                        getClass().getClassLoader(), new Class[]{ResultSet.class}, (rp, rm, ra) -> {
                                                            if (rm.getName().equals("next")) return true;
                                                            if (rm.getName().equals("getInt")) return 1;
                                                            if (rm.getName().equals("close")) return null;
                                                            if (rm.getReturnType().isPrimitive()) return 0;
                                                            return null;
                                                        });
                                            }
                                            if (mm.getName().equals("close")) return null;
                                            if (mm.getReturnType().isPrimitive()) return 0;
                                            return null;
                                        });
                            }
                            if (m.getReturnType().isPrimitive()) return 0;
                            return null;
                        }));
            }

            @Override
            public String name() {
                return "test-provider";
            }
        };

        setBridgeProvider(bridge, provider);

        Properties p = baseProps();
        p.setProperty("db.secret", "x");
        DbConfig cfg = DbConfig.from(p, envWithPassword(null));

        DataSource ds1 = bridge.getOrCreate(cfg);
        DataSource ds2 = bridge.getOrCreate(cfg);

        assertNotNull(ds1);
        assertNotNull(ds2);
        assertEquals(2, createCalls.get(), "Provider createDataSource should be called per getOrCreate call");
    }

                            PreparedStatement ps = (PreparedStatement) java.lang.reflect.Proxy.newProxyInstance(
                                    getClass().getClassLoader(), new Class[]{PreparedStatement.class}, (pp, mm, aa) -> {
                                        switch (mm.getName()) {

                                                return (ResultSet) java.lang.reflect.Proxy.newProxyInstance(
                                                        getClass().getClassLoader(), new Class[]{ResultSet.class}, (rp, rm, ra) -> {
                                                            return switch (rm.getName()) {
                                                                case "next" -> true;

                                            }
                                            case "close" -> { return null; }
                                            default -> { return mm.getReturnType().isPrimitive() ? 0 : null; }
                                        }
                                    });
                            return ps;
                        }
                        case "close" -> { return null; }
                        default -> { return m.getReturnType().isPrimitive() ? 0 : null; }
                    }
                });

        DataSource ds = new StubDataSource(conn);

        assertDoesNotThrow(() -> tester.testConnection(ds));
        assertEquals(1, prepareCalls.get());
