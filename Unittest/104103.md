    private static DpaiDataSourceProperties props(String name, String url, DpaiDataSourceProperties.AuthMode mode,
                                                  String username, String password, Map<String, String> extra) {
        DpaiDataSourceProperties p = new DpaiDataSourceProperties();

    @Test
    void tc01_toAavaConfig_basic_includesUsernamePasswordAndProperties() {
        Map<String, String> extra = new LinkedHashMap<>();
        extra.put("ssl", "true");
        extra.put("connectTimeoutSeconds", "15");

        DpaiDataSourceProperties p = props("primary", "jdbc:postgresql://x/y",
                DpaiDataSourceProperties.AuthMode.BASIC, "user1", "secret", extra);

        Map<String, Object> cfg = AavaPropertyMapper.toAavaConfig(p);

        assertEquals("jdbc:postgresql://x/y", cfg.get("jdbcUrl"));
        assertEquals("BASIC", cfg.get("authMode"));
        assertEquals("user1", cfg.get("username"));
        assertEquals("secret", cfg.get("password"));

        assertTrue(cfg.containsKey("properties"));
        assertEquals(extra, cfg.get("properties"));
    }

    @Test

        Map<String, Object> cfg = AavaPropertyMapper.toAavaConfig(p);
        assertFalse(cfg.containsKey("password"));

        extra.put("ssl", "true");

        DpaiDataSourceProperties p = props("primary", "jdbc:postgresql://x/y",
                DpaiDataSourceProperties.AuthMode.BASIC, "user1", "secret", extra);

        DataSourceConfiguration cfg = new DataSourceConfiguration(p);

        DataSource expected = mock(DataSource.class);
        com.ascendion.aava.dpai.aava.AavaDataSourceProvider provider = mock(com.ascendion.aava.dpai.aava.AavaDataSourceProvider.class);
        when(provider.getDataSource(same(p))).thenReturn(expected);

    private static void invokeValidateAuthMode(DataSourceConfiguration cfg) throws Exception {
        Method m = DataSourceConfiguration.class.getDeclaredMethod("validateAuthMode");
        m.setAccessible(true);
        try {
            m.invoke(cfg);
        } catch (java.lang.reflect.InvocationTargetException ite) {
            Throwable cause = ite.getTargetException();
            if (cause instanceof RuntimeException re) {
                throw re;
            }
            if (cause instanceof Error err) {
                throw err;
            }
            throw ite;
        }
    }

    @Test
    void tc23_health_happyPath_returnsExpectedKeys() throws Exception {
        DataSource ds = mock(DataSource.class);
        Connection c = mock(Connection.class);
        DatabaseMetaData meta = mock(DatabaseMetaData.class);

        when(ds.getConnection()).thenReturn(c);
        when(c.getMetaData()).thenReturn(meta);
        when(meta.getDatabaseProductName()).thenReturn("PostgreSQL");
        when(meta.getDatabaseProductVersion()).thenReturn("16");
        when(meta.getURL()).thenReturn("jdbc:postgresql://localhost:5432/app");
        when(meta.getUserName()).thenReturn("app_user");

        DataSourceHealthController controller = new DataSourceHealthController(ds);
        Map<String, Object> out = controller.health();

        assertNotNull(out.get("timestamp"));
        assertEquals("UP", out.get("status"));
        assertEquals("PostgreSQL", out.get("databaseProduct"));
        assertEquals("16", out.get("databaseVersion"));
        assertEquals("jdbc:postgresql://localhost:5432/app", out.get("jdbcUrl"));
        assertEquals("app_user", out.get("user"));

        verify(c, times(1)).close();
    }
