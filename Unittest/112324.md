import org.junit.jupiter.api.Test;

import java.io.ByteArrayOutputStream;
import java.io.PrintStream;
import java.lang.reflect.Constructor;
import java.lang.reflect.InvocationTargetException;
import java.lang.reflect.Method;

import static org.junit.jupiter.api.Assertions.*;

/**
 * Unit tests for DataStudioDecommissionValidator (Story 112324).
 *
 * Notes:
 * - The production class uses private nested records and private methods.
 *   Tests use reflection to keep production code unchanged.
 * - Tests avoid logging/printing sensitive data and keep assertions deterministic.
 */
class DataStudioDecommissionValidatorTest {

    // ---------- Reflection helpers ----------

    private static Class<?> statusClass() throws Exception {
        return Class.forName("DataStudioDecommissionValidator$DecommissionStatus");
    }

    private static Class<?> resultClass() throws Exception {
        return Class.forName("DataStudioDecommissionValidator$ValidationResult");
    }

    private static Object newStatus(boolean backend,
                                    boolean ui,
                                    boolean deps,
                                    boolean monitoring,
                                    boolean docs,
                                    boolean post) throws Exception {
        Class<?> cls = statusClass();
        // record canonical ctor with components in declared order
        Constructor<?> ctor = cls.getDeclaredConstructor(boolean.class, boolean.class, boolean.class,
                boolean.class, boolean.class, boolean.class);
        ctor.setAccessible(true);
        return ctor.newInstance(backend, ui, deps, monitoring, docs, post);
    }

    private static Object invokeFromArgs(String... args) throws Exception {
        Method m = statusClass().getDeclaredMethod("fromArgs", String[].class);
        m.setAccessible(true);
        return m.invoke(null, (Object) args);
    }

    private static boolean getFlag(Object status, String accessor) throws Exception {
        Method m = status.getClass().getDeclaredMethod(accessor);
        m.setAccessible(true);
        return (boolean) m.invoke(status);
    }

    private static Object invokeValidate(Object status) throws Exception {
        Method m = DataStudioDecommissionValidator.class.getDeclaredMethod("validate", statusClass());
        m.setAccessible(true);
        return m.invoke(null, status);
    }

    private static String invokeFormat(Object result) throws Exception {
        Method m = resultClass().getDeclaredMethod("format");
        m.setAccessible(true);
        return (String) m.invoke(result);
    }

    private static boolean getSuccess(Object result) throws Exception {
        Method m = result.getClass().getDeclaredMethod("success");
        m.setAccessible(true);
        return (boolean) m.invoke(result);
    }

    private static String getDetails(Object result) throws Exception {
        Method m = result.getClass().getDeclaredMethod("details");
        m.setAccessible(true);
        return (String) m.invoke(result);
    }

    private static boolean invokeAppendCheck(StringBuilder sb, String label, boolean passed) throws Exception {
        Method m = DataStudioDecommissionValidator.class.getDeclaredMethod("appendCheck", StringBuilder.class, String.class, boolean.class);
        m.setAccessible(true);
        return (boolean) m.invoke(null, sb, label, passed);
    }

    // ---------- Tests ----------

    @Test
    void validate_allTrue_successAndAllPassLines() throws Exception {
        Object status = newStatus(true, true, true, true, true, true);
        Object result = invokeValidate(status);

        assertTrue(getSuccess(result));

        String formatted = invokeFormat(result);
        assertTrue(formatted.contains("Decommission Validation Result: SUCCESS"));
        assertEquals(6, formatted.lines().filter(l -> l.startsWith("[PASS] ")).count());
        assertEquals(0, formatted.lines().filter(l -> l.startsWith("[FAIL] ")).count());
    }

    @Test
    void validate_oneFalse_failsOverallAndContainsFailLine() throws Exception {
        Object status = newStatus(true, false, true, true, true, true);
        Object result = invokeValidate(status);

        assertFalse(getSuccess(result));

        String formatted = invokeFormat(result);
        assertTrue(formatted.contains("Decommission Validation Result: FAILED"));
        assertTrue(formatted.contains("[FAIL] dqg-ui service is successfully decommissioned"));
        assertEquals(5, formatted.lines().filter(l -> l.startsWith("[PASS] ")).count());
        assertEquals(1, formatted.lines().filter(l -> l.startsWith("[FAIL] ")).count());
    }

    @Test
    void validate_allFalse_allFailLines() throws Exception {
        Object status = newStatus(false, false, false, false, false, false);
        Object result = invokeValidate(status);

        assertFalse(getSuccess(result));
        String formatted = invokeFormat(result);
        assertEquals(0, formatted.lines().filter(l -> l.startsWith("[PASS] ")).count());
        assertEquals(6, formatted.lines().filter(l -> l.startsWith("[FAIL] ")).count());
    }

    @Test
    void main_noArgs_printsUsageOnly() {
        PrintStream origOut = System.out;
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        System.setOut(new PrintStream(baos));
        try {
            DataStudioDecommissionValidator.main(new String[0]);
        } finally {
            System.setOut(origOut);
        }

        String out = baos.toString();
        assertTrue(out.contains("Story 112324:"));
        assertTrue(out.contains("Usage:"));
        assertFalse(out.contains("Decommission Validation Result:"));
    }

    @Test
    void fromArgs_setsProvidedKeysAndDefaultsOthersFalse() throws Exception {
        Object status = invokeFromArgs(
                "service:dqg-backend=true",
                "dependenciesCleared=true"
        );

        assertTrue(getFlag(status, "dqgBackendDecommissioned"));
        assertFalse(getFlag(status, "dqgUiDecommissioned"));
        assertTrue(getFlag(status, "dependenciesCleared"));
        assertFalse(getFlag(status, "monitoringRemoved"));
        assertFalse(getFlag(status, "documentationUpdated"));
        assertFalse(getFlag(status, "postValidationPassed"));
    }

    @Test
    void fromArgs_duplicateKey_lastWins() throws Exception {
        Object status = invokeFromArgs(
                "service:dqg-ui=true",
                "service:dqg-ui=false"
        );
        assertFalse(getFlag(status, "dqgUiDecommissioned"));
    }

    @Test
    void fromArgs_unknownKey_ignored() throws Exception {
        Object status = invokeFromArgs(
                "unknownKey=true",
                "service:dqg-backend=true"
        );
        assertTrue(getFlag(status, "dqgBackendDecommissioned"));
        assertFalse(getFlag(status, "dqgUiDecommissioned"));
    }

    @Test
    void fromArgs_malformedTokenIgnored_noThrow() throws Exception {
        Object status = invokeFromArgs(
                "service:dqg-backend=true",
                "not-a-pair",
                "dependenciesCleared=true"
        );
        assertTrue(getFlag(status, "dqgBackendDecommissioned"));
        assertTrue(getFlag(status, "dependenciesCleared"));
    }

    @Test
    void fromArgs_emptyValue_treatedAsFalse() throws Exception {
        Object status = invokeFromArgs("service:dqg-ui=");
        assertFalse(getFlag(status, "dqgUiDecommissioned"));
    }

    @Test
    void fromArgs_nonBooleanValue_treatedAsFalse() throws Exception {
        Object status = invokeFromArgs("monitoringRemoved=YES");
        assertFalse(getFlag(status, "monitoringRemoved"));
    }

    @Test
    void fromArgs_trimsWhitespaceAroundKeyAndValue() throws Exception {
        Object status = invokeFromArgs("  service:dqg-ui = true  ");
        assertTrue(getFlag(status, "dqgUiDecommissioned"));
    }

    @Test
    void fromArgs_nullArray_throwsNpe() throws Exception {
        Method m = statusClass().getDeclaredMethod("fromArgs", String[].class);
        m.setAccessible(true);

        InvocationTargetException ex = assertThrows(InvocationTargetException.class, () -> m.invoke(null, (Object) null));
        assertTrue(ex.getCause() instanceof NullPointerException);
    }

    @Test
    void format_containsHeaderAndDetailsNoTrailingBlankLine() throws Exception {
        Object status = newStatus(true, true, true, true, true, true);
        Object result = invokeValidate(status);

        String formatted = invokeFormat(result);
        assertTrue(formatted.startsWith("Decommission Validation Result:"));
        assertFalse(formatted.endsWith(System.lineSeparator() + System.lineSeparator()));
    }

    @Test
    void details_ordering_isStable() throws Exception {
        Object status = newStatus(false, false, false, false, false, false);
        Object result = invokeValidate(status);

        String details = getDetails(result);
        String[] lines = details.split(System.lineSeparator());
        assertEquals(6, lines.length);
        assertTrue(lines[0].contains("dqg-backend"));
        assertTrue(lines[1].contains("dqg-ui"));
        assertTrue(lines[2].contains("dependencies"));
        assertTrue(lines[3].toLowerCase().contains("monitoring"));
        assertTrue(lines[4].toLowerCase().contains("documentation"));
        assertTrue(lines[5].toLowerCase().contains("post-decommission"));
    }

    @Test
    void appendCheck_returnsSameBooleanAndWritesPrefix() throws Exception {
        StringBuilder sb = new StringBuilder();

        boolean r1 = invokeAppendCheck(sb, "label1", true);
        boolean r2 = invokeAppendCheck(sb, "label2", false);

        assertTrue(r1);
        assertFalse(r2);

        String out = sb.toString();
        assertTrue(out.contains("[PASS] label1"));
        assertTrue(out.contains("[FAIL] label2"));
    }

    @Test
    void main_partialArgs_resultsInFailedAndMixedLines() {
        PrintStream origOut = System.out;
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        System.setOut(new PrintStream(baos));
        try {
            DataStudioDecommissionValidator.main(new String[]{"service:dqg-backend=true"});
        } finally {
            System.setOut(origOut);
        }

        String out = baos.toString();
        assertTrue(out.contains("Decommission Validation Result: FAILED"));
        assertTrue(out.contains("[PASS] dqg-backend service is successfully decommissioned"));
        assertTrue(out.contains("[FAIL] dqg-ui service is successfully decommissioned"));
    }

    @Test
    void main_acceptsUnknownAndMalformedArgs_withoutThrow() {
        PrintStream origOut = System.out;
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        System.setOut(new PrintStream(baos));
        try {
            DataStudioDecommissionValidator.main(new String[]{"unknown=true", "malformed", "dependenciesCleared=true"});
        } finally {
            System.setOut(origOut);
        }

        String out = baos.toString();
        assertTrue(out.contains("Decommission Validation Result:"));
        // since only deps is true, overall must fail
        assertTrue(out.contains("FAILED"));
        assertTrue(out.contains("[PASS] No active dependencies or integrations are impacted"));
    }

    @Test
    void output_doesNotEchoRawArgs() {
        // Attempt to inject a newline and a fake PASS line via an unknown key.
        // The program should not include raw args in output.
        String injected = "unknownKey=\\n[PASS] injected";

        PrintStream origOut = System.out;
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        System.setOut(new PrintStream(baos));
        try {
            DataStudioDecommissionValidator.main(new String[]{injected});
        } finally {
            System.setOut(origOut);
        }

        String out = baos.toString();
        assertFalse(out.contains("injected"));
    }
}
