    private static Object invokeEvaluate(Object request) {
        try {
            Method m = DataStudioServiceDecommissionTool.class.getDeclaredMethod(
                    "evaluate",
                    DataStudioServiceDecommissionTool.Request.class
            );
            m.setAccessible(true);
            return m.invoke(null, request);
        } catch (NoSuchMethodException | IllegalAccessException e) {
            throw new AssertionError("Failed to access evaluate() via reflection", e);
        } catch (InvocationTargetException e) {
            // Unwrap
            Throwable t = e.getTargetException();
            if (t instanceof RuntimeException re) throw re;
            throw new AssertionError("evaluate() threw unexpected exception", t);
        }
    }

    private static boolean boolField(Object obj, String fieldName) {
        try {
            Field f = obj.getClass().getDeclaredField(fieldName);
            f.setAccessible(true);
            return f.getBoolean(obj);
        } catch (ReflectiveOperationException e) {
            throw new AssertionError("Missing/invalid boolean field: " + fieldName, e);
        }
    }

    @SuppressWarnings("unchecked")
    private static List<String> assumptions(Object result) {

        } catch (ReflectiveOperationException e) {
            throw new AssertionError("Missing/invalid assumptions field", e);
        }
    }

    private static String format(Object result) {
        try {
            Method m = result.getClass().getDeclaredMethod("format");
            m.setAccessible(true);
            return (String) m.invoke(result);
        } catch (ReflectiveOperationException e) {
            throw new AssertionError("Failed to invoke format()", e);
        }
    }
