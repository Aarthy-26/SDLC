package com.example.radiotable;

import com.example.radiotable.RadioButtonTableApp.SelectionListener;
import com.example.radiotable.RadioButtonTableApp.SelectionModel;
import com.example.radiotable.RadioButtonTableApp.TableRow;
import com.example.radiotable.RadioButtonTableApp.ValidationException;
import org.junit.jupiter.api.Test;

import java.util.ArrayList;
import java.util.List;
import java.util.Optional;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicReference;

import static org.junit.jupiter.api.Assertions.*;

/**
 * Unit tests for {@link RadioButtonTableApp} domain logic.
 *
 * Security/enterprise notes:
 * - Tests avoid logging untrusted input.
 * - No external IO/network.
 * - Deterministic assertions; no flaky timing dependencies.
 */
class RadioButtonTableAppTest {

    @Test
    void tableRow_validTrimsWhitespace() {
        TableRow row = new TableRow("  r1  ", "  Alpha  ");
        assertEquals("r1", row.id());
        assertEquals("Alpha", row.label());
    }

    @Test
    void tableRow_nullIdRejected() {
        IllegalArgumentException ex = assertThrows(IllegalArgumentException.class,
                () -> new TableRow(null, "Alpha"));
        assertTrue(ex.getMessage().contains("id must be non-blank"));
    }

    @Test
    void tableRow_blankIdRejected() {
        IllegalArgumentException ex = assertThrows(IllegalArgumentException.class,
                () -> new TableRow("   ", "Alpha"));
        assertTrue(ex.getMessage().contains("id must be non-blank"));
    }

    @Test
    void tableRow_nullLabelRejected() {
        IllegalArgumentException ex = assertThrows(IllegalArgumentException.class,
                () -> new TableRow("r1", null));
        assertTrue(ex.getMessage().contains("label must be non-blank"));
    }

    @Test
    void tableRow_blankLabelRejected() {
        IllegalArgumentException ex = assertThrows(IllegalArgumentException.class,
                () -> new TableRow("r1", "\t \n"));
        assertTrue(ex.getMessage().contains("label must be non-blank"));
    }

    @Test
    void radioInputId_allowedCharactersUnchanged() {
        TableRow row = new TableRow("AZaz09_-", "L");
        assertEquals("radio-AZaz09_-", row.radioInputId());
    }

    @Test
    void radioInputId_sanitizesDisallowedCharacters() {
        TableRow row = new TableRow("r 1!", "L");
        assertEquals("radio-r-1-", row.radioInputId());
    }

    @Test
    void radioInputId_allDisallowedCharactersStillNonEmptyAfterSanitize() {
        TableRow row = new TableRow("!!!", "L");
        assertEquals("radio---", row.radioInputId());
    }

    @Test
    void radioGroupName_isStable() {
        assertEquals("table-selection", new TableRow("r1", "A").radioGroupName());
        assertEquals("table-selection", new TableRow("r2", "B").radioGroupName());
    }

    @Test
    void selectionModel_nullRowsRejected() {
        IllegalArgumentException ex = assertThrows(IllegalArgumentException.class,
                () -> new SelectionModel(null));
        assertTrue(ex.getMessage().contains("rows must not be null"));
    }

    @Test
    void selectionModel_emptyRows_constructsAndNoSelection() {
        SelectionModel model = new SelectionModel(List.of());
        assertEquals(Optional.empty(), model.getSelectedRowId());
    }

    @Test
    void getSelectedRowId_initiallyEmpty() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        assertTrue(model.getSelectedRowId().isEmpty());
    }

    @Test
    void selectExistingRow_setsSelection() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A"), new TableRow("r2", "B")));
        model.selectRowById("r2");
        assertEquals(Optional.of("r2"), model.getSelectedRowId());
    }

    @Test
    void selectRowById_normalizesWhitespace() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        model.selectRowById("  r1  ");
        assertEquals(Optional.of("r1"), model.getSelectedRowId());
    }

    @Test
    void selectUnknownRow_throwsAndDoesNotChangeSelection() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        model.selectRowById("r1");

        ValidationException ex = assertThrows(ValidationException.class,
                () -> model.selectRowById("r999"));
        assertTrue(ex.getMessage().contains("Unknown rowId: r999"));

        assertEquals(Optional.of("r1"), model.getSelectedRowId());
    }

    @Test
    void selectNullRowId_throws() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        ValidationException ex = assertThrows(ValidationException.class,
                () -> model.selectRowById(null));
        assertTrue(ex.getMessage().contains("rowId must be provided"));
    }

    @Test
    void selectBlankRowId_throws() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        ValidationException ex = assertThrows(ValidationException.class,
                () -> model.selectRowById("   "));
        assertTrue(ex.getMessage().contains("rowId must be provided"));
    }

    @Test
    void selectSameRowId_isIdempotentAndDoesNotFireListener() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        AtomicInteger calls = new AtomicInteger(0);
        model.setSelectionListener(selected -> calls.incrementAndGet());

        model.selectRowById("r1");
        model.selectRowById("r1");
        model.selectRowById("  r1  ");

        assertEquals(Optional.of("r1"), model.getSelectedRowId());
        assertEquals(1, calls.get(), "Listener must be called only once for idempotent re-selection");
    }

    @Test
    void switchingSelection_updatesSelectedIdAndFiresListenerOncePerChange() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A"), new TableRow("r2", "B")));
        List<String> events = new ArrayList<>();
        model.setSelectionListener(events::add);

        model.selectRowById("r1");
        model.selectRowById("r2");

        assertEquals(Optional.of("r2"), model.getSelectedRowId());
        assertEquals(List.of("r1", "r2"), events);
    }

    @Test
    void clearSelection_idempotentWhenAlreadyEmpty() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        AtomicInteger calls = new AtomicInteger(0);
        model.setSelectionListener(selected -> calls.incrementAndGet());

        model.clearSelection();

        assertTrue(model.getSelectedRowId().isEmpty());
        assertEquals(0, calls.get());
    }

    @Test
    void clearSelection_afterSelectionFiresNullAndClears() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        AtomicReference<String> last = new AtomicReference<>("__unset__");
        model.setSelectionListener(last::set);

        model.selectRowById("r1");
        model.clearSelection();

        assertTrue(model.getSelectedRowId().isEmpty());
        assertNull(last.get(), "Listener must receive null on clear");
    }

    @Test
    void nullListener_isSafe() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        model.setSelectionListener(null);
        assertDoesNotThrow(() -> model.selectRowById("r1"));
        assertEquals(Optional.of("r1"), model.getSelectedRowId());
        assertDoesNotThrow(model::clearSelection);
        assertTrue(model.getSelectedRowId().isEmpty());
    }

    @Test
    void listenerNotCalledOnInvalidSelectionAttempt() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        AtomicInteger calls = new AtomicInteger(0);
        model.setSelectionListener(selected -> calls.incrementAndGet());

        assertThrows(ValidationException.class, () -> model.selectRowById("unknown"));
        assertEquals(0, calls.get());
    }

    @Test
    void defensiveCopy_mutatingOriginalListDoesNotAffectSelectionResolution() {
        List<TableRow> rows = new ArrayList<>();
        rows.add(new TableRow("r1", "A"));
        SelectionModel model = new SelectionModel(rows);

        // Mutate original list after model creation
        rows.add(new TableRow("r2", "B"));
        rows.clear();

        // Model should still know about original snapshot containing only r1
        model.selectRowById("r1");
        assertEquals(Optional.of("r1"), model.getSelectedRowId());
        assertThrows(ValidationException.class, () -> model.selectRowById("r2"));
    }

    @Test
    void volume_largeRowSet_selectionWorks() {
        int n = 10_000;
        List<TableRow> rows = new ArrayList<>(n);
        for (int i = 0; i < n; i++) {
            rows.add(new TableRow("r" + i, "L" + i));
        }
        SelectionModel model = new SelectionModel(rows);

        model.selectRowById("r9999");
        assertEquals(Optional.of("r9999"), model.getSelectedRowId());
    }

    @Test
    void volume_repeatedIdempotentSelectionDoesNotCauseRepeatedCallbacks() {
        SelectionModel model = new SelectionModel(List.of(new TableRow("r1", "A")));
        AtomicInteger calls = new AtomicInteger(0);
        model.setSelectionListener(selected -> calls.incrementAndGet());

        for (int i = 0; i < 1_000; i++) {
            model.selectRowById("r1");
        }

        assertEquals(Optional.of("r1"), model.getSelectedRowId());
        assertEquals(1, calls.get());
    }
}
