package com.ascendion.datapowerAI;

import org.junit.jupiter.api.Test;

import java.lang.reflect.Method;
import java.util.List;

import static org.junit.jupiter.api.Assertions.*;

/**
 * Unit tests for {@link ComprehensiveTestCoveragePlanner}.
 *
 * Enterprise notes:
 * - No external I/O, no reflection beyond invoking private methods for deterministic unit coverage.
 * - No secrets/PII.
 * - Uses JUnit 5.
 */
class ComprehensiveTestCoveragePlannerTest {

    @Test
    void tc01_generateReport_returnsReportWithCorrectStoryAndTitle() {
        var report = ComprehensiveTestCoveragePlanner.generateReport();
        assertNotNull(report);
        assertEquals("111425", report.storyId());
        assertEquals("Implement Comprehensive Unit Test Coverage for All Controller and Service Classes", report.title());
    }

    @Test
    void tc02_generateReport_populatesConventionsExpectedValues() {
        var report = ComprehensiveTestCoveragePlanner.generateReport();
        var c = report.conventions();
        assertNotNull(c);
        assertEquals("com.ascendion.datapowerAI", c.basePackage());
        assertEquals("ApiResponseGenric", c.responseWrapper());
        assertEquals("ApiResponseUtil", c.responseHelper());
        assertEquals("GlobalExceptionHandlerDpai", c.exceptionHandler());
    }

    @Test
    void tc03_conventions_authorizationPatternsContainExpectedEntries() {
        var c = ComprehensiveTestCoveragePlanner.generateReport().conventions();
        assertEquals(4, c.authorizationPatterns().size());
        assertEquals("@PreAuthorize(RoleType.TOOL_ADMIN)", c.authorizationPatterns().get(0));
        assertTrue(c.authorizationPatterns().contains("@PreAuthorize(RoleType.APP_EDITOR)"));
    }

    @Test
    void tc04_conventions_persistenceStyleAndHelpersPresent() {
        var c = ComprehensiveTestCoveragePlanner.generateReport().conventions();
        assertNotNull(c.persistenceStyle());
        assertFalse(c.persistenceStyle().isBlank());
        assertEquals(6, c.similarHelpers().size());
        assertTrue(c.similarHelpers().contains("DuplicateNameGenerator"));
    }

    @Test
    void tc05_inventory_controllerCountAndSampleMapping() {
        var inv = ComprehensiveTestCoveragePlanner.generateReport().inventory();
        assertEquals(37, inv.controllers().size());
        var first = inv.controllers().get(0);
        assertEquals("AavaAuthController", first.className());
        assertEquals("AavaAuthControllerTest", first.testClassName());
        assertEquals(ComprehensiveTestCoveragePlanner.Layer.CONTROLLER, first.layer());
    }

    @Test
    void tc06_inventory_serviceCountAndSampleMapping() {
        var inv = ComprehensiveTestCoveragePlanner.generateReport().inventory();
        assertEquals(78, inv.services().size());
        assertTrue(inv.services().stream().anyMatch(p -> p.className().equals("APIExecutionService") && p.testClassName().equals("APIExecutionServiceTest")));
        assertTrue(inv.services().stream().anyMatch(p -> p.className().equals("ReplaceAppImportStrategy") && p.testClassName().equals("ReplaceAppImportStrategyTest")));
    }

    @Test
    void tc07_allControllerPlans_haveTestSuffixAndCorrectLayer() {
        var inv = ComprehensiveTestCoveragePlanner.generateReport().inventory();
        for (var p : inv.controllers()) {
            assertEquals(p.className() + "Test", p.testClassName());
            assertEquals(ComprehensiveTestCoveragePlanner.Layer.CONTROLLER, p.layer());
        }
    }

    @Test
    void tc08_allServicePlans_haveTestSuffixAndCorrectLayer() {
        var inv = ComprehensiveTestCoveragePlanner.generateReport().inventory();
        for (var p : inv.services()) {
            assertEquals(p.className() + "Test", p.testClassName());
            assertEquals(ComprehensiveTestCoveragePlanner.Layer.SERVICE, p.layer());
        }
    }

    @Test
    void tc09_defaultControllerScenarios_matchContract_viaReflection() throws Exception {
        Method m = ComprehensiveTestCoveragePlanner.class.getDeclaredMethod("defaultScenarios", ComprehensiveTestCoveragePlanner.Layer.class);
        m.setAccessible(true);
        @SuppressWarnings("unchecked")
        List<String> scenarios = (List<String>) m.invoke(null, ComprehensiveTestCoveragePlanner.Layer.CONTROLLER);
        assertEquals(List.of("success", "validation", "error-response", "authorization", "boundary-input"), scenarios);
    }

    @Test
    void tc10_defaultServiceScenarios_matchContract_viaReflection() throws Exception {
        Method m = ComprehensiveTestCoveragePlanner.class.getDeclaredMethod("defaultScenarios", ComprehensiveTestCoveragePlanner.Layer.class);
        m.setAccessible(true);
        @SuppressWarnings("unchecked")
        List<String> scenarios = (List<String>) m.invoke(null, ComprehensiveTestCoveragePlanner.Layer.SERVICE);
        assertEquals(List.of("success", "rule-check", "dependency-failure", "not-found-or-empty", "exception-propagation"), scenarios);
    }

    @Test
    void tc11_completionTotals_matchInventorySizes() {
        var report = ComprehensiveTestCoveragePlanner.generateReport();
        assertEquals(report.inventory().controllers().size(), report.completionReport().totalControllersIdentified());
        assertEquals(report.inventory().services().size(), report.completionReport().totalServicesIdentified());
    }

    @Test
    void tc12_completion_controllerTestClassesProjectedCorrectly() {
        var report = ComprehensiveTestCoveragePlanner.generateReport();
        assertEquals(37, report.completionReport().controllerTestClasses().size());
        assertTrue(report.completionReport().controllerTestClasses().contains("WhitelistControllerTest"));
    }

    @Test
    void tc13_completion_serviceTestClassesProjectedCorrectly() {
        var report = ComprehensiveTestCoveragePlanner.generateReport();
        assertEquals(78, report.completionReport().serviceTestClasses().size());
        assertTrue(report.completionReport().serviceTestClasses().contains("ReplaceAppImportStrategyTest"));
    }

    @Test
    void tc14_completion_importantScenariosContainExpectedItems() {
        var imp = ComprehensiveTestCoveragePlanner.generateReport().completionReport().importantScenariosCovered();
        assertEquals(8, imp.size());
        assertTrue(imp.contains("Authorization and role-based access"));
        assertTrue(imp.contains("Import/export and multipart endpoint behavior"));
    }

    @Test
    void tc15_completion_metadataStrings_areNonBlankAndContainExpectedPhrases() {
        var cr = ComprehensiveTestCoveragePlanner.generateReport().completionReport();
        assertTrue(cr.dependenciesAddedOrModified().toLowerCase().contains("none"));
        assertTrue(cr.mavenTestExecutionResults().toLowerCase().contains("not executed"));
        assertTrue(cr.jacocoCoverage().toLowerCase().contains("not measured"));
        assertTrue(cr.remainingGaps().toLowerCase().contains("known remaining gap"));
    }

    @Test
    void tc16_inventoryLists_areImmutable() {
        var inv = ComprehensiveTestCoveragePlanner.generateReport().inventory();
        assertThrows(UnsupportedOperationException.class, () -> inv.controllers().add(null));
        assertThrows(UnsupportedOperationException.class, () -> inv.services().add(null));
    }

    @Test
    void tc17_conventionLists_areImmutable() {
        var c = ComprehensiveTestCoveragePlanner.generateReport().conventions();
        assertThrows(UnsupportedOperationException.class, () -> c.authorizationPatterns().add("x"));
        assertThrows(UnsupportedOperationException.class, () -> c.similarHelpers().add("x"));
    }

    @Test
    void tc18_recordComponents_nonNullWhereExpected() {
        var report = ComprehensiveTestCoveragePlanner.generateReport();
        assertNotNull(report.conventions().authorizationPatterns());
        assertNotNull(report.conventions().similarHelpers());
        assertNotNull(report.completionReport().importantScenariosCovered());
    }

    @Test
    void tc19_controllerOrdering_isDeterministic() {
        var controllers = ComprehensiveTestCoveragePlanner.generateReport().inventory().controllers();
        assertEquals("AavaAuthController", controllers.get(0).className());
        assertEquals("WhitelistController", controllers.get(controllers.size() - 1).className());
    }

    @Test
    void tc20_serviceOrdering_isDeterministic() {
        var services = ComprehensiveTestCoveragePlanner.generateReport().inventory().services();
        assertEquals("APIExecutionService", services.get(0).className());
        assertEquals("ReplaceAppImportStrategy", services.get(services.size() - 1).className());
    }
}
