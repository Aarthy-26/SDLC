package com.ascendion.datapowerAI;

import org.junit.jupiter.api.Test;

import java.util.List;

import static org.junit.jupiter.api.Assertions.*;

/**
 * Unit tests for {@link ComprehensiveUnitTestCoveragePlanner}.
 *
 * Notes (enterprise/compliance):
 * - No secrets, no network, deterministic tests.
 * - Tests validate business logic (counts, inventories) and template/string content.
 */
class ComprehensiveUnitTestCoveragePlannerTest {

    @Test
    void inventorySample_returnsNonNullLists() {
        ComprehensiveUnitTestCoveragePlanner.InventoryReport inv = ComprehensiveUnitTestCoveragePlanner.InventoryReport.sample();
        assertNotNull(inv.controllers);
        assertNotNull(inv.services);
        assertNotNull(inv.serviceImplementations);
        assertNotNull(inv.helpers);
        assertFalse(inv.controllers.isEmpty());
        assertFalse(inv.services.isEmpty());
        assertFalse(inv.serviceImplementations.isEmpty());
        assertFalse(inv.helpers.isEmpty());
    }

    @Test
    void inventorySample_containsKnownController() {
        var inv = ComprehensiveUnitTestCoveragePlanner.InventoryReport.sample();
        assertTrue(inv.controllers.contains("AppController"));
    }

    @Test
    void inventorySample_containsKnownService() {
        var inv = ComprehensiveUnitTestCoveragePlanner.InventoryReport.sample();
        assertTrue(inv.services.contains("AppService"));
    }

    @Test
    void inventorySample_containsKnownServiceImplementation() {
        var inv = ComprehensiveUnitTestCoveragePlanner.InventoryReport.sample();
        assertTrue(inv.serviceImplementations.contains("UserServiceImpl"));
    }

    @Test
    void inventorySample_containsKnownHelper() {
        var inv = ComprehensiveUnitTestCoveragePlanner.InventoryReport.sample();
        assertTrue(inv.helpers.contains("ApiResponseUtil"));
    }

    @Test
    void completionReportService_calculatesCountsCorrectly() {
        ComprehensiveUnitTestCoveragePlanner.CompletionReportService svc = new ComprehensiveUnitTestCoveragePlanner.CompletionReportService();
        ComprehensiveUnitTestCoveragePlanner.CompletionSummary summary = svc.summarize();

        int expectedControllerCount = ComprehensiveUnitTestCoveragePlanner.INVENTORY.controllers.size();
        int expectedServiceCount = ComprehensiveUnitTestCoveragePlanner.INVENTORY.services.size()
                + ComprehensiveUnitTestCoveragePlanner.INVENTORY.serviceImplementations.size();

        assertEquals(expectedControllerCount, summary.controllerClassesIdentified());
        assertEquals(expectedServiceCount, summary.serviceClassesIdentified());
        assertEquals(expectedControllerCount + expectedServiceCount, summary.plannedTestClasses());
    }

    @Test
    void completionReportService_exposesCommands() {
        ComprehensiveUnitTestCoveragePlanner.CompletionSummary summary =
                new ComprehensiveUnitTestCoveragePlanner.CompletionReportService().summarize();
        assertEquals("mvn test", summary.mavenCommand());
        assertEquals("mvn test jacoco:report", summary.jacocoCommand());
    }

    @Test
    void completionReportService_dependenciesContainExpected() {
        ComprehensiveUnitTestCoveragePlanner.CompletionSummary summary =
                new ComprehensiveUnitTestCoveragePlanner.CompletionReportService().summarize();
        assertTrue(summary.dependencies().containsAll(List.of("JUnit Jupiter", "Mockito", "MockMvc", "JaCoCo")));
    }

    @Test
