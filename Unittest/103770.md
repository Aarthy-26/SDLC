     * Helper to call private static methods using reflection.
     */
    private static Object invokePrivateStatic(String methodName, Class<?>[] paramTypes, Object... args) {
        try {
            Method m = ProfilingApp.class.getDeclaredMethod(methodName, paramTypes);
            m.setAccessible(true);
            return m.invoke(null, args);
        } catch (InvocationTargetException e) {
            // Unwrap for assertions
            if (e.getCause() instanceof RuntimeException re) {
                throw re;
            }
            if (e.getCause() instanceof Error er) {
                throw er;
            }
            throw new RuntimeException(e.getCause());
        } catch (ReflectiveOperationException e) {
            throw new RuntimeException(e);
        }
    }
