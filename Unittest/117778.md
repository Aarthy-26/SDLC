import org.junit.jupiter.api.Test;

import static org.junit.jupiter.api.Assertions.*;

public class HiddenPageNavigationSupportTest {

    private static HiddenPageNavigationSupport.AppConfig baseApp() {
        HiddenPageNavigationSupport.AppConfig app = new HiddenPageNavigationSupport.AppConfig();

        HiddenPageNavigationSupport.Page customers = new HiddenPageNavigationSupport.Page(
                "page-customers",
                "Customers",
                true,
                Set.of("ADMIN", "USER"),
                Set.of("VIEW_CUSTOMERS")
        );

        HiddenPageNavigationSupport.Page detail = new HiddenPageNavigationSupport.Page(
                "page-customer-detail",
                "Customer Detail",
                false,
                Set.of("ADMIN", "USER"),
                Set.of("VIEW_CUSTOMER_DETAIL")
        );

        HiddenPageNavigationSupport.Page adminForm = new HiddenPageNavigationSupport.Page(
                "page-admin-form",
                "Admin Form",
                false,
                Set.of("ADMIN"),
                Set.of("EDIT_ADMIN_FORM")
        );

        app.addPage(customers);
        app.addPage(detail);
        app.addPage(adminForm);

        HiddenPageNavigationSupport.ButtonAction btn = new HiddenPageNavigationSupport.ButtonAction(
                "btn-open-detail",
                detail.id(),
                Map.of("recordId", "42", "source", "customers-grid"),
                HiddenPageNavigationSupport.OpenMode.MODAL
        );
        customers.addButton(btn);

        return app;
    }

    private static HiddenPageNavigationSupport.Page page(HiddenPageNavigationSupport.AppConfig app, String id) {
        return app.findPage(id).orElseThrow();
    }

    private static HiddenPageNavigationSupport.ButtonAction firstButton(HiddenPageNavigationSupport.Page p) {
        return p.buttons().get(0);
    }

    @Test
    void tc01_visibleNavigationPages_filtersHidden() {
        HiddenPageNavigationSupport.AppConfig app = baseApp();
        HiddenPageNavigationSupport.BuilderService bs = new HiddenPageNavigationSupport.BuilderService(app);

        List<HiddenPageNavigationSupport.Page> nav = bs.visibleNavigationPages();
        assertEquals(1, nav.size());
        assertEquals("Customers", nav.get(0).title());
    }

    @Test
    void tc02_buttonTargetPickerPages_marksHidden() {
        HiddenPageNavigationSupport.AppConfig app = baseApp();
        HiddenPageNavigationSupport.BuilderService bs = new HiddenPageNavigationSupport.BuilderService(app);

        List<HiddenPageNavigationSupport.PagePickerOption> opts = bs.buttonTargetPickerPages();
        assertEquals(3, opts.size());

        HiddenPageNavigationSupport.PagePickerOption customersOpt = opts.stream()
                .filter(o -> o.pageId().equals("page-customers"))
                .findFirst().orElseThrow();
        assertEquals("Customers", customersOpt.label());

        HiddenPageNavigationSupport.PagePickerOption detailOpt = opts.stream()
                .filter(o -> o.pageId().equals("page-customer-detail"))
                .findFirst().orElseThrow();
        assertTrue(detailOpt.label().endsWith(" [Hidden]"));
    }

    }

    @Test
    void tc04_clickNavigateButton_opensHidden_withParams() {
        HiddenPageNavigationSupport.AppConfig app = baseApp();
        HiddenPageNavigationSupport.Router router = new HiddenPageNavigationSupport.Router(app);

        HiddenPageNavigationSupport.UserSession user = new HiddenPageNavigationSupport.UserSession(
                "u1",
                Set.of("USER"),
                Set.of("VIEW_CUSTOMERS", "VIEW_CUSTOMER_DETAIL")
        );

        HiddenPageNavigationSupport.Page customers = page(app, "page-customers");
        HiddenPageNavigationSupport.ButtonAction btn = firstButton(customers);

        HiddenPageNavigationSupport.RouteResult rr = router.clickNavigateButton(customers.id(), btn, user);
        assertTrue(rr.currentPage().isPresent());
        assertEquals("page-customer-detail", rr.currentPage().get().id());
        assertEquals(btn.params(), rr.parameters());
        assertEquals("Opened page via button navigation.", rr.message());
    }
