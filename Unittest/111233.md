package com.aava.datapowerai.service;

import com.aava.datapowerai.dto.common.ApiResponse;
import com.aava.datapowerai.model.UserModel;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import org.mockito.ArgumentCaptor;

import org.mockito.junit.jupiter.MockitoExtension;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.ResponseEntity;

import java.util.Optional;
import java.util.UUID;

import static org.junit.jupiter.api.Assertions.*;

import static org.mockito.Mockito.*;

/**
 * Unit tests for User Story 111233 - SSO resolve/provision.
 *
 * Enterprise/security notes:
 * - Tests avoid logging PII.
 * - Tests verify parameter binding behaviors and ensure no accidental persistence side effects.
 */
@ExtendWith(MockitoExtension.class)
class ImplementSsoUserProvisioningTest {

    @Mock
    private ImplementSsoUserProvisioning.SsoUserRepository repo;

    private ImplementSsoUserProvisioning.SsoUserProvisioningService service;

    private ImplementSsoUserProvisioning.SsoProvisioningController controller;

    @Captor
    private ArgumentCaptor<UUID> uuidCaptor;

    @Captor
    private ArgumentCaptor<String> usernameCaptor;

    @Captor
    private ArgumentCaptor<String> emailCaptor;

    @BeforeEach
    void setUp() {
        service = new ImplementSsoUserProvisioning.SsoUserProvisioningService(repo);
        controller = new ImplementSsoUserProvisioning.SsoProvisioningController(service);
    }

    private static UserModel user(UUID id, String username, String email) {
        UserModel u = new UserModel();
        u.setId(id);
        u.setUsername(username);
        u.setEmail(email);
        u.setEnabled(true);
        return u;
    }

    @Test
    void tc01_resolveExistingByUsername() {
        UUID id = UUID.randomUUID();
        UserModel existing = user(id, "alice", "alice@example.com");

        when(repo.findByUsername("alice")).thenReturn(Optional.of(existing));

        UserModel out = service.resolveOrProvisionUser("alice", "alice@example.com");

        assertSame(existing, out);
        verify(repo, times(1)).findByUsername("alice");
        verify(repo, never()).findByEmailIgnoreCase(anyString());
        verify(repo, never()).insertSsoUser(any(), anyString(), anyString());
    }

    @Test
    void tc02_fallbackToEmailIgnoreCase() {
        UUID id = UUID.randomUUID();
        UserModel byEmail = user(id, "manualUser", "Bob@Example.com");

        when(repo.findByUsername("bob")).thenReturn(Optional.empty());
        when(repo.findByEmailIgnoreCase("bob@example.com")).thenReturn(Optional.of(byEmail));

        UserModel out = service.resolveOrProvisionUser("bob", "bob@example.com");

        assertSame(byEmail, out);
        verify(repo).findByUsername("bob");
        verify(repo).findByEmailIgnoreCase("bob@example.com");
        verify(repo, never()).insertSsoUser(any(), anyString(), anyString());
        verify(repo, never()).findById(any());
    }

    @Test
    void tc03_createNewUser_whenNotFound() {
        when(repo.findByUsername("carol")).thenReturn(Optional.empty());
        when(repo.findByEmailIgnoreCase("carol@example.com")).thenReturn(Optional.empty());

        // Insert should happen; then findById should return created.
        when(repo.findById(any(UUID.class))).thenAnswer(inv -> {
            UUID id = inv.getArgument(0, UUID.class);
            return Optional.of(user(id, "carol", "carol@example.com"));
        });

        UserModel out = service.resolveOrProvisionUser("carol", "carol@example.com");

        assertNotNull(out.getId());
        assertEquals("carol", out.getUsername());
        assertEquals("carol@example.com", out.getEmail());

        verify(repo).findByUsername("carol");
        verify(repo).findByEmailIgnoreCase("carol@example.com");
        verify(repo, times(1)).insertSsoUser(any(UUID.class), eq("carol"), eq("carol@example.com"));
        verify(repo, times(1)).findById(any(UUID.class));
    }

    @Test
    void tc04_concurrency_duplicateInsert_thenReadByUsername() {
        UUID existingId = UUID.randomUUID();
        UserModel existing = user(existingId, "dave", "dave@example.com");

        when(repo.findByUsername("dave"))
                .thenReturn(Optional.empty())
                .thenReturn(Optional.of(existing));
        when(repo.findByEmailIgnoreCase("dave@example.com")).thenReturn(Optional.empty());
        doThrow(new DataIntegrityViolationException("duplicate")).when(repo)
                .insertSsoUser(any(UUID.class), eq("dave"), eq("dave@example.com"));

        UserModel out = service.resolveOrProvisionUser("dave", "dave@example.com");

        assertSame(existing, out);
        verify(repo, times(2)).findByUsername("dave");
        verify(repo, times(1)).insertSsoUser(any(UUID.class), eq("dave"), eq("dave@example.com"));
    }

    @Test
    void tc05_concurrency_duplicateInsert_andReadStillMissing_rethrows() {
        when(repo.findByUsername("erin"))
                .thenReturn(Optional.empty())
                .thenReturn(Optional.empty());
        when(repo.findByEmailIgnoreCase("erin@example.com")).thenReturn(Optional.empty());
        DataIntegrityViolationException ex = new DataIntegrityViolationException("duplicate");
        doThrow(ex).when(repo).insertSsoUser(any(UUID.class), eq("erin"), eq("erin@example.com"));

        DataIntegrityViolationException thrown = assertThrows(
                DataIntegrityViolationException.class,
                () -> service.resolveOrProvisionUser("erin", "erin@example.com")
        );

        assertSame(ex, thrown);
        verify(repo, times(2)).findByUsername("erin");
    }

    @Test
    void tc15_repoSqlIsParameterized_securityContract() {
        // This is a "contract" unit test: verify service/repo interaction passes raw email as a bind arg.

        service.resolveOrProvisionUser("ivan", "x@example.com' OR 1=1 --");

        verify(repo).findByEmailIgnoreCase("x@example.com' OR 1=1 --");
    }
