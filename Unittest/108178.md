import static org.junit.jupiter.api.Assertions.*;

import org.junit.jupiter.api.*;

import java.io.*;
import java.lang.reflect.*;
import java.nio.charset.StandardCharsets;

import java.util.*;

/**
 * Unit tests for User Story 108178 (Remove Unused Variables from ConfigMap).
 *
 * Notes:
 * - The repository version of 108178.java contains only a placeholder class and is not compilable as a runnable tool.
 * - These tests target the intended implementation described in the story context (class US108178), using reflection
 *   to access its internal methods (removeUnusedFromDataSection and Args.parse) without weakening encapsulation in prod.
 * - To run these tests, ensure the implementation class exists on the test classpath (e.g., US108178 in same module).
 *
 * Security/enterprise:
 * - Tests do not log secrets; they validate that no YAML values leak to audit output.
 */
public class US108178Test {

    private Method removeMethod;
    private Class<?> removalResultClass;

    @BeforeEach
    void locateMethods() throws Exception {
        Class<?> clazz = Class.forName("US108178");
        removeMethod = clazz.getDeclaredMethod("removeUnusedFromDataSection", List.class, Set.class);
        removeMethod.setAccessible(true);

        // Find nested class US108178$RemovalResult
        removalResultClass = Class.forName("US108178$RemovalResult");
    }

    @SuppressWarnings("unchecked")
    private List<String> cleanedLines(Object removalResult) throws Exception {
        Field f = removalResultClass.getDeclaredField("cleanedLines");
        f.setAccessible(true);
        return (List<String>) f.get(removalResult);
    }

    @SuppressWarnings("unchecked")
    private Set<String> removedKeys(Object removalResult) throws Exception {
        Field f = removalResultClass.getDeclaredField("removedKeys");
        f.setAccessible(true);
        return (Set<String>) f.get(removalResult);
    }

    private boolean dataSectionFound(Object removalResult) throws Exception {
        Field f = removalResultClass.getDeclaredField("dataSectionFound");
        f.setAccessible(true);
        return (boolean) f.get(removalResult);
    }

    private Object runRemove(List<String> lines, Set<String> unused) throws Exception {
        return removeMethod.invoke(null, lines, unused);
    }

    private static List<String> lines(String... s) {
        return Arrays.asList(s);
    }

    private static String join(List<String> lines) {
        return String.join("\n", lines) + "\n";
    }

    @Test
    void removeSingleKey_simpleMapping_happyPath() throws Exception {
        List<String> in = lines(
                "apiVersion: v1",
                "kind: ConfigMap",
                "data:",
                "  A: 1",
                "  B: 2"
        );
        Object r = runRemove(in, new LinkedHashSet<>(Set.of("A")));
        String out = join(cleanedLines(r));

        assertTrue(dataSectionFound(r));
        assertFalse(out.contains("A: 1"));
        assertTrue(out.contains("B: 2"));
        assertEquals(Set.of("A"), removedKeys(r));
    }

    @Test
    void removeMultipleKeys_happyPath() throws Exception {
        List<String> in = lines(
                "data:",
                "  A: 1",
                "  B: 2",
                "  C: 3"
        );
        Object r = runRemove(in, new LinkedHashSet<>(Set.of("A", "C")));
        String out = join(cleanedLines(r));
        assertFalse(out.contains("A: 1"));
        assertTrue(out.contains("B: 2"));
        assertFalse(out.contains("C: 3"));
        assertEquals(new LinkedHashSet<>(List.of("A", "C")), removedKeys(r));
    }

    @Test
    void noDataSection_preservesInput() throws Exception {
        List<String> in = lines(
                "apiVersion: v1",
                "kind: ConfigMap",
                "metadata:",
                "  name: x"
        );
        Object r = runRemove(in, new LinkedHashSet<>(Set.of("A")));
        assertFalse(dataSectionFound(r));
        assertEquals(join(in), join(cleanedLines(r)));
        assertTrue(removedKeys(r).isEmpty());
    }

    @Test
    void respectsSectionBoundary_onlyDataProcessed() throws Exception {
        List<String> in = lines(
                "data:",
                "  A: 1",
                "metadata:",
                "  A: shouldStay"
        );
        Object r = runRemove(in, new LinkedHashSet<>(Set.of("A")));
        String out = join(cleanedLines(r));
        assertFalse(out.contains("  A: 1"));
        assertTrue(out.contains("  A: shouldStay"));
    }

    @Test
    void exactKeyMatchOnly_doesNotRemoveSimilar() throws Exception {
        List<String> in = lines(
                "data:",
                "  A: 1",
                "  AA: 2"
        );
        Object r = runRemove(in, new LinkedHashSet<>(Set.of("A")));
        String out = join(cleanedLines(r));
        assertFalse(out.contains("A: 1"));
        assertTrue(out.contains("AA: 2"));
    }

    @Test
    void removeBlockScalar_removesWholeBlock() throws Exception {
        List<String> in = lines(
                "data:",
                "  SCRIPT: |",
                "    echo hello",
                "    echo world",
                "  KEEP: ok"
        );
        Object r = runRemove(in, new LinkedHashSet<>(Set.of("SCRIPT")));
        String out = join(cleanedLines(r));
        assertFalse(out.contains("SCRIPT:"));
        assertFalse(out.contains("echo hello"));
        assertTrue(out.contains("KEEP: ok"));
        assertEquals(Set.of("SCRIPT"), removedKeys(r));
    }

    @Test
    void blockScalarWithBlankLines_removedIncludingBlankLines() throws Exception {
        List<String> in = lines(

                "  SCRIPT: |",
                "    line1",
                "",
                "    # comment in block",
                "    line2",
                "  KEEP: ok"

        assertFalse(out.contains("SCRIPT:"));
        assertFalse(out.contains("line1"));
        assertFalse(out.contains("line2"));
        assertFalse(out.contains("comment in block"));
        assertTrue(out.contains("KEEP: ok"));
    }

    @Test
    void malformedKeyWithSpaces_preserved() throws Exception {
        List<String> in = lines(
                "data:",
                "  BAD KEY: 1",
                "  GOOD: 2"

        Object r = runRemove(in, new LinkedHashSet<>(Set.of("BAD KEY", "GOOD")));

        Object r = runRemove(in, new LinkedHashSet<>(Set.of("A")));

        assertFalse(out.contains("A: 1"));

    private static Object parseArgs(String... argv) throws Exception {
        Class<?> argsClass = Class.forName("US108178$Args");
        Method parse = argsClass.getDeclaredMethod("parse", String[].class);
        parse.setAccessible(true);
        return parse.invoke(null, (Object) argv);

    private static Set<String> unusedKeysFromArgs(Object argsObj) throws Exception {
        Field f = argsObj.getClass().getDeclaredField("unusedKeys");
        f.setAccessible(true);
        @SuppressWarnings("unchecked")
        Set<String> v = (Set<String>) f.get(argsObj);
        return v;
    }

    @Test
    void argsParse_unusedCsv_trimsAndSkipsEmptyTokens() throws Exception {
        Object argsObj = parseArgs("--unused", " A , ,B ");
        assertEquals(new LinkedHashSet<>(List.of("A", "B")), unusedKeysFromArgs(argsObj));
    }

    @Test
    void argsParse_unusedFile_ignoresCommentsAndBlankLines() throws Exception {
        Path tmp = Files.createTempFile("unused", ".txt");
        Files.writeString(tmp, "\n# comment\nA\n\nB\n", StandardCharsets.UTF_8);

        Object argsObj = parseArgs("--unused-file", tmp.toString());
        assertEquals(new LinkedHashSet<>(List.of("A", "B")), unusedKeysFromArgs(argsObj));
    }

    @Test

        assertThrows(InvocationTargetException.class, () -> parseArgs("--unused-file"));
        assertThrows(InvocationTargetException.class, () -> parseArgs("--in"));
    }

    @Test
    void auditLog_doesNotPrintValues_onlyKeys() throws Exception {
        Class<?> clazz = Class.forName("US108178");
        Method audit = clazz.getDeclaredMethod("audit", Class.forName("US108178$Args"), removalResultClass, PrintStream.class);
        audit.setAccessible(true);

        // Create Args instance via reflection and set fields.
        Object argsObj = parseArgs("--unused", "SECRET_KEY");
        // ensure inPath is null (stdin)

        List<String> in = lines(
                "data:",
                "  SECRET_KEY: super-secret-value",
                "  OTHER: ok"
        );
        Object r = runRemove(in, new LinkedHashSet<>(Set.of("SECRET_KEY")));

        ByteArrayOutputStream errBuf = new ByteArrayOutputStream();
        PrintStream ps = new PrintStream(errBuf, true, StandardCharsets.UTF_8);
        audit.invoke(null, argsObj, r, ps);

        String err = errBuf.toString(StandardCharsets.UTF_8);
        assertTrue(err.contains("SECRET_KEY"));
        assertFalse(err.contains("super-secret-value"));
    }
