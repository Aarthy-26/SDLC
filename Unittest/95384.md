    @Test
    @Timeout(5)
    void ttlCache_preventsStampede_loaderRunsOncePerKey() throws Exception {
        MutableClock clock = new MutableClock(Instant.parse("2026-01-01T00:00:00Z"));
        TtlCache<String, String> cache = new TtlCache<>(clock, Duration.ofSeconds(5));

        int threads = 16;
        ExecutorService pool = Executors.newFixedThreadPool(threads);
        CountDownLatch ready = new CountDownLatch(threads);
        CountDownLatch start = new CountDownLatch(1);
        AtomicInteger calls = new AtomicInteger();

        Callable<String> task = () -> {
            ready.countDown();
            start.await();
            return cache.get("k", () -> {
                calls.incrementAndGet();
                // slow loader to amplify stampede risk
                try {
                    Thread.sleep(50);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
                return "value";
            });
        };

        List<Future<String>> futures = new ArrayList<>();
        for (int i = 0; i < threads; i++) {
            futures.add(pool.submit(task));
        }
        ready.await();
        start.countDown();

        for (Future<String> f : futures) {
            assertEquals("value", f.get(1, TimeUnit.SECONDS));
        }
        pool.shutdownNow();

        assertEquals(1, calls.get(), "Loader should have been called once due to compute() per-key lock");
    }
