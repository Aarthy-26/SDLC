import codeoutput.us105821.EntityColumnMappingTool.ColumnFinding;

/**
 * Unit tests for User Story 105821: EntityColumnMappingTool.
 *
 * Security/enterprise notes:
 * - Tests use @TempDir; no fixed filesystem paths.
 * - No secrets or external network.
 * - All file writes are confined to temp directories.

public class EntityColumnMappingToolTest {

    @TempDir
    Path tempDir;

    // ---------- ColumnNamingStrategy / parsing ----------

    @Test
    void fromName_resolvesKnownNames_caseInsensitive() {
        Optional<ColumnNamingStrategy> s = ColumnNamingStrategy.fromName("snake_upper");
        assertTrue(s.isPresent());
        assertEquals(ColumnNamingStrategy.SNAKE_UPPER, s.get());
    }

    @Test
    void fromName_null_returnsEmpty() {
        assertTrue(ColumnNamingStrategy.fromName(null).isEmpty());
    }

    @Test
    void fromName_unknown_returnsEmpty() {
        assertTrue(ColumnNamingStrategy.fromName("does_not_exist").isEmpty());
    }

    @Test
    void toSnake_handlesAcronymBoundary_URLValue() {
        assertEquals("URL_Value", ColumnNamingStrategy.toSnake("URLValue"));
    }

    @Test
    void toSnake_replacesHyphen_andCollapsesUnderscores() {
        assertEquals("a_b_c", ColumnNamingStrategy.toSnake("a-b__c"));
    }

    @Test
    void toSnake_blank_returnsEmpty() {
        assertEquals("", ColumnNamingStrategy.toSnake("   "));
    }

    @Test
    void normalize_snakeUpper() {
        assertEquals("CREATED_AT", ColumnNamingStrategy.SNAKE_UPPER.normalize("createdAt"));
    }

    @Test
    void normalize_snakeLower() {
        assertEquals("created_at", ColumnNamingStrategy.SNAKE_LOWER.normalize("createdAt"));
    }

    @Test
    void normalize_asIs() {
        assertEquals("createdAt", ColumnNamingStrategy.AS_IS.normalize("createdAt"));
    }

    // ---------- Scanner ----------

    @Test
    void scan_throwsWhenRootIsNotDirectory() {
        EntitySourceScanner scanner = new EntitySourceScanner(ColumnNamingStrategy.SNAKE_UPPER);
        ToolException ex = assertThrows(ToolException.class,
                () -> scanner.scan(tempDir.resolve("not-a-dir"), StandardCharsets.UTF_8));
        assertTrue(ex.getMessage().contains("--root must be an existing directory"));
    }

    @Test
    void scanFile_ignoresNonJavaFiles() throws IOException {
        Path f = tempDir.resolve("a.txt");
        Files.writeString(f, "@Column(name=\"X\")\nprivate String a;\n", StandardCharsets.UTF_8);

        EntitySourceScanner scanner = new EntitySourceScanner(ColumnNamingStrategy.SNAKE_UPPER);
        assertTrue(scanner.scanFile(f, StandardCharsets.UTF_8).isEmpty());
    }

    @Test
    void scanFile_detectsJavaxColumnNameBeforeField() throws IOException {
        Path f = tempDir.resolve("CustomerEntity.java");
        String src = String.join(System.lineSeparator(),
                "package demo;",
                "import javax.persistence.Column;",
                "public class CustomerEntity {",
                "  @Column(name=\"customerName\")",
                "  private String customerName;",
                "}");
        Files.writeString(f, src, StandardCharsets.UTF_8);

        EntitySourceScanner scanner = new EntitySourceScanner(ColumnNamingStrategy.SNAKE_UPPER);
        List<ColumnFinding> findings = scanner.scanFile(f, StandardCharsets.UTF_8);

        assertEquals(1, findings.size());
        ColumnFinding c = findings.get(0);
        assertEquals("CustomerEntity", c.className);
        assertEquals("customerName", c.fieldName);
        assertEquals("customerName", c.currentColumnName);
        assertEquals("CUSTOMER_NAME", c.expectedColumnName);
        assertEquals(4, c.lineNumber);
    }

    @Test

    // ---------- Rewriter ----------

    @Test
    void rewriter_rewritesNonCompliant_andCreatesBackupAndHashes() throws IOException {
        Path root = Files.createDirectories(tempDir.resolve("src"));
        Path f = root.resolve("E.java");

        String src = String.join(System.lineSeparator(),
                "package demo;",
                "import javax.persistence.Column;",
                "public class E {",
                "  @Column(name=\"wrongName\")",
                "  private String createdAt;",
                "}");
        Files.writeString(f, src, StandardCharsets.UTF_8);

        EntitySourceScanner scanner = new EntitySourceScanner(ColumnNamingStrategy.SNAKE_UPPER);
        ScanReport report = scanner.scan(tempDir, StandardCharsets.UTF_8);
        assertEquals(1, report.nonCompliantCount());

        EntitySourceRewriter rewriter = new EntitySourceRewriter(ColumnNamingStrategy.SNAKE_UPPER);
        RewriteReport rr = rewriter.apply(report, StandardCharsets.UTF_8);

        assertEquals(1, rr.results.size());
        var res = rr.results.get(f);
        assertNotNull(res);
        assertTrue(res.changed);
        assertNotNull(res.beforeSha256);
        assertNotNull(res.afterSha256);
        assertNotEquals(res.beforeSha256, res.afterSha256);

        // backup created
        Path bak = f.resolveSibling("E.java.bak");
        assertTrue(Files.exists(bak));

        String updated = Files.readString(f, StandardCharsets.UTF_8);
        assertTrue(updated.contains("@Column(name=\"CREATED_AT\")"));
    }

    @Test
    void rewriter_doesNotRewriteOnContentDrift() throws IOException {
        Path root = Files.createDirectories(tempDir.resolve("src"));
        Path f = root.resolve("E.java");

        String src = String.join(System.lineSeparator(),
                "package demo;",
                "import javax.persistence.Column;",
                "public class E {",
                "  @Column(name=\"wrongName\")",
                "  private String createdAt;",
                "}");
        Files.writeString(f, src, StandardCharsets.UTF_8);

        EntitySourceScanner scanner = new EntitySourceScanner(ColumnNamingStrategy.SNAKE_UPPER);
        ScanReport report = scanner.scan(tempDir, StandardCharsets.UTF_8);

        // Introduce drift: change the line the finding points to
        List<String> lines = Files.readAllLines(f, StandardCharsets.UTF_8);
        lines.set(3, "  @Column(name=\"drifted\")");
        Files.writeString(f, String.join(System.lineSeparator(), lines) + System.lineSeparator(), StandardCharsets.UTF_8);

        EntitySourceRewriter rewriter = new EntitySourceRewriter(ColumnNamingStrategy.SNAKE_UPPER);
        RewriteReport rr = rewriter.apply(report, StandardCharsets.UTF_8);
        var res = rr.results.get(f);

        assertNotNull(res);
        assertFalse(res.changed);

        String unchanged = Files.readString(f, StandardCharsets.UTF_8);
        assertTrue(unchanged.contains("@Column(name=\"drifted\")"));
        assertFalse(Files.exists(f.resolveSibling("E.java.bak")));
    }

    @Test
    void rewriter_defensiveWhenPatternDoesNotMatch_lineUnchanged() throws IOException {
        Path root = Files.createDirectories(tempDir.resolve("src"));
        Path f = root.resolve("E.java");

        // Multiline/odd format that scanner can detect (line has @Column(name="X"))
        // but rewriter pattern won't match due to missing closing ")" on the same line.
        String src = String.join(System.lineSeparator(),
                "package demo;",
                "import javax.persistence.Column;",
                "public class E {",
                "  @Column(name=\"wrongName\"",
                "  )",
                "  private String createdAt;",
                "}");
        Files.writeString(f, src, StandardCharsets.UTF_8);

        // Scanner will NOT detect because annotation pattern requires ) on same line; ensure no findings.
        EntitySourceScanner scanner = new EntitySourceScanner(ColumnNamingStrategy.SNAKE_UPPER);
        ScanReport report = scanner.scan(tempDir, StandardCharsets.UTF_8);
        assertEquals(0, report.allFindings().size());

        // So apply results should be empty and file unchanged
        EntitySourceRewriter rewriter = new EntitySourceRewriter(ColumnNamingStrategy.SNAKE_UPPER);
        RewriteReport rr = rewriter.apply(report, StandardCharsets.UTF_8);
        assertTrue(rr.results.isEmpty());
        assertEquals(src, Files.readString(f, StandardCharsets.UTF_8));
    }

    @Test

        Path root = Files.createDirectories(tempDir.resolve("root"));
        Path target = Files.createDirectories(root.resolve("target"));
        Path git = Files.createDirectories(root.resolve(".git"));

        Files.writeString(target.resolve("X.java"), "class X{}", StandardCharsets.UTF_8);
        Files.writeString(git.resolve("Y.java"), "class Y{}", StandardCharsets.UTF_8);

        AtomicInteger visited = new AtomicInteger();
        JavaFileVisitor v = new JavaFileVisitor(p -> visited.incrementAndGet());
        Files.walkFileTree(root, v);

        assertEquals(0, visited.get());
    }

        assertDoesNotThrow(SelfTest::run);
