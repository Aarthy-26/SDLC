        Class<?> modeEnum = Class.forName("com.aava.dpai.migration.EntityColumnMappingUpdater$CliOptions$Mode");

        Constructor<?> ctor = cliClass.getDeclaredConstructor(Path.class, Path.class, modeEnum, Path.class, Path.class, boolean.class);

        assertTrue(first.updatedSource().contains("@Deprecated"));
        assertTrue(first.updatedSource().contains("getOldField()"));

        var second = EntityColumnMappingUpdater.JavaSourceUpdater.applyMappings(first.updatedSource(), info, rows, true);
