// File: DpaiSsoConfigCheckerTest.java
// JUnit 5 tests for DpaiSsoConfigChecker (Java 17+)
// Security/enterprise notes:
// - Tests never log or assert on actual secret values; they only assert masking behavior.
// - System properties are restored after each test to avoid cross-test contamination.

import org.junit.jupiter.api.*;

import java.io.ByteArrayOutputStream;
import java.io.PrintStream;
import java.lang.reflect.Field;
import java.lang.reflect.Method;
import java.nio.charset.StandardCharsets;
import java.util.*;

import static org.junit.jupiter.api.Assertions.*;

    private static final List<String> REQUIRED_KEYS = List.of(
            "AAVA_SSO_ISSUER",
            "AAVA_SSO_CLIENT_ID",
            "AAVA_SSO_REDIRECT_URI",
            "AAVA_SSO_LOGIN_URL"
    );

    private static final List<String> OPTIONAL_KEYS = List.of(
            "AAVA_SSO_CLIENT_SECRET",
            "AAVA_SSO_SCOPES",
            "AAVA_SSO_AUDIENCE"
    );

    private Properties originalSystemProps;
    private PrintStream originalOut;

    @BeforeEach
    void beforeEach() {
        originalSystemProps = (Properties) System.getProperties().clone();
        originalOut = System.out;
        clearAllRelevantSystemProperties();
    }

    @AfterEach
    void afterEach() {
        System.setProperties(originalSystemProps);
        System.setOut(originalOut);
    }

    private static void clearAllRelevantSystemProperties() {
        for (String k : REQUIRED_KEYS) {
            System.clearProperty(k);
        }
        for (String k : OPTIONAL_KEYS) {
            System.clearProperty(k);
        }
    }

    private static Map<String, String> envSnapshot() {
        // Copy to avoid accidental mutation attempts.
        return new HashMap<>(System.getenv());
    }

    /**
     * Sets an environment variable for the current process via reflection.
     * If the platform/JDK forbids it, test will be skipped.
     */
    private static void setEnvOrSkip(String key, String value) {
        Assumptions.assumeTrue(canMutateEnv(), "Environment mutation not supported on this runtime");
        try {
            // Works on many JDKs, but is not guaranteed; hence skip if it fails.
            Map<String, String> env = System.getenv();
            Class<?> cl = env.getClass();

            // Common for Oracle/OpenJDK: java.util.Collections$UnmodifiableMap wrapping a mutable map in field "m"
            Field mField = cl.getDeclaredField("m");
            mField.setAccessible(true);
            @SuppressWarnings("unchecked")
            Map<String, String> mutable = (Map<String, String>) mField.get(env);

            if (value == null) {
                mutable.remove(key);
            } else {
                mutable.put(key, value);
            }
        } catch (Exception e) {
            Assumptions.abort("Env mutation failed: " + e.getClass().getSimpleName() + ": " + e.getMessage());
        }
    }

    private static boolean canMutateEnv() {
        try {
            Map<String, String> env = System.getenv();
            env.getClass().getDeclaredField("m");
            return true;
        } catch (Exception e) {
            return false;
        }
    }

    private static String runMainCapturingStdout(String... args) {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        System.setOut(new PrintStream(baos, true, StandardCharsets.UTF_8));
        DpaiSsoConfigChecker.main(args);
        return baos.toString(StandardCharsets.UTF_8);
    }

    private static Method getPrivateStaticMethod(String name, Class<?>... paramTypes) throws Exception {
        Method m = DpaiSsoConfigChecker.class.getDeclaredMethod(name, paramTypes);
        m.setAccessible(true);
        return m;
    }

    @Test
    void tc05_envPrecedence_overSystemProperty_whenEnvNonBlank() {
        Map<String, String> before = envSnapshot();
        try {
            // sysprops
            System.setProperty("AAVA_SSO_LOGIN_URL", "https://login.sysprop/auth");
            System.setProperty("AAVA_SSO_REDIRECT_URI", "https://redirect.sysprop/cb");
            System.setProperty("AAVA_SSO_ISSUER", "https://issuer.sysprop");
            System.setProperty("AAVA_SSO_CLIENT_ID", "client-sysprop");

            // env overrides
            setEnvOrSkip("AAVA_SSO_LOGIN_URL", "https://login.env/auth");
            setEnvOrSkip("AAVA_SSO_REDIRECT_URI", "https://redirect.env/cb");
            setEnvOrSkip("AAVA_SSO_ISSUER", "https://issuer.env");
            setEnvOrSkip("AAVA_SSO_CLIENT_ID", "client-env");

            String out = runMainCapturingStdout();
            assertTrue(out.contains("Status: PASS"));
            assertTrue(out.contains("https://login.env/auth"));
            assertTrue(out.contains("https://redirect.env/cb"));
            assertFalse(out.contains("https://login.sysprop/auth"));
        } finally {
            // Best effort restore env keys modified by this test
            for (String k : REQUIRED_KEYS) {
                String v = before.get(k);
                setEnvOrSkip(k, v);
            }
        }
    }

    @Test
    void tc12_concat_preservesOrderAndSize() throws Exception {
        Method concat = getPrivateStaticMethod("concat", List.class, List.class);

        // Reflectively obtain REQUIRED and OPTIONAL lists
        Field requiredField = DpaiSsoConfigChecker.class.getDeclaredField("REQUIRED");
        requiredField.setAccessible(true);
        @SuppressWarnings("unchecked")
        List<Object> required = (List<Object>) requiredField.get(null);

        Field optionalField = DpaiSsoConfigChecker.class.getDeclaredField("OPTIONAL");
        optionalField.setAccessible(true);
        @SuppressWarnings("unchecked")
        List<Object> optional = (List<Object>) optionalField.get(null);

        @SuppressWarnings("unchecked")
        List<Object> out = (List<Object>) concat.invoke(null, required, optional);

        assertEquals(required.size() + optional.size(), out.size());
        assertSame(required.get(0), out.get(0));
        assertSame(required.get(required.size() - 1), out.get(required.size() - 1));
        assertSame(optional.get(0), out.get(required.size()));
    }

    private static int countOccurrences(String s, String needle) {
        int count = 0;
        int idx = 0;
        while ((idx = s.indexOf(needle, idx)) >= 0) {
            count++;
            idx += needle.length();
        }
        return count;
    }
