/**
 * Unit tests for Jackson deserialization + bean validation + controller behavior.
 *
 * Enterprise notes:
 * - No secrets.
 * - No network calls.
 * - Uses Spring's configured ObjectMapper for fidelity in MVC tests.
 */
class ColumnResponseUnitTests {

    private static ValidatorFactory validatorFactory;
    private static Validator validator;

    private final ObjectMapper objectMapper = new ObjectMapper();

    @BeforeAll
    static void setupValidator() {
        validatorFactory = Validation.buildDefaultValidatorFactory();
        validator = validatorFactory.getValidator();
    }

    @AfterAll
    static void tearDownValidator() {
        if (validatorFactory != null) {
            validatorFactory.close();
        }
    }

    @Test
    void shouldDeserialize_withAllFields_present() throws Exception {
        String json = "{\"name\":\"colA\",\"type\":\"STRING\",\"nullable\":true}";
        ColumnResponse dto = objectMapper.readValue(json, ColumnResponse.class);

        assertNotNull(dto);
        assertEquals("colA", dto.getName());
        assertEquals("STRING", dto.getType());
        assertTrue(dto.isNullable());
    }

    @Test
    void shouldDeserialize_andIgnoreUnknownProperties() throws Exception {
        String json = "{\"name\":\"colA\",\"type\":\"STRING\",\"nullable\":true,\"extra\":123,\"other\":{\"x\":\"y\"}}";
        ColumnResponse dto = objectMapper.readValue(json, ColumnResponse.class);

        assertNotNull(dto);
        assertEquals("colA", dto.getName());
        assertEquals("STRING", dto.getType());
        assertTrue(dto.isNullable());
    }

    @Test
    void shouldDeserialize_withOnlyName_present() throws Exception {
        String json = "{\"name\":\"colA\"}";
        ColumnResponse dto = objectMapper.readValue(json, ColumnResponse.class);

        assertNotNull(dto);
        assertEquals("colA", dto.getName());
        assertNull(dto.getType());
        assertFalse(dto.isNullable());
    }

    @Test
    void shouldDeserialize_withTypeNull() throws Exception {
        String json = "{\"name\":\"colA\",\"type\":null,\"nullable\":false}";
        ColumnResponse dto = objectMapper.readValue(json, ColumnResponse.class);

        assertNotNull(dto);
        assertEquals("colA", dto.getName());
        assertNull(dto.getType());
        assertFalse(dto.isNullable());
    }

    @Test
    void validation_shouldFail_whenNameNull() throws Exception {
        String json = "{\"name\":null,\"type\":\"STRING\",\"nullable\":true}";
        ColumnResponse dto = objectMapper.readValue(json, ColumnResponse.class);

        Set<ConstraintViolation<ColumnResponse>> violations = validator.validate(dto);
        assertThat(violations).isNotEmpty();
        assertThat(violations.stream().anyMatch(v -> v.getPropertyPath().toString().equals("name"))).isTrue();
    }

    @Test
    void validation_shouldFail_whenNameBlank() throws Exception {
        String json = "{\"name\":\"   \",\"type\":\"STRING\",\"nullable\":true}";
        ColumnResponse dto = objectMapper.readValue(json, ColumnResponse.class);

        Set<ConstraintViolation<ColumnResponse>> violations = validator.validate(dto);
        assertThat(violations).isNotEmpty();
        assertThat(violations.stream().anyMatch(v -> v.getPropertyPath().toString().equals("name"))).isTrue();
    }

    @Test
    void validation_shouldPass_whenNameLengthIs200() {
        String name200 = "a".repeat(200);
        ColumnResponse dto = new ColumnResponse(name200, "T", true);

        Set<ConstraintViolation<ColumnResponse>> violations = validator.validate(dto);
        assertThat(violations).isEmpty();
    }

    @Test
    void validation_shouldFail_whenNameLengthIs201() {
        String name201 = "a".repeat(201);
        ColumnResponse dto = new ColumnResponse(name201, "T", true);

        Set<ConstraintViolation<ColumnResponse>> violations = validator.validate(dto);
        assertThat(violations).isNotEmpty();
        assertThat(violations.stream().anyMatch(v -> v.getPropertyPath().toString().equals("name"))).isTrue();
    }

    @Test
    void validation_shouldPass_whenTypeLengthIs100() {
        String type100 = "t".repeat(100);
        ColumnResponse dto = new ColumnResponse("colA", type100, false);

        Set<ConstraintViolation<ColumnResponse>> violations = validator.validate(dto);
        assertThat(violations).isEmpty();
    }

    @Test
    void validation_shouldFail_whenTypeLengthIs101() {
        String type101 = "t".repeat(101);
        ColumnResponse dto = new ColumnResponse("colA", type101, false);

        Set<ConstraintViolation<ColumnResponse>> violations = validator.validate(dto);
        assertThat(violations).isNotEmpty();
        assertThat(violations.stream().anyMatch(v -> v.getPropertyPath().toString().equals("type"))).isTrue();
    }

    @Test
    void shouldFailDeserialization_whenNullableHasWrongType() {
        String json = "{\"name\":\"colA\",\"type\":\"STRING\",\"nullable\":\"true\"}";
        assertThrows(MismatchedInputException.class, () -> objectMapper.readValue(json, ColumnResponse.class));
    }

    @Test
    void shouldFailDeserialization_whenJsonMalformed() {
        String json = "{\"name\":\"colA\",\"type\":\"STRING\",\"nullable\":true"; // missing closing brace
        assertThrows(JsonProcessingException.class, () -> objectMapper.readValue(json, ColumnResponse.class));
    }

    @Test
    void shouldRoundTripSerializeDeserialize() throws Exception {
        ColumnResponse in = new ColumnResponse("colA", "STRING", true);
        String json = objectMapper.writeValueAsString(in);
        ColumnResponse out = objectMapper.readValue(json, ColumnResponse.class);

        assertEquals(in.getName(), out.getName());
        assertEquals(in.getType(), out.getType());
        assertEquals(in.isNullable(), out.isNullable());
    }

    @Test
    void volume_shouldDeserializeLargeList() throws Exception {
        int n = 10_000;
        StringBuilder sb = new StringBuilder(n * 64);
        sb.append("{\"columns\":[");
        for (int i = 0; i < n; i++) {
            if (i > 0) sb.append(',');
            sb.append("{\"name\":\"c").append(i).append("\",\"type\":\"STRING\",\"nullable\":").append(i % 2 == 0).append('}');
        }
        sb.append("]}");

        ImportRequest req = objectMapper.readValue(sb.toString(), ImportRequest.class);
        assertNotNull(req);
        assertNotNull(req.columns());
        assertEquals(n, req.columns().size());
    }
}

@WebMvcTest(controllers = AppImportController.class)
@Import({com.ascendion.aava.appimport.api.AppImportController.class})
class AppImportControllerWebMvcTests {

    @Autowired
    MockMvc mockMvc;

    @Autowired
    ObjectMapper objectMapper;

    @Test
    void importApp_shouldEchoColumns_happyPath() throws Exception {
        String json = "{\"columns\":[{\"name\":\"colA\",\"type\":\"STRING\",\"nullable\":true}]}";

        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .accept(MediaType.APPLICATION_JSON)
                        .content(json))
                .andExpect(status().isOk())
                .andExpect(content().contentTypeCompatibleWith(MediaType.APPLICATION_JSON))
                .andExpect(jsonPath("$[0].name").value("colA"))
                .andExpect(jsonPath("$[0].type").value("STRING"))
                .andExpect(jsonPath("$[0].nullable").value(true));
    }

    @Test
    void importApp_shouldReturnEmptyList_whenColumnsEmpty() throws Exception {
        String json = "{\"columns\":[]}";

        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .accept(MediaType.APPLICATION_JSON)
                        .content(json))
                .andExpect(status().isOk())
                .andExpect(content().json("[]"));
    }

    @Test
    void importApp_shouldFail_whenColumnsMissing() throws Exception {
        String json = "{}";

        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .accept(MediaType.APPLICATION_JSON)
                        .content(json))
                // Default Spring Boot validation error is 400
                .andExpect(status().isBadRequest());
    }

    @Test
    void importApp_shouldFail_whenColumnsNull() throws Exception {
        String json = "{\"columns\":null}";

        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .accept(MediaType.APPLICATION_JSON)
                        .content(json))
                .andExpect(status().isBadRequest());
    }

    @Test
    void importApp_shouldFail_whenNestedNameMissing() throws Exception {
        String json = "{\"columns\":[{\"type\":\"STRING\",\"nullable\":true}]}";

        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .accept(MediaType.APPLICATION_JSON)
                        .content(json))
                .andExpect(status().isBadRequest());
    }

    @Test
    void importApp_shouldFail_whenNestedNameBlank() throws Exception {
        String json = "{\"columns\":[{\"name\":\"   \",\"type\":\"STRING\",\"nullable\":true}]}";

        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .accept(MediaType.APPLICATION_JSON)
                        .content(json))
                .andExpect(status().isBadRequest());
    }

    @Test
    void importApp_shouldFail_whenNestedNameTooLong() throws Exception {
        String longName = "a".repeat(201);
        String json = objectMapper.writeValueAsString(new Object() {
            public final List<Object> columns = List.of(new Object() {
                public final String name = longName;
                public final String type = "STRING";
                public final boolean nullable = true;
            });
        });

        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .accept(MediaType.APPLICATION_JSON)
                        .content(json))
                .andExpect(status().isBadRequest());
    }

    @Test
    void importApp_shouldFail_whenNestedTypeTooLong() throws Exception {
        String longType = "t".repeat(101);
        String json = objectMapper.writeValueAsString(new Object() {
            public final List<Object> columns = List.of(new Object() {
                public final String name = "colA";
                public final String type = longType;
                public final boolean nullable = true;
            });
        });

        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .accept(MediaType.APPLICATION_JSON)
                        .content(json))
                .andExpect(status().isBadRequest());
    }

    @Test
    void importApp_shouldIgnoreUnknownPropertiesInColumnItems() throws Exception {
        String json = "{\"columns\":[{\"name\":\"colA\",\"type\":\"STRING\",\"nullable\":true,\"unknown\":\"x\"}]}";

        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .accept(MediaType.APPLICATION_JSON)
                        .content(json))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$[0].name").value("colA"));
    }

    @Test
    void importApp_shouldDefensivelyCopyList_notExposeBackingListReference() {
        AppImportController controller = new AppImportController();

        List<ColumnResponse> cols = new ArrayList<>();
        cols.add(new ColumnResponse("colA", "STRING", true));
        ImportRequest req = new ImportRequest(cols);

        List<ColumnResponse> out = controller.importApp(req);
        assertEquals(1, out.size());

        // Mutate input list after call; output list must not change.
        cols.add(new ColumnResponse("colB", "INT", false));
        assertEquals(1, out.size());
    }
}
