package com.aava.datastudio.appimport;

import com.aava.datastudio.appimport.dto.ColumnResponse;
import com.aava.datastudio.appimport.dto.ImportRequest;
import com.fasterxml.jackson.databind.ObjectMapper;
import jakarta.validation.ConstraintViolation;
import jakarta.validation.Validation;
import jakarta.validation.Validator;
import jakarta.validation.ValidatorFactory;
import org.junit.jupiter.api.AfterAll;
import org.junit.jupiter.api.BeforeAll;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.web.servlet.AutoConfigureMockMvc;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;

import java.util.ArrayList;
import java.util.List;
import java.util.Set;

import static org.assertj.core.api.Assertions.assertThat;
import static org.junit.jupiter.api.Assertions.assertThrows;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

/**
 * Unit/integration tests for User Story 95383 (Jackson constructibility fix for ColumnResponse).
 *
 * Enterprise/Security notes:
 * - No external network calls.
 * - No secrets/logging of sensitive values.
 * - Uses Spring's MockMvc for controller tests.
 */
@SpringBootTest
@AutoConfigureMockMvc
class UserStory95383_JacksonDeserializationTests {

    @Autowired
    private ObjectMapper objectMapper;

    @Autowired
    private MockMvc mockMvc;

    private static ValidatorFactory validatorFactory;
    private static Validator validator;

    @BeforeAll
    static void initValidator() {
        validatorFactory = Validation.buildDefaultValidatorFactory();
        validator = validatorFactory.getValidator();
    }

    @AfterAll
    static void closeValidator() {
        if (validatorFactory != null) {
            validatorFactory.close();
        }
    }

    @Test
    void deserializeColumnResponse_allFields_present() throws Exception {
        ColumnResponse cr = objectMapper.readValue(
                "{\"name\":\"id\",\"type\":\"integer\",\"nullable\":false}",
                ColumnResponse.class);

        assertThat(cr.name()).isEqualTo("id");
        assertThat(cr.type()).isEqualTo("integer");
        assertThat(cr.nullable()).isFalse();
    }

    @Test
    void deserializeColumnResponse_optionalFields_missing_defaultsToNull() throws Exception {
        ColumnResponse cr = objectMapper.readValue(
                "{\"name\":\"id\"}",
                ColumnResponse.class);

        assertThat(cr.name()).isEqualTo("id");
        assertThat(cr.type()).isNull();
        assertThat(cr.nullable()).isNull();
    }

    @Test
    void deserializeColumnResponse_unknownFields_ignored() throws Exception {
        ColumnResponse cr = objectMapper.readValue(
                "{\"name\":\"id\",\"unknown\":\"x\"}",
                ColumnResponse.class);

        assertThat(cr.name()).isEqualTo("id");
    }

    @Test
    void deserializeImportRequest_happyPath_singleColumn() throws Exception {
        ImportRequest req = objectMapper.readValue(
                "{\"columns\":[{\"name\":\"id\",\"type\":\"integer\",\"nullable\":false}]}",
                ImportRequest.class);

        assertThat(req.columns()).hasSize(1);
        assertThat(req.columns().get(0).name()).isEqualTo("id");
        assertThat(req.columns().get(0).type()).isEqualTo("integer");
        assertThat(req.columns().get(0).nullable()).isFalse();
    }

    @Test
    void deserializeImportRequest_unknownTopLevelFields_ignored() throws Exception {
        ImportRequest req = objectMapper.readValue(
                "{\"columns\":[{\"name\":\"id\"}],\"futureField\":123}",
                ImportRequest.class);

        assertThat(req.columns()).hasSize(1);
        assertThat(req.columns().get(0).name()).isEqualTo("id");
    }

    @Test
    void deserializeImportRequest_columnsNull_isNormalizedToEmptyList() throws Exception {
        ImportRequest req = objectMapper.readValue(
                "{\"columns\":null}",
                ImportRequest.class);

        assertThat(req.columns()).isNotNull();
        assertThat(req.columns()).isEmpty();
    }

    @Test
    void deserializeImportRequest_columnsMissing_isNormalizedToEmptyList() throws Exception {
        ImportRequest req = objectMapper.readValue(
                "{}",
                ImportRequest.class);

        assertThat(req.columns()).isNotNull();
        assertThat(req.columns()).isEmpty();
    }

    @Test
    void importRequest_defensiveCopy_isUnmodifiable() {
        List<ColumnResponse> mutable = new ArrayList<>();
        mutable.add(new ColumnResponse("id", null, null));

        ImportRequest req = new ImportRequest(mutable);

        assertThrows(UnsupportedOperationException.class,
                () -> req.columns().add(new ColumnResponse("x", null, null)));
    }

    @Test
    void beanValidation_columnResponse_nameBlank_fails() {
        Set<ConstraintViolation<ColumnResponse>> violations = validator.validate(
                new ColumnResponse(" ", "integer", Boolean.FALSE));

        assertThat(violations).isNotEmpty();
        assertThat(violations.stream().anyMatch(v -> v.getPropertyPath().toString().equals("name"))).isTrue();
    }

    @Test
    void web_importApp_validPayload_returns200_andEchoes() throws Exception {
        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content("{\"columns\":[{\"name\":\"id\",\"type\":\"integer\",\"nullable\":false}]}"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.columns").isArray())
                .andExpect(jsonPath("$.columns[0].name").value("id"))
                .andExpect(jsonPath("$.columns[0].type").value("integer"))
                .andExpect(jsonPath("$.columns[0].nullable").value(false));
    }

    @Test
    void web_importApp_missingName_triggersValidation400_withFieldErrors() throws Exception {
        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content("{\"columns\":[{\"type\":\"integer\"}]}"))
                .andExpect(status().isBadRequest())
                .andExpect(jsonPath("$.error").value("validation_error"))
                .andExpect(jsonPath("$.status").value(400))
                .andExpect(jsonPath("$.fieldErrors").exists());
    }

    @Test
    void web_importApp_invalidJson_triggers400_invalid_json() throws Exception {
        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content("{not-json}"))
                .andExpect(status().isBadRequest())
                .andExpect(jsonPath("$.error").value("invalid_json"))
                .andExpect(jsonPath("$.status").value(400))
                .andExpect(jsonPath("$.message").exists());
    }

    @Test
    void web_importApp_unknownColumnFields_areIgnored() throws Exception {
        mockMvc.perform(post("/api/import")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content("{\"columns\":[{\"name\":\"id\",\"unknown\":true}]}"))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.columns[0].name").value("id"));
    }
}
