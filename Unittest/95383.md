package com.example.appimport.dto;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.DeserializationFeature;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.exc.MismatchedInputException;
import org.junit.jupiter.api.Test;

import java.util.ArrayList;
import java.util.List;

import static org.junit.jupiter.api.Assertions.*;

/**
 * Unit test inventory implementation for User Story 95383.
 *
 * Enterprise/security notes:
 * - No external calls, no secrets, deterministic assertions.
 * - Exceptions are asserted by type; messages are not relied on.
 */
class ColumnResponseUnitTest {

    private static ObjectMapper mapperFailOnUnknownDisabled() {
        return new ObjectMapper().configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);
    }

    @Test
    void tc01_deserialize_validJson_allFields() throws Exception {
        ObjectMapper mapper = mapperFailOnUnknownDisabled();
        String json = "{\"name\":\"colA\",\"type\":\"STRING\",\"nullable\":true}";

        ColumnResponse r = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(r);
        assertEquals("colA", r.getName());
        assertEquals("STRING", r.getType());
        assertEquals(Boolean.TRUE, r.getNullable());
    }

    @Test
    void tc02_deserialize_validJson_nullableFalse() throws Exception {
        ObjectMapper mapper = mapperFailOnUnknownDisabled();
        String json = "{\"name\":\"colB\",\"type\":\"INT\",\"nullable\":false}";

        ColumnResponse r = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(r);
        assertEquals("colB", r.getName());
        assertEquals("INT", r.getType());
        assertEquals(Boolean.FALSE, r.getNullable());
    }

    @Test
    void tc03_deserialize_missingOptional_type() throws Exception {
        ObjectMapper mapper = mapperFailOnUnknownDisabled();
        String json = "{\"name\":\"colC\",\"nullable\":true}";

        ColumnResponse r = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(r);
        assertEquals("colC", r.getName());
        assertNull(r.getType());
        assertEquals(Boolean.TRUE, r.getNullable());
    }

    @Test
    void tc04_deserialize_missingOptional_nullable() throws Exception {
        ObjectMapper mapper = mapperFailOnUnknownDisabled();
        String json = "{\"name\":\"colD\",\"type\":\"STRING\"}";

        ColumnResponse r = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(r);
        assertEquals("colD", r.getName());
        assertEquals("STRING", r.getType());
        assertNull(r.getNullable());
    }

    @Test
    void tc05_deserialize_allFieldsExplicitNull() throws Exception {
        ObjectMapper mapper = mapperFailOnUnknownDisabled();
        String json = "{\"name\":null,\"type\":null,\"nullable\":null}";

        ColumnResponse r = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(r);
        assertNull(r.getName());
        assertNull(r.getType());
        assertNull(r.getNullable());
    }

    @Test
    void tc06_deserialize_unknownField_ignored() throws Exception {
        ObjectMapper mapper = new ObjectMapper(); // annotation @JsonIgnoreProperties(ignoreUnknown=true) should handle
        String json = "{\"name\":\"colE\",\"type\":\"STRING\",\"nullable\":true,\"unknown\":123}";

        ColumnResponse r = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(r);
        assertEquals("colE", r.getName());
        assertEquals("STRING", r.getType());

    @Test

        String json = "{\"name\":\"colF\",\"type\":\"STRING\",\"nullable\":false,\"extra\":{\"a\":1,\"b\":[1,2]}}";

        ColumnResponse r = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(r);

    @Test

        ColumnResponse a = new ColumnResponse("n", "t", Boolean.TRUE);
        String s = a.toString();
        assertNotNull(s);
        assertTrue(s.contains("name='n'") || s.contains("name=\"n\"") || s.contains("name"));

    @Test
    void tc21_volume_deserializeLargeArray_10k() throws Exception {
        ObjectMapper mapper = mapperFailOnUnknownDisabled();

        int n = 10_000;
        StringBuilder sb = new StringBuilder(n * 64);
        sb.append("[");
        for (int i = 0; i < n; i++) {
            sb.append("{\"name\":\"c").append(i).append("\",\"type\":\"STRING\",\"nullable\":true}");
            if (i < n - 1) sb.append(",");
        }
        sb.append("]");

        ColumnResponse[] arr = mapper.readValue(sb.toString(), ColumnResponse[].class);
        assertNotNull(arr);
        assertEquals(n, arr.length);
        assertEquals("c0", arr[0].getName());
        assertEquals("c9999", arr[n - 1].getName());
    }
