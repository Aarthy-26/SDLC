package com.aava.datastudio;

/**
 * Unit/integration tests for US-94727 API error handling.
 *
 * Security/enterprise notes:
 * - Assert JSON Content-Type for error responses.
 * - Assert sanitized 500 responses do not leak internal exception messages.
 * - Avoid logging sensitive values in assertions.

@SpringBootTest
@AutoConfigureMockMvc
class GlobalExceptionHandlerInventoryTest {

    @Autowired
    private MockMvc mockMvc;

    @Test
    @DisplayName("1 - POST /api/widgets happy path returns widget")
    void createWidget_happyPath_returns200AndWidgetResponse() throws Exception {
        mockMvc.perform(post("/api/widgets")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content("{\"name\":\"Widget-A\"}"))
                .andExpect(status().isOk())
                .andExpect(content().contentTypeCompatibleWith(MediaType.APPLICATION_JSON))
                .andExpect(jsonPath("$.id", not(isBlankOrNullString())))
                .andExpect(jsonPath("$.name").value("Widget-A"));
    }

    @Test
    @DisplayName("2 - Validation error: empty name")
    void createWidget_emptyName_returns400ValidationEnvelope() throws Exception {
        mockMvc.perform(post("/api/widgets")
                        .contentType(MediaType.APPLICATION_JSON)
                        .content("{\"name\":\"\"}"))
                .andExpect(status().isBadRequest())
                .andExpect(content().contentTypeCompatibleWith(MediaType.APPLICATION_JSON))
                .andExpect(jsonPath("$.code").value("VALIDATION_ERROR"))
                .andExpect(jsonPath("$.message").value("Request validation failed"))
                .andExpect(jsonPath("$.timestamp", notNullValue()))
                .andExpect(jsonPath("$.path").value("/api/widgets"))
                .andExpect(jsonPath("$.fieldErrors", isA(List.class)))
                .andExpect(jsonPath("$.fieldErrors", not(empty())))
                .andExpect(jsonPath("$.fieldErrors[*].field", hasItem("name")));
    }

    @Test
    @DisplayName("3 - Validation error: missing name key")
    void createWidget_missingName_returns400ValidationEnvelope() throws Exception {

                .andExpect(status().isBadRequest())

                .andExpect(jsonPath("$.path").value("/api/widgets"))

    @Test

    void apiErrorResponse_ofWithFields_setsFieldErrors() {
        Instant now = Instant.now();
        List<ApiErrorResponse.FieldErrorItem> fes = List.of(new ApiErrorResponse.FieldErrorItem("f", "e"));
        ApiErrorResponse r = ApiErrorResponse.ofWithFields("C", "M", now, "/p", fes);

        assertEquals(now, r.getTimestamp());
        assertEquals("/p", r.getPath());
        assertNotNull(r.getFieldErrors());
        assertEquals(1, r.getFieldErrors().size());

        assertEquals("e", r.getFieldErrors().get(0).getError());
