    @Test
    void tc01_process_validRequest_normalizesAndDedupes() {
        var service = newService();

        List<ImsCodeRefactoring.Operation> ops = List.of(
                new ImsCodeRefactoring.Operation(ImsCodeRefactoring.OperationType.TRIM, " module_name ", Optional.empty()),
                new ImsCodeRefactoring.Operation(ImsCodeRefactoring.OperationType.STANDARDIZE_KEY, "customer_id", Optional.empty()),
                new ImsCodeRefactoring.Operation(ImsCodeRefactoring.OperationType.REMOVE_DUPLICATES, "tags", Optional.empty()),
                new ImsCodeRefactoring.Operation(ImsCodeRefactoring.OperationType.REMOVE_DUPLICATES, "tags", Optional.empty())
        );

        var response = service.process(req("REQ-1", "user", Instant.parse("2026-01-01T00:00:00Z"), " IMS Processing ", ops));

        assertTrue(response.success());
        assertEquals("REQ-1", response.requestId());
        assertNotNull(response.data());
        assertEquals("IMS Processing", response.data().moduleName());
        assertEquals(3, response.data().operationsApplied());
        assertEquals(1, response.data().duplicateOperationsRemoved());
        assertEquals(List.of("Processed using shared IMS refactoring pipeline"), response.messages());

        // summary and changes align to distinct operations
        assertEquals(3, response.data().changes().size());
        assertTrue(response.data().operationSummary().containsKey(ImsCodeRefactoring.OperationType.TRIM));
        assertTrue(response.data().operationSummary().containsKey(ImsCodeRefactoring.OperationType.STANDARDIZE_KEY));
        assertTrue(response.data().operationSummary().containsKey(ImsCodeRefactoring.OperationType.REMOVE_DUPLICATES));

        // duplicate note present
        assertTrue(response.data().notes().stream().anyMatch(n -> n.contains("Removed 1 duplicate")));
    }

            // bypass compact constructor by using reflection still calls constructor; but keep as documentation placeholder
