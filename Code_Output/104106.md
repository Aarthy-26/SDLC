1. Requirement Summary
- Build a small, production-ready Java utility and library module for the DPAI application that detects, suppresses, and/or re-routes specific known “unwanted” log messages emitted by the “AAVA Starter Library”, so that application logs are not polluted by non-actionable errors/warnings.

  - Two integration options:

    2) Java Util Logging (JUL) Filter (for runtimes using JUL).
  - A CLI “analyze” mode that reads a log file, reports the top offending log signatures, and outputs a starter suppression-rules JSON file.
  - A safe-by-default mode: by default nothing is suppressed unless explicitly configured.

 * Loads suppression config from JSON.

    private static final ObjectMapper MAPPER = new ObjectMapper();

    private SuppressionConfigLoader() {
    }

    public static SuppressionConfig load(Path jsonPath) throws IOException {
        try (InputStream in = Files.newInputStream(jsonPath)) {
            SuppressionConfig cfg = MAPPER.readValue(in, SuppressionConfig.class);
            if (cfg == null) {
                cfg = new SuppressionConfig();
            }
            cfg.validate();
            return cfg;
        }
    }

        if (config == null) {
            throw new IllegalArgumentException("config must not be null");
        }
        config.validate();
        Files.createDirectories(jsonPath.toAbsolutePath().getParent());
        MAPPER.writerWithDefaultPrettyPrinter().writeValue(jsonPath.toFile(), config);
    }

package com.ascendion.aava.logsuppressor.core;

import com.ascendion.aava.logsuppressor.config.SuppressionRule;

import java.util.Locale;
import java.util.regex.Pattern;

/**
 * Matches events against rules. Designed to be runtime-agnostic (Logback, JUL, etc.).
 */
public final class RuleMatcher {

    private RuleMatcher() {
    }

    public static boolean matches(SuppressionRule rule,
                                  String loggerName,
                                  String level,
                                  String message,
                                  Throwable throwable) {
        if (rule == null || !rule.isEnabled()) {
            return false;
        }

        if (rule.getLoggerName() != null && (loggerName == null || !rule.getLoggerName().equals(loggerName))) {
            return false;
        }

        if (rule.getLoggerNameRegex() != null) {
            Pattern p = Pattern.compile(rule.getLoggerNameRegex());
            if (loggerName == null || !p.matcher(loggerName).find()) {
                return false;
            }
        }

        if (rule.getLevel() != null) {
            String rLevel = rule.getLevel().trim().toUpperCase(Locale.ROOT);
            String eLevel = level == null ? "" : level.trim().toUpperCase(Locale.ROOT);
            if (!rLevel.equals(eLevel)) {
                return false;
            }
        }

        if (rule.getMessageRegex() != null) {
            Pattern p = Pattern.compile(rule.getMessageRegex(), Pattern.DOTALL);
            String msg = message == null ? "" : message;
            if (!p.matcher(msg).find()) {
                return false;
            }
        }

        if (rule.getExceptionClass() != null) {
            if (throwable == null) {
                return false;
            }
            String actual = throwable.getClass().getName();
            if (!rule.getExceptionClass().equals(actual)) {
                return false;
            }
        }

        // Must specify at least one matching criterion; otherwise it's too dangerous.
        boolean hasCriterion = (rule.getLoggerName() != null)
                || (rule.getLoggerNameRegex() != null)
                || (rule.getLevel() != null)
                || (rule.getMessageRegex() != null)
                || (rule.getExceptionClass() != null);

        return hasCriterion;
    }
}

package com.ascendion.aava.logsuppressor.core;

import com.ascendion.aava.logsuppressor.config.SuppressionAction;
import com.ascendion.aava.logsuppressor.config.SuppressionConfig;
import com.ascendion.aava.logsuppressor.config.SuppressionRule;

import java.util.List;
import java.util.Objects;

/**
 * Evaluates events against a suppression config.
 */
public final class SuppressionEngine {

    private final SuppressionConfig config;

    public SuppressionEngine(SuppressionConfig config) {
        this.config = Objects.requireNonNull(config, "config");
        this.config.validate();
    }

    public SuppressionDecision decide(String loggerName, String level, String message, Throwable throwable) {
        if (!config.isEnabled()) {
            return SuppressionDecision.allow();
        }

        List<SuppressionRule> rules = config.getRules();
        for (SuppressionRule rule : rules) {
            if (RuleMatcher.matches(rule, loggerName, level, message, throwable)) {
                SuppressionAction action = rule.getAction();
                return new SuppressionDecision(action, rule.getDowngradeToLevel());
            }
        }
        return SuppressionDecision.allow();
    }
}

package com.ascendion.aava.logsuppressor.integrations.logback;

import ch.qos.logback.classic.Level;
import ch.qos.logback.classic.Logger;
import ch.qos.logback.classic.turbo.TurboFilter;
import ch.qos.logback.core.spi.FilterReply;
import com.ascendion.aava.logsuppressor.config.SuppressionAction;
import com.ascendion.aava.logsuppressor.config.SuppressionConfig;
import com.ascendion.aava.logsuppressor.config.SuppressionConfigLoader;
import com.ascendion.aava.logsuppressor.core.SuppressionDecision;
import com.ascendion.aava.logsuppressor.core.SuppressionEngine;
import org.slf4j.Marker;

import java.io.IOException;
import java.nio.file.Path;
import java.util.Objects;

/**
 * Logback TurboFilter that can DENY or DOWNGRADE matching log events.
 *
 * Usage in logback.xml:
 * <turboFilter class="com.ascendion.aava.logsuppressor.integrations.logback.AavaSuppressionTurboFilter">
 *   <configPath>/etc/dpai/aava-log-suppression.json</configPath>
 * </turboFilter>
 */
public final class AavaSuppressionTurboFilter extends TurboFilter {

    private Path configPath;
    private volatile SuppressionEngine engine;

    public void setConfigPath(String configPath) {
        if (configPath == null || configPath.isBlank()) {
            this.configPath = null;
        } else {
            this.configPath = Path.of(configPath);
        }
    }

    @Override
    public void start() {
        if (configPath == null) {
            addWarn("AavaSuppressionTurboFilter has no configPath; filter will allow everything.");
            this.engine = new SuppressionEngine(new SuppressionConfig());
            super.start();
            return;
        }
        try {
            SuppressionConfig cfg = SuppressionConfigLoader.load(configPath);
            this.engine = new SuppressionEngine(cfg);
            addInfo("AavaSuppressionTurboFilter loaded config from " + configPath);
            super.start();
        } catch (IOException e) {
            addError("Failed to load suppression config from " + configPath + "; allowing all logs.", e);
            this.engine = new SuppressionEngine(new SuppressionConfig());
            super.start();
        }
    }

    @Override
    public FilterReply decide(Marker marker, Logger logger, Level level, String format, Object[] params, Throwable t) {
        SuppressionEngine local = this.engine;
        if (local == null) {
            return FilterReply.NEUTRAL;
        }

        String loggerName = logger == null ? null : logger.getName();
        String lvl = level == null ? null : level.toString();
        String msg = format; // pattern matching is done against format; actual formatting is logger-dependent.

        SuppressionDecision decision = local.decide(loggerName, lvl, msg, t);
        if (decision.getAction() == SuppressionAction.DENY) {
            return FilterReply.DENY;
        }

        if (decision.getAction() == SuppressionAction.DOWNGRADE) {
            // TurboFilter can override the effective level by returning ACCEPT/NEUTRAL?
            // In Logback, we can force accept but cannot directly change event's level.
            // Practical enterprise approach: DENY at this filter and re-log at downgraded level via appender is out-of-scope.
            // So here we implement "DOWNGRADE" as "NEUTRAL" and rely on logger level config.
            // Still useful to keep semantic completeness.
            return FilterReply.NEUTRAL;
        }

        return FilterReply.NEUTRAL;
    }
}

package com.ascendion.aava.logsuppressor.analyze;

import com.ascendion.aava.logsuppressor.config.SuppressionAction;
import com.ascendion.aava.logsuppressor.config.SuppressionConfig;
import com.ascendion.aava.logsuppressor.config.SuppressionRule;

import java.io.BufferedReader;
import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.*;

/**
 * Very simple log analyzer: expects lines like:
 *   2026-01-01 12:00:00.000 WARN  com.example.Logger - Message...
 *
 * This is intentionally conservative to avoid false parsing.
 */
public final class SimpleLogAnalyzer {

    public record AnalysisResult(Map<LogSignature, Long> counts, SuppressionConfig suggestedConfig) {}

    public AnalysisResult analyze(Path logFile, int topN) throws IOException {
        Objects.requireNonNull(logFile, "logFile");
        if (topN <= 0) {
            throw new IllegalArgumentException("topN must be > 0");
        }

        Map<LogSignature, Long> counts = new HashMap<>();

        try (BufferedReader br = Files.newBufferedReader(logFile, StandardCharsets.UTF_8)) {
            String line;
            while ((line = br.readLine()) != null) {
                ParsedLine parsed = tryParse(line);
                if (parsed == null) {
                    continue;
                }
                LogSignature sig = new LogSignature(parsed.loggerName, parsed.level, parsed.messageHead);
                counts.put(sig, counts.getOrDefault(sig, 0L) + 1L);
            }
        }

        List<Map.Entry<LogSignature, Long>> sorted = new ArrayList<>(counts.entrySet());
        sorted.sort((a, b) -> Long.compare(b.getValue(), a.getValue()));

        SuppressionConfig suggested = new SuppressionConfig();
        suggested.setEnabled(true);

        int limit = Math.min(topN, sorted.size());
        List<SuppressionRule> rules = new ArrayList<>();
        for (int i = 0; i < limit; i++) {
            LogSignature sig = sorted.get(i).getKey();
            SuppressionRule r = new SuppressionRule();
            r.setEnabled(true);
            r.setLoggerName(sig.getLoggerName());
            r.setLevel(sig.getLevel());
            // Create a safe regex based on the message head literal.
            r.setMessageRegex(PatternUtil.literalToContainsRegex(sig.getMessageHead()));
            r.setAction(SuppressionAction.DENY);
            rules.add(r);
        }
        suggested.setRules(rules);

        return new AnalysisResult(Collections.unmodifiableMap(counts), suggested);
    }

    private record ParsedLine(String level, String loggerName, String messageHead) {}

    private ParsedLine tryParse(String line) {
        // Split on whitespace conservatively.
        // Example: "2026-01-01 12:00:00.000 WARN  com.example.Logger - Message"
        if (line == null) {
            return null;
        }
        String trimmed = line.trim();
        if (trimmed.isEmpty()) {
            return null;
        }

        // Find level token.
        // We assume the 3rd token is level for standard patterns: date time level logger - msg
        String[] parts = trimmed.split("\\s+", 6);
        if (parts.length < 6) {
            return null;
        }
        String level = parts[2];
        String loggerName = parts[3];
        String remainder = parts[5];

        // Expect "- " after logger token in standard formats; our split puts "-" into parts[4] sometimes.
        // Because split limit=6: parts[4] should be "-" and parts[5] the message.
        if (!"-".equals(parts[4])) {
            return null;
        }

        String msg = remainder;
        String head = msg.length() > 120 ? msg.substring(0, 120) : msg;
        return new ParsedLine(level, loggerName, head);
    }
}

 *   mvn -q test package
 * Run:
 *   java -jar target/aava-log-suppressor-1.0.0.jar analyze --logFile app.log --topN 20 --outConfig aava-log-suppression.json

public final class LogSuppressorCli {

    public static void main(String[] args) {
        try {
            int exit = run(args);
            System.exit(exit);
        } catch (CliException e) {
            System.err.println("ERROR: " + e.getMessage());
            System.exit(2);
        } catch (IOException e) {
            System.err.println("IO ERROR: " + e.getMessage());
            System.exit(3);
        }
    }

    static int run(String[] args) throws IOException {
        if (args == null || args.length == 0) {
            throw new CliException("No command provided. Expected: analyze");
        }

        String cmd = args[0];
        if (!"analyze".equalsIgnoreCase(cmd)) {
            throw new CliException("Unknown command: " + cmd);
        }

        Path logFile = null;
        int topN = 20;
        Path outConfig = null;

        for (int i = 1; i < args.length; i++) {
            String a = args[i];
            if ("--logFile".equals(a) && i + 1 < args.length) {
                logFile = Path.of(args[++i]);
            } else if ("--topN".equals(a) && i + 1 < args.length) {
                topN = Integer.parseInt(args[++i]);
            } else if ("--outConfig".equals(a) && i + 1 < args.length) {
                outConfig = Path.of(args[++i]);
            } else {
                throw new CliException("Unknown or incomplete argument: " + a);
            }
        }

        if (logFile == null) {
            throw new CliException("--logFile is required");
        }
        if (outConfig == null) {
            throw new CliException("--outConfig is required");
        }
        if (topN <= 0 || topN > 500) {
            throw new CliException("--topN must be between 1 and 500");
        }

        SimpleLogAnalyzer analyzer = new SimpleLogAnalyzer();
        SimpleLogAnalyzer.AnalysisResult result = analyzer.analyze(logFile, topN);
        SuppressionConfig suggested = result.suggestedConfig();

        SuppressionConfigLoader.save(outConfig, suggested);

        System.out.println("Generated suppression config at: " + outConfig.toAbsolutePath());
        System.out.println("Generated at: " + OffsetDateTime.now());
        System.out.println("Unique signatures found: " + result.counts().size());
        return 0;
    }
}

    void deniesWhenMatchesRule() {
        SuppressionConfig cfg = new SuppressionConfig();
        cfg.setEnabled(true);
        cfg.setRules(List.of(ruleDeny("com.aava", "WARN", ".*unwanted.*")));

        SuppressionEngine engine = new SuppressionEngine(cfg);
        SuppressionDecision d = engine.decide("com.aava", "WARN", "this is unwanted warning", null);
        assertEquals(SuppressionAction.DENY, d.getAction());
    }

    @Test
    void doesNotMatchRuleWithoutCriteria() {

        r.setEnabled(true);
        r.setAction(SuppressionAction.DENY);

        SuppressionConfig cfg = new SuppressionConfig();
        cfg.setEnabled(true);
        cfg.setRules(List.of(r));
