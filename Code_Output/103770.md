Execute automated data profiling and diagnostic scans across seven source tables in the `staging` schema (`mod_record`, `baseline_record`, `impact_record`, `task_record`, `subtask`, `completion_gate`, `audit_event`) to capture per-column statistical metrics and detect high-risk data quality issues prior to transforming data into PostgreSQL.

- Database connectivity parameters are provided via environment variables (no secrets in code): `DB_URL`, `DB_USER`, `DB_PASSWORD`.
- Foreign key relationships are inferred by name and acceptance criteria:
  - `subtask.task_record_id` → `task_record.id`
  - `completion_gate.task_record_id` → `task_record.id`
  - `impact_record.mod_record_id` → `mod_record.id`
  (If actual PK/FK columns differ, update the relationship config in code.)
- High-risk checks implemented:
  - Missing mandatory `mod_ref` in `mod_record` (null/blank)
  - Invalid timestamp formats for columns of string type whose name contains `time`, `timestamp`, or ends with `_ts` (validated against ISO-8601) 
  - Unmapped sensitivity codes in any column named `sensitivity_code` (allowed set configurable; default: `PUBLIC`, `INTERNAL`, `CONFIDENTIAL`, `RESTRICTED`)
- “Date ranges” are computed for date/time typed columns (DATE/TIMESTAMP/TIMESTAMPTZ) using `MIN`/`MAX`.

/**
 * Command-line data profiling and diagnostic scan tool for PostgreSQL.
 *
 * <p>Runs profiling across the 7 specified tables in schema {@code staging}, capturing:
 * total row counts, per-column null counts/percentages, distinct cardinality,
 * min/max string lengths, and date ranges. Also executes orphan FK checks and
 * highlights high-risk anomalies.
 *
 * <p>Run:
 * <pre>
 *   export DB_URL='jdbc:postgresql://localhost:5432/mydb'
 *   export DB_USER='myuser'
 *   export DB_PASSWORD='mypassword'
 *   mvn -q -DskipTests package
 *   java -jar target/data-profiler-1.0.0.jar
 * </pre>

public final class ProfilerApp {

    private static final String SCHEMA = "staging";

    private static final List<String> TABLES = List.of(
            "mod_record",
            "baseline_record",
            "impact_record",
            "task_record",
            "subtask",
            "completion_gate",
            "audit_event"
    );

    /** Default allowed sensitivity codes; update if your domain differs. */
    private static final Set<String> ALLOWED_SENSITIVITY_CODES = Set.of(
            "PUBLIC", "INTERNAL", "CONFIDENTIAL", "RESTRICTED"

    private ProfilerApp() {
    }

    public static void main(String[] args) throws Exception {
        String dbUrl = Env.require("DB_URL");
        String dbUser = Env.require("DB_USER");
        String dbPassword = Env.require("DB_PASSWORD");

        ProfileReport report = new ProfileReport();
        report.setUserStoryId("103770");
        report.setGeneratedAt(Instant.now());
        report.setSchema(SCHEMA);

        try (Connection connection = DriverManager.getConnection(dbUrl, dbUser, dbPassword)) {
            connection.setReadOnly(true);

            List<DataQualityIssue> issues = new ArrayList<>();
            List<ForeignKeyCheck> fkChecks = new ArrayList<>();

                tableProfiles.add(profileTable(connection, SCHEMA, table, issues));
            }

            // Orphan record checks (configurable if PK/FK columns differ)

                    new ForeignKeyCheck("subtask", "task_record", "task_record_id", "id"), issues));
            fkChecks.add(checkOrphans(connection,
                    new ForeignKeyCheck("completion_gate", "task_record", "task_record_id", "id"), issues));
            fkChecks.add(checkOrphans(connection,
                    new ForeignKeyCheck("impact_record", "mod_record", "mod_record_id", "id"), issues));

            report.setTables(tableProfiles);
            report.setForeignKeyChecks(fkChecks);
            report.setIssues(issues);

        mapper.writeValue(out, report);

    private static TableProfile profileTable(Connection connection,
                                             String schema,
                                             String table,
                                             List<DataQualityIssue> issues) throws SQLException {

        String qualified = qIdent(schema) + "." + qIdent(table);

        long totalRows = queryForLong(connection, "SELECT COUNT(*) FROM " + qualified);

        TableProfile tp = new TableProfile();
        tp.setSchema(schema);
        tp.setTable(table);
        tp.setTotalRows(totalRows);

        List<ColumnProfile> colProfiles = new ArrayList<>();

        List<ColumnMeta> cols = listColumns(connection, schema, table);
        for (ColumnMeta col : cols) {
            ColumnProfile cp = new ColumnProfile();
            cp.setColumn(col.columnName);
            cp.setDataType(col.dataType);
            cp.setJdbcType(col.jdbcType);

            // Null counts
            String nullCountSql = "SELECT COUNT(*) FROM " + qualified + " WHERE " + qIdent(col.columnName) + " IS NULL";
            long nullCount = queryForLong(connection, nullCountSql);
            cp.setNullCount(nullCount);
            cp.setNullPercentage(totalRows == 0 ? 0.0d : (nullCount * 100.0d) / totalRows);

            // Distinct values
            String distinctSql = "SELECT COUNT(DISTINCT " + qIdent(col.columnName) + ") FROM " + qualified;
            long distinct = queryForLong(connection, distinctSql);
            cp.setDistinctCount(distinct);

            // String length metrics
            if (isStringType(col.jdbcType)) {
                String lenSql = "SELECT MIN(LENGTH(" + qIdent(col.columnName) + ")), MAX(LENGTH(" + qIdent(col.columnName) + ")) " +
                        "FROM " + qualified + " WHERE " + qIdent(col.columnName) + " IS NOT NULL";
                try (PreparedStatement ps = connection.prepareStatement(lenSql);
                     ResultSet rs = ps.executeQuery()) {
                    if (rs.next()) {
                        cp.setMinStringLength((Integer) rs.getObject(1));
                        cp.setMaxStringLength((Integer) rs.getObject(2));
                    }
                }

                // Timestamp-format anomaly scan (only for string columns with likely timestamp name)
                if (looksLikeTimestampColumn(col.columnName)) {
                    List<String> badSamples = findInvalidIso8601Samples(connection, qualified, col.columnName, 10);
                    if (!badSamples.isEmpty()) {
                        DataQualityIssue issue = DataQualityIssue.highRisk(
                                schema,
                                table,
                                col.columnName,
                                "INVALID_TIMESTAMP_FORMAT",
                                "Found non-ISO-8601 timestamp strings in column: " + col.columnName,
                                badSamples
                        );
                        issues.add(issue);
                    }
                }

                // Sensitivity code validation
                if ("sensitivity_code".equalsIgnoreCase(col.columnName)) {
                    List<String> unmapped = findUnmappedSensitivityCodes(connection, qualified, col.columnName, 20);
                    if (!unmapped.isEmpty()) {
                        DataQualityIssue issue = DataQualityIssue.highRisk(
                                schema,
                                table,
                                col.columnName,
                                "UNMAPPED_SENSITIVITY_CODE",
                                "Found sensitivity codes outside allowed set: " + ALLOWED_SENSITIVITY_CODES,
                                unmapped
                        );
                        issues.add(issue);
                    }
                }

            }

            // Date ranges for date/time types
            if (isDateTimeType(col.jdbcType)) {
                String rangeSql = "SELECT MIN(" + qIdent(col.columnName) + "), MAX(" + qIdent(col.columnName) + ") " +
                        "FROM " + qualified + " WHERE " + qIdent(col.columnName) + " IS NOT NULL";
                try (PreparedStatement ps = connection.prepareStatement(rangeSql);
                     ResultSet rs = ps.executeQuery()) {
                    if (rs.next()) {
                        cp.setMinDateTime(stringifyTemporal(rs.getObject(1)));
                        cp.setMaxDateTime(stringifyTemporal(rs.getObject(2)));
                    }
                }
            }

            colProfiles.add(cp);
        }

        tp.setColumns(colProfiles);

        // High-risk issue: missing mandatory mod_ref in mod_record
        if ("mod_record".equalsIgnoreCase(table) && hasColumn(cols, "mod_ref")) {
            String sql = "SELECT COUNT(*) FROM " + qualified + " WHERE mod_ref IS NULL OR BTRIM(mod_ref) = ''";
            long missing = queryForLong(connection, sql);
            if (missing > 0) {
                issues.add(DataQualityIssue.highRisk(
                        schema,
                        table,
                        "mod_ref",
                        "MISSING_MANDATORY_MOD_REF",
                        "mod_ref is missing (NULL/blank) in " + missing + " row(s)",
                        List.of(String.valueOf(missing))
                ));
            }
        }

        return tp;
    }

    private static ForeignKeyCheck checkOrphans(Connection connection,
                                                ForeignKeyCheck check,
                                                List<DataQualityIssue> issues) throws SQLException {
        Objects.requireNonNull(check, "check");

        String childQualified = qIdent(SCHEMA) + "." + qIdent(check.getChildTable());
        String parentQualified = qIdent(SCHEMA) + "." + qIdent(check.getParentTable());

        // Orphans: child rows whose FK is not null and does not exist in parent PK
        String sql = "SELECT COUNT(*) FROM " + childQualified + " c " +
                "LEFT JOIN " + parentQualified + " p ON c." + qIdent(check.getChildFkColumn()) + " = p." + qIdent(check.getParentPkColumn()) + " " +
                "WHERE c." + qIdent(check.getChildFkColumn()) + " IS NOT NULL AND p." + qIdent(check.getParentPkColumn()) + " IS NULL";

        long orphanCount = queryForLong(connection, sql);
        check.setOrphanCount(orphanCount);

        if (orphanCount > 0) {
            issues.add(DataQualityIssue.highRisk(
                    SCHEMA,
                    check.getChildTable(),
                    check.getChildFkColumn(),
                    "ORPHAN_FOREIGN_KEYS",
                    "Orphan foreign keys detected: " + check.getChildTable() + "." + check.getChildFkColumn() + " -> " + check.getParentTable() + "." + check.getParentPkColumn(),
                    List.of(String.valueOf(orphanCount))
            ));
        }

        return check;
    }

    private static List<String> findInvalidIso8601Samples(Connection connection,
                                                          String qualifiedTable,
                                                          String column,
                                                          int limit) throws SQLException {
        if (limit <= 0) {
            throw new IllegalArgumentException("limit must be > 0");
        }

        String sql = "SELECT " + qIdent(column) + " FROM " + qualifiedTable + " " +
                "WHERE " + qIdent(column) + " IS NOT NULL " +
                "LIMIT 5000";

        List<String> bad = new ArrayList<>();
        try (PreparedStatement ps = connection.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            while (rs.next()) {
                String v = rs.getString(1);
                if (v == null) {
                    continue;
                }
                String trimmed = v.trim();
                if (trimmed.isEmpty()) {
                    continue;
                }
                if (!Iso8601.isValid(trimmed)) {
                    bad.add(trimmed);
                    if (bad.size() >= limit) {
                        break;
                    }
                }
            }
        }
        return bad;
    }

    private static List<String> findUnmappedSensitivityCodes(Connection connection,
                                                             String qualifiedTable,
                                                             String column,
                                                             int limit) throws SQLException {
        if (limit <= 0) {
            throw new IllegalArgumentException("limit must be > 0");
        }

        // Fetch distinct codes and filter in Java to avoid dynamic IN lists.
        String sql = "SELECT DISTINCT " + qIdent(column) + " FROM " + qualifiedTable + " WHERE " + qIdent(column) + " IS NOT NULL";
        List<String> unmapped = new ArrayList<>();
        try (PreparedStatement ps = connection.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            while (rs.next()) {
                String code = rs.getString(1);
                if (code == null) {
                    continue;
                }
                String normalized = code.trim().toUpperCase(Locale.ROOT);
                if (normalized.isEmpty()) {
                    continue;
                }
                if (!ALLOWED_SENSITIVITY_CODES.contains(normalized)) {
                    unmapped.add(code);
                    if (unmapped.size() >= limit) {
                        break;
                    }
                }
            }
        }
        return unmapped;
    }

    private static boolean looksLikeTimestampColumn(String name) {
        String n = name.toLowerCase(Locale.ROOT);
        return n.contains("timestamp") || n.contains("time") || n.endsWith("_ts");
    }

    private static boolean hasColumn(List<ColumnMeta> cols, String name) {
        for (ColumnMeta c : cols) {
            if (c.columnName.equalsIgnoreCase(name)) {
                return true;
            }
        }
        return false;
    }

    private static boolean isStringType(int jdbcType) {
        return jdbcType == Types.VARCHAR || jdbcType == Types.CHAR || jdbcType == Types.LONGVARCHAR
                || jdbcType == Types.NVARCHAR || jdbcType == Types.NCHAR || jdbcType == Types.LONGNVARCHAR;
    }

    private static boolean isDateTimeType(int jdbcType) {
        return jdbcType == Types.DATE || jdbcType == Types.TIMESTAMP || jdbcType == Types.TIMESTAMP_WITH_TIMEZONE || jdbcType == Types.TIME;
    }

    private static String stringifyTemporal(Object o) {
        if (o == null) {
            return null;
        }
        if (o instanceof Timestamp ts) {
            return ts.toInstant().toString();
        }
        if (o instanceof Date d) {
            return d.toString();
        }
        if (o instanceof LocalDate ld) {
            return ld.toString();
        }
        if (o instanceof OffsetDateTime odt) {
            return odt.toString();
        }
        return String.valueOf(o);
    }

    private static long queryForLong(Connection connection, String sql) throws SQLException {
        try (PreparedStatement ps = connection.prepareStatement(sql);
             ResultSet rs = ps.executeQuery()) {
            if (!rs.next()) {
                throw new SQLException("No rows returned for query: " + sql);
            }
            return rs.getLong(1);
        }
    }

    private static List<ColumnMeta> listColumns(Connection connection, String schema, String table) throws SQLException {
        DatabaseMetaData md = connection.getMetaData();
        List<ColumnMeta> cols = new ArrayList<>();
        try (ResultSet rs = md.getColumns(null, schema, table, null)) {
            while (rs.next()) {
                String columnName = rs.getString("COLUMN_NAME");
                String typeName = rs.getString("TYPE_NAME");
                int dataType = rs.getInt("DATA_TYPE");
                cols.add(new ColumnMeta(columnName, typeName, dataType));
            }
        }
        cols.sort(Comparator.comparing(c -> c.columnName.toLowerCase(Locale.ROOT)));
        return cols;
    }

    /**
     * Safely quotes SQL identifiers for PostgreSQL.
     *
     * <p>This prevents issues with reserved words/special characters.
     */
    private static String qIdent(String ident) {
        if (ident == null || ident.isBlank()) {
            throw new IllegalArgumentException("Identifier must not be null/blank");
        }
        // Double-quote and escape any embedded quotes.
        return "\"" + ident.replace("\"", "\"\"") + "\"";
    }

    private record ColumnMeta(String columnName, String dataType, int jdbcType) {
    }
}

package com.aava.datastudio.profiler.model;

import java.util.ArrayList;
import java.util.List;
import java.util.Objects;

/**
 * A detected data quality issue.
 *
 * <p>Severity levels are modeled as strings to keep JSON output simple.
 */
public final class DataQualityIssue {
    private String severity; // e.g., HIGH
    private String schema;
    private String table;
    private String column;
    private String code;
    private String message;
    private List<String> samples = new ArrayList<>();

    public static DataQualityIssue highRisk(String schema,
                                            String table,
                                            String column,
                                            String code,
                                            String message,
                                            List<String> samples) {
        Objects.requireNonNull(code, "code");
        Objects.requireNonNull(message, "message");

        DataQualityIssue i = new DataQualityIssue();
        i.setSeverity("HIGH");
        i.setSchema(schema);
        i.setTable(table);
        i.setColumn(column);
        i.setCode(code);
        i.setMessage(message);
        if (samples != null) {
            i.setSamples(samples);
        }
        return i;
    }

    public String getSeverity() {
        return severity;

/** Environment variable helper to avoid hardcoding secrets. */
public final class Env {
    private Env() {
    }

    /**
     * Reads a required environment variable.
     *
     * @throws IllegalStateException if missing/blank
     */
    public static String require(String name) {
        if (name == null || name.isBlank()) {
            throw new IllegalArgumentException("name must not be null/blank");
        }
        String v = System.getenv(name);
        if (v == null || v.isBlank()) {
            throw new IllegalStateException("Missing required environment variable: " + name);
        }
        return v;
    }
}

package com.aava.datastudio.profiler.util;

import java.time.Instant;
import java.time.OffsetDateTime;
import java.time.format.DateTimeParseException;

/** ISO-8601 parsing utility for timestamp format checks. */
public final class Iso8601 {
    private Iso8601() {
    }

    /**
     * Validates whether a string can be parsed as ISO-8601 instant or offset date-time.
     *
     * <p>Examples accepted:
     * <ul>
     *   <li>2024-01-31T12:34:56Z</li>
     *   <li>2024-01-31T12:34:56.123Z</li>
     *   <li>2024-01-31T12:34:56+05:30</li>
     * </ul>
     */
    public static boolean isValid(String value) {
        if (value == null) {
            return false;
        }
        String v = value.trim();
        if (v.isEmpty()) {
            return false;
        }

        try {
            Instant.parse(v);
            return true;
        } catch (DateTimeParseException ignored) {
            // fall through
        }

        try {
            OffsetDateTime.parse(v);
            return true;
        } catch (DateTimeParseException ignored) {
            return false;
        }
    }
}
