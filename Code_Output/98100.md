        private static String sanitizeForId(String raw) {
            // Strictly allow [A-Za-z0-9_-] for safety; replace others.
            StringBuilder out = new StringBuilder(raw.length());
            for (int i = 0; i < raw.length(); i++) {
                char c = raw.charAt(i);
                if ((c >= 'a' && c <= 'z')

                        || (c >= '0' && c <= '9')

        public void selectRowById(String rowId) {

            if (selectionListener != null) {

        private Optional<TableRow> findRow(String rowId) {
