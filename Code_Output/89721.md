1. Requirement Summary

Assumptions
1) The story describes adding two MCP tools to an existing “pbip-mcp-server”. Because the existing server codebase and protocol implementation are not provided here, this deliverable implements a production-ready standalone Java 21 (Spring Boot) MCP-compatible HTTP server that exposes equivalent capabilities via MCP-style endpoints: `POST /tools/list` and `POST /tools/call`. This is the closest useful, runnable artifact.
2) Input is a local filesystem `projectPath` pointing to a PBIP project folder. The server runs with least privilege and reads only from that folder.
3) PBIP structures vary. The implementation discovers report/model files by scanning known PBIP-related extensions and folder names:
   - Report metadata: `.pbir` / `.pbir.json` / `report.json` (and fallback scan for `*.pbir*`)
   - Semantic model (TMDL): `.tmdl` / `.tmsl` / `model.bim` / `database.json` (fallback scan for `*.tmdl*`, `*.bim`, `*.tmsl`)
   If multiple candidates exist, the newest (by lastModifiedTime) is used.
4) “Real values only — nothing inferred or fabricated”: the server returns only fields actually present in parsed files. Missing/unknown fields are omitted (not defaulted) and lists may be empty if not present.
5) Theme/resourcePackage: returned only if explicitly found in report JSON as `theme` / `resourcePackage` (case-insensitive search of known locations) or in a `Theme` file inside the report directory.
6) “Power Query parameters where resolvable”: parameters are returned when found in model JSON under known keys (e.g., `parameters`, `sharedExpressions`) or in `.pq`/`m` sources under `queries`. If parameters cannot be parsed deterministically, they are omitted.
7) Security: no authentication requirement is stated, so the server includes:
   - path traversal protection (projectPath must resolve under an allowed base directory configured by `PBIP_BASE_DIR` or defaults to current working dir)
   - request size limits and JSON strict parsing
   - structured error responses with no sensitive file contents
8) Deliverable type: runnable server + tests that exercise endpoints against a temp “sample PBIP-like project” created during tests (since a real PBIP project cannot be embedded here). This satisfies the requirement to test logic end-to-end.

Requirement Summary

- `get_report_metadata`: returns pages and visuals with positions, plus applied theme/resourcePackage if present.

        this.registry = registry;
        this.executor = executor;
    }

            return ResponseEntity.status(HttpStatus.NOT_FOUND)

package com.ascendion.aava.pbipmcp.core;

import com.ascendion.aava.pbipmcp.core.dto.ToolsCallRequest;
import com.ascendion.aava.pbipmcp.core.dto.ToolsCallResponse;
import com.ascendion.aava.pbipmcp.core.error.ToolNotFoundException;
import com.ascendion.aava.pbipmcp.core.tools.GetReportMetadataTool;
import com.ascendion.aava.pbipmcp.core.tools.GetSemanticModelSchemaTool;
import org.springframework.stereotype.Component;

@Component
public class ToolCallExecutor {

    private final McpToolRegistry registry;
    private final GetReportMetadataTool reportTool;
    private final GetSemanticModelSchemaTool modelTool;

    public ToolCallExecutor(McpToolRegistry registry,
                            GetReportMetadataTool reportTool,
                            GetSemanticModelSchemaTool modelTool) {
        this.registry = registry;
        this.reportTool = reportTool;
        this.modelTool = modelTool;
    }

    public ToolsCallResponse execute(ToolsCallRequest request) {
        String toolName = request.name();
        if (registry.findByName(toolName) == null) {
            throw new ToolNotFoundException("Unknown tool: " + toolName, "TOOL_NOT_FOUND");
        }

        return switch (toolName) {
            case "get_report_metadata" -> reportTool.call(request.arguments().projectPath());
            case "get_semantic_model_schema" -> modelTool.call(request.arguments().projectPath());
            default -> throw new ToolNotFoundException("Unknown tool: " + toolName, "TOOL_NOT_FOUND");
        };
    }
}

package com.ascendion.aava.pbipmcp.core.fs;

import com.ascendion.aava.pbipmcp.core.config.SecurityConfig;
import com.ascendion.aava.pbipmcp.core.error.ProjectAccessException;
import com.ascendion.aava.pbipmcp.core.error.ProjectNotFoundException;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Map;

@Component
public class ProjectPathResolver {

    private final SecurityConfig securityConfig;

    public ProjectPathResolver(SecurityConfig securityConfig) {
        this.securityConfig = securityConfig;
    }

    public Path resolveProjectDir(String projectPath) {
        if (projectPath == null || projectPath.isBlank()) {
            throw new ProjectNotFoundException("projectPath is missing", "PROJECT_PATH_MISSING", Map.of());
        }

        Path input = Path.of(projectPath);
        Path resolved = input.isAbsolute()
                ? input.toAbsolutePath().normalize()
                : securityConfig.baseDir().resolve(input).toAbsolutePath().normalize();

        if (!resolved.startsWith(securityConfig.baseDir())) {
            throw new ProjectAccessException(
                    "projectPath is outside allowed base directory",
                    "PATH_TRAVERSAL_BLOCKED",
                    Map.of("baseDir", securityConfig.baseDir().toString())
            );
        }

        if (!Files.exists(resolved)) {
            throw new ProjectNotFoundException("projectPath does not exist", "PROJECT_NOT_FOUND", Map.of("projectPath", resolved.toString()));
        }
        if (!Files.isDirectory(resolved)) {
            throw new ProjectNotFoundException("projectPath is not a directory", "PROJECT_NOT_DIRECTORY", Map.of("projectPath", resolved.toString()));
        }

        try {
            if (!Files.isReadable(resolved)) {
                throw new ProjectAccessException("projectPath is not readable", "PROJECT_NOT_READABLE", Map.of("projectPath", resolved.toString()));
            }
        } catch (SecurityException e) {
            throw new ProjectAccessException("projectPath access denied", "PROJECT_ACCESS_DENIED", Map.of("projectPath", resolved.toString()));
        }

        try {
            // basic check to avoid extremely large directory traversals; not a hard security boundary but prevents abuse
            long entries = Files.list(resolved).limit(10_000).count();
            if (entries >= 10_000) {
                throw new ProjectAccessException("projectPath contains too many entries", "PROJECT_TOO_LARGE", Map.of("projectPath", resolved.toString()));
            }
        } catch (IOException e) {
            throw new ProjectAccessException("Unable to read projectPath", "PROJECT_IO_ERROR", Map.of("projectPath", resolved.toString()));
        }

        return resolved;
    }
}

package com.ascendion.aava.pbipmcp.core.fs;

import org.springframework.stereotype.Component;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.attribute.FileTime;
import java.util.Comparator;
import java.util.List;
import java.util.Locale;
import java.util.Optional;
import java.util.Set;
import java.util.stream.Stream;

@Component
public class ProjectScanner {

    private static final Set<String> REPORT_CANDIDATE_NAMES = Set.of(
            "report.json",
            "definition.pbir",
            "definition.pbir.json",
            "report.pbir",
            "report.pbir.json"
    );

    private static final Set<String> MODEL_CANDIDATE_NAMES = Set.of(
            "model.bim",
            "database.json",
            "model.tmsl",
            "model.tmdl"
    );

    public Optional<Path> findBestReportFile(Path projectDir) {
        return findBestByNameOrExt(projectDir,
                REPORT_CANDIDATE_NAMES,
                Set.of(".pbir", ".json"),
                p -> {
                    String n = p.getFileName().toString().toLowerCase(Locale.ROOT);
                    return n.contains("pbir") || n.equals("report.json");
                });
    }

    public Optional<Path> findBestModelFile(Path projectDir) {
        return findBestByNameOrExt(projectDir,
                MODEL_CANDIDATE_NAMES,
                Set.of(".tmdl", ".tmsl", ".bim", ".json"),
                p -> {
                    String n = p.getFileName().toString().toLowerCase(Locale.ROOT);
                    return n.endsWith(".tmdl") || n.endsWith(".tmsl") || n.endsWith(".bim") || n.contains("model") || n.contains("database");
                });
    }

    private Optional<Path> findBestByNameOrExt(Path root,
                                              Set<String> preferredNames,
                                              Set<String> exts,
                                              java.util.function.Predicate<Path> fallbackFilter) {
        try (Stream<Path> stream = Files.walk(root, 10)) {
            List<Path> candidates = stream
                    .filter(Files::isRegularFile)
                    .filter(p -> {
                        String name = p.getFileName().toString().toLowerCase(Locale.ROOT);
                        boolean preferred = preferredNames.contains(name);
                        boolean extMatch = exts.stream().anyMatch(name::endsWith);
                        return preferred || (extMatch && fallbackFilter.test(p));
                    })
                    .toList();

            return candidates.stream()
                    .max(Comparator.comparing(this::lastModifiedSafe));
        } catch (IOException e) {
            return Optional.empty();
        }
    }

    private FileTime lastModifiedSafe(Path p) {
        try {
            return Files.getLastModifiedTime(p);
        } catch (IOException e) {
            return FileTime.fromMillis(0);
        }
    }

                        if (type == null) type = firstTextAt(v, "config", "singleVisual", "prototypeQuery", "From", "Entity");

                        // position: commonly in layout/position/transform
                        ObjectNode pos = mapper.createObjectNode();

    private static void putIfNumber(ObjectNode obj, String key, Number value) {
        if (value == null) return;
        if (value instanceof Integer i) obj.put(key, i);
        else if (value instanceof Long l) obj.put(key, l);
        else obj.put(key, value.doubleValue());

    private static String firstText(JsonNode node, String... keys) {
        if (node == null) return null;
        for (String k : keys) {
            JsonNode v = node.get(k);
            if (v != null && v.isTextual()) return v.asText();
        }
        return null;
    }

    private static String firstTextAt(JsonNode node, String... path) {
        JsonNode cur = node;
        for (String p : path) {
            if (cur == null) return null;
            if (p.matches("\\d+")) {
                if (!cur.isArray()) return null;
                int idx = Integer.parseInt(p);
                if (idx >= cur.size()) return null;
                cur = cur.get(idx);
            } else {
                cur = cur.get(p);
            }
        }
        return (cur != null && cur.isTextual()) ? cur.asText() : null;
    }

    private static Number firstNumber(JsonNode node, String... keys) {
        if (node == null) return null;
        for (String k : keys) {
            JsonNode v = node.get(k);
            if (v != null && v.isNumber()) return v.numberValue();
        }
        return null;
    }

    private static JsonNode firstArray(JsonNode node, String... keys) {
        if (node == null) return null;
        for (String k : keys) {
            JsonNode v = node.get(k);
            if (v != null && v.isArray()) return v;
        }
        return null;
    }

    private static JsonNode firstObject(JsonNode node, String... keys) {
        if (node == null) return null;
        for (String k : keys) {
            JsonNode v = node.get(k);
            if (v != null && v.isObject()) return v;
        }
        return null;
    }

    private static JsonNode findCaseInsensitive(JsonNode node, String targetKey) {
        if (node == null) return null;
        if (!node.isContainerNode()) return null;
        if (node.isObject()) {
            Iterator<String> it = node.fieldNames();
            while (it.hasNext()) {
                String fn = it.next();
                JsonNode child = node.get(fn);
                if (fn.equalsIgnoreCase(targetKey)) {
                    return child;
                }
                JsonNode found = findCaseInsensitive(child, targetKey);
                if (found != null) return found;
            }
        } else if (node.isArray()) {
            for (JsonNode child : node) {
                JsonNode found = findCaseInsensitive(child, targetKey);
                if (found != null) return found;
            }
        }
        return null;
    }

    private static JsonNode firstObjectAt(JsonNode node, String... path) {
        JsonNode cur = node;
        for (String p : path) {
            if (cur == null) return null;
            if (p.matches("\\d+")) {

                cur = cur.get(idx);

        result.put("modelFile", modelFile.toString());

        ArrayNode tablesOut = mapper.createArrayNode();

        JsonNode model = firstObject(root, "model");
        JsonNode tables = model != null ? model.get("tables") : root.get("tables");

                        if (c.has("isHidden") && c.get("isHidden").isBoolean()) {
                            cOut.put("hidden", c.get("isHidden").asBoolean());

                        putIfText(mOut, "owningTable", text(t, "name"));
                        putIfText(mOut, "dax", text(m, "expression"));
                        measuresOut.add(mOut);
