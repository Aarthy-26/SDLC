1. **Requirement Summary**

Optimize the performance of two existing API endpoints:
- `GET /api/apps`
- `GET /api/projects/{id}/details`

**Assumptions (due to missing story details/acceptance criteria):**
- Java 17 + Spring Boot 3.x application.
- Endpoints are read-only and should return JSON.
- Primary latency causes are inefficient DB access (N+1 queries), lack of caching, and missing pagination.
- Add server-side caching using Caffeine with safe defaults and cache invalidation hooks.
- Add pagination + sorting to `/api/apps`.
- Implement efficient repository queries (fetch joins) to avoid N+1.
- Add request validation and structured error handling.
- Add lightweight timing logs and an opt-in correlation id.
- Security boundary is assumed to be handled by upstream auth; this change must not weaken security.

import com.github.benmanes.caffeine.cache.Caffeine;
import org.springframework.cache.CacheManager;
import org.springframework.cache.annotation.EnableCaching;
import org.springframework.cache.caffeine.CaffeineCacheManager;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import java.time.Duration;

@Configuration
@EnableCaching
public class CacheConfig {

    public static final String APPS_LIST_CACHE = "apps.list";
    public static final String PROJECT_DETAILS_CACHE = "projects.details";

    @Bean
    public Caffeine<Object, Object> caffeine() {
        return Caffeine.newBuilder()
                .maximumSize(10_000)
                .expireAfterWrite(Duration.ofMinutes(5));
    }

    @Bean
    public CacheManager cacheManager(Caffeine<Object, Object> caffeine) {
        CaffeineCacheManager mgr = new CaffeineCacheManager(APPS_LIST_CACHE, PROJECT_DETAILS_CACHE);
        mgr.setCaffeine(caffeine);
        return mgr;
    }
}

package com.example.datastudio.config;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.slf4j.MDC;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.Optional;
import java.util.UUID;

@Component
public class CorrelationIdFilter extends OncePerRequestFilter {

    public static final String HEADER = "X-Correlation-Id";
    public static final String MDC_KEY = "correlationId";

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {
        String correlationId = Optional.ofNullable(request.getHeader(HEADER))
                .filter(v -> !v.isBlank())
                .orElse(UUID.randomUUID().toString());

        MDC.put(MDC_KEY, correlationId);
        response.setHeader(HEADER, correlationId);
        try {
            filterChain.doFilter(request, response);
        } finally {
            MDC.remove(MDC_KEY);
        }
    }
}

package com.example.datastudio.service;

import com.example.datastudio.api.dto.AppDto;
import com.example.datastudio.config.CacheConfig;
import com.example.datastudio.domain.AppEntity;
import com.example.datastudio.repo.AppRepository;
import jakarta.validation.constraints.Max;
import jakarta.validation.constraints.Min;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.data.domain.*;
import org.springframework.stereotype.Service;
import org.springframework.util.StringUtils;

import java.util.Objects;

@Service
public class AppService {

    private final AppRepository appRepository;

    public AppService(AppRepository appRepository) {
        this.appRepository = Objects.requireNonNull(appRepository, "appRepository");
    }

    @Cacheable(cacheNames = CacheConfig.APPS_LIST_CACHE,
            key = "T(java.lang.String).format('%s|%s|%s|%s|%s', #page, #size, #sort, #ownerFilter, #nameFilter)")
    public Page<AppDto> listApps(
            @Min(0) int page,
            @Min(1) @Max(200) int size,
            String sort,
            String ownerFilter,
            String nameFilter
    ) {
        Sort sortObj = parseSort(sort);
        Pageable pageable = PageRequest.of(page, size, sortObj);

        Page<AppEntity> entityPage;
        if (StringUtils.hasText(ownerFilter) && StringUtils.hasText(nameFilter)) {
            // If both filters are provided, do in-memory filtering on one dimension after DB query on the other.
            // This avoids writing complex query logic here; for production, replace with a custom specification.
            entityPage = appRepository.findAllByOwnerContainingIgnoreCase(ownerFilter.trim(), pageable)
                    .map(a -> a);
            entityPage = entityPage.map(a -> a) // no-op mapping to keep Page type
                    .filter(a -> a.getName().toLowerCase().contains(nameFilter.trim().toLowerCase()));
        } else if (StringUtils.hasText(ownerFilter)) {
            entityPage = appRepository.findAllByOwnerContainingIgnoreCase(ownerFilter.trim(), pageable);
        } else if (StringUtils.hasText(nameFilter)) {
            entityPage = appRepository.findAllByNameContainingIgnoreCase(nameFilter.trim(), pageable);
        } else {
            entityPage = appRepository.findAll(pageable);
        }

        return entityPage.map(this::toDto);
    }

    private AppDto toDto(AppEntity e) {
        return new AppDto(e.getId(), e.getName(), e.getOwner());
    }

    private Sort parseSort(String sort) {
        // Expected: "name,asc" or "owner,desc". Default: id asc.
        if (!StringUtils.hasText(sort)) {
            return Sort.by(Sort.Direction.ASC, "id");
        }
        String[] parts = sort.split(",");
        String property = parts[0].trim();
        if (property.isEmpty()) {
            return Sort.by(Sort.Direction.ASC, "id");
        }
        Sort.Direction dir = Sort.Direction.ASC;
        if (parts.length > 1) {
            String d = parts[1].trim();
            if (d.equalsIgnoreCase("desc")) {
                dir = Sort.Direction.DESC;
            }
        }

        // Allowlist sortable fields to prevent unexpected property paths.
        if (!property.equals("id") && !property.equals("name") && !property.equals("owner")) {
            property = "id";
        }
        return Sort.by(dir, property);
    }
}

package com.example.datastudio.service;

import com.example.datastudio.api.dto.AppDto;
import com.example.datastudio.api.dto.ProjectDetailsDto;
import com.example.datastudio.config.CacheConfig;
import com.example.datastudio.domain.ProjectEntity;
import com.example.datastudio.repo.ProjectRepository;
import com.example.datastudio.web.NotFoundException;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Objects;

@Service
public class ProjectService {

    private final ProjectRepository projectRepository;

    public ProjectService(ProjectRepository projectRepository) {
        this.projectRepository = Objects.requireNonNull(projectRepository, "projectRepository");
    }

    @Cacheable(cacheNames = CacheConfig.PROJECT_DETAILS_CACHE, key = "#projectId")
    public ProjectDetailsDto getProjectDetails(long projectId) {
        if (projectId <= 0) {
            throw new IllegalArgumentException("projectId must be positive");
        }
        ProjectEntity project = projectRepository.findWithAppsById(projectId)
                .orElseThrow(() -> new NotFoundException("Project not found: " + projectId));

        List<AppDto> apps = project.getApps().stream()
                .map(a -> new AppDto(a.getId(), a.getName(), a.getOwner()))
                .toList();

        return new ProjectDetailsDto(project.getId(), project.getName(), project.getDescription(), apps);
    }
}

package com.example.datastudio.web;

import com.example.datastudio.api.dto.AppDto;
import com.example.datastudio.api.dto.ProjectDetailsDto;
import com.example.datastudio.service.AppService;
import com.example.datastudio.service.ProjectService;
import jakarta.validation.constraints.Max;
import jakarta.validation.constraints.Min;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.data.domain.Page;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

import java.util.Objects;

@RestController
@RequestMapping("/api")
@Validated
public class ApiController {

    private static final Logger log = LoggerFactory.getLogger(ApiController.class);

    private final AppService appService;
    private final ProjectService projectService;

    public ApiController(AppService appService, ProjectService projectService) {
        this.appService = Objects.requireNonNull(appService, "appService");
        this.projectService = Objects.requireNonNull(projectService, "projectService");
    }

    @GetMapping("/apps")
    public Page<AppDto> listApps(
            @RequestParam(defaultValue = "0") @Min(0) int page,
            @RequestParam(defaultValue = "50") @Min(1) @Max(200) int size,
            @RequestParam(required = false) String sort,
            @RequestParam(required = false) String owner,
            @RequestParam(required = false) String name
    ) {
        long start = System.nanoTime();
        try {
            return appService.listApps(page, size, sort, owner, name);
        } finally {
            long tookMs = (System.nanoTime() - start) / 1_000_000;
            if (tookMs > 250) {
                log.warn("Slow request: GET /api/apps took {}ms (page={}, size={})", tookMs, page, size);
            } else {
                log.info("GET /api/apps took {}ms", tookMs);
            }
        }
    }

    @GetMapping("/projects/{id}/details")
    public ProjectDetailsDto projectDetails(@PathVariable("id") long id) {
        long start = System.nanoTime();
        try {
            return projectService.getProjectDetails(id);
        } finally {
            long tookMs = (System.nanoTime() - start) / 1_000_000;
            if (tookMs > 250) {
                log.warn("Slow request: GET /api/projects/{}/details took {}ms", id, tookMs);
            } else {
                log.info("GET /api/projects/{}/details took {}ms", id, tookMs);
            }
        }
    }
}

package com.example.datastudio.web;

import com.example.datastudio.api.dto.ErrorResponse;
import com.example.datastudio.config.CorrelationIdFilter;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.validation.ConstraintViolationException;
import org.slf4j.MDC;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.time.Instant;

@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(NotFoundException.class)
    public ResponseEntity<ErrorResponse> notFound(NotFoundException ex, HttpServletRequest req) {
        return build(HttpStatus.NOT_FOUND, ex.getMessage(), req);
    }

    @ExceptionHandler({IllegalArgumentException.class, ConstraintViolationException.class, MethodArgumentNotValidException.class})
    public ResponseEntity<ErrorResponse> badRequest(Exception ex, HttpServletRequest req) {
        String msg = ex.getMessage();
        if (ex instanceof MethodArgumentNotValidException manv && manv.getBindingResult().hasErrors()) {
            msg = manv.getBindingResult().getAllErrors().get(0).getDefaultMessage();
        }
        return build(HttpStatus.BAD_REQUEST, msg, req);
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ErrorResponse> serverError(Exception ex, HttpServletRequest req) {
        return build(HttpStatus.INTERNAL_SERVER_ERROR, "Unexpected error", req);
    }

    private ResponseEntity<ErrorResponse> build(HttpStatus status, String message, HttpServletRequest req) {
        String cid = MDC.get(CorrelationIdFilter.MDC_KEY);
        ErrorResponse body = new ErrorResponse(
                Instant.now(),
                status.value(),
                status.getReasonPhrase(),
                message,
                req.getRequestURI(),
                cid
        );
        return ResponseEntity.status(status).body(body);
    }
}

package com.example.datastudio.config;

import com.example.datastudio.domain.AppEntity;
import com.example.datastudio.domain.ProjectEntity;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import java.util.List;

@Configuration
public class DataSeeder {

    @Bean
    CommandLineRunner seed(AppRepository appRepository, ProjectRepository projectRepository) {
        return args -> {
            if (appRepository.count() > 0 || projectRepository.count() > 0) {
                return;
            }

            AppEntity a1 = appRepository.save(new AppEntity("Ingestion Service", "platform"));
            AppEntity a2 = appRepository.save(new AppEntity("Analytics Engine", "data"));
            AppEntity a3 = appRepository.save(new AppEntity("Reporting UI", "product"));

            ProjectEntity p1 = new ProjectEntity("Data Studio", "Project for data exploration");
            p1.addApp(a1);
            p1.addApp(a2);
            projectRepository.save(p1);

            ProjectEntity p2 = new ProjectEntity("Insights", "Project for dashboard insights");
            p2.addApp(a2);
            p2.addApp(a3);
            projectRepository.save(p2);

            // Add more apps to make pagination relevant.
            List<String> owners = List.of("platform", "data", "product", "security");
            for (int i = 1; i <= 250; i++) {
                appRepository.save(new AppEntity("App-" + i, owners.get(i % owners.size())));
            }
        };
    }
}
