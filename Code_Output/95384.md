    public List<App> findAll() {
        synchronized (store) {
            return List.copyOf(store);

        synchronized (store) {
            store.clear();
            store.addAll(apps);
        }
    }

package com.aava.datastudio.service.cache;

import java.time.Clock;
import java.time.Duration;
import java.util.Objects;
import java.util.concurrent.ConcurrentHashMap;
import java.util.function.Supplier;

/**
 * Simple thread-safe TTL cache with read-through loading.
 *
 * This avoids external dependencies (like Caffeine) while demonstrating latency mitigation.
 */
public final class TtlCache<K, V> {

    private final ConcurrentHashMap<K, CacheEntry<V>> map = new ConcurrentHashMap<>();
    private final Clock clock;
    private final Duration ttl;

    public TtlCache(Clock clock, Duration ttl) {
        this.clock = Objects.requireNonNull(clock, "clock must not be null");
        this.ttl = Objects.requireNonNull(ttl, "ttl must not be null");
        if (ttl.isNegative() || ttl.isZero()) {
            throw new IllegalArgumentException("ttl must be > 0");
        }
    }

    /**
     * Returns cached value if present and not expired; otherwise loads it using loader.
     */
    public V get(K key, Supplier<V> loader) {
        Objects.requireNonNull(key, "key must not be null");
        Objects.requireNonNull(loader, "loader must not be null");

        long now = clock.millis();
        CacheEntry<V> existing = map.get(key);
        if (existing != null && !existing.isExpired(now)) {
            return existing.value();
        }

        // compute() ensures only one loader execution per key at a time.
        CacheEntry<V> updated = map.compute(key, (k, current) -> {
            long t = clock.millis();
            if (current != null && !current.isExpired(t)) {
                return current;
            }
            V loaded = loader.get();
            return new CacheEntry<>(loaded, t + ttl.toMillis());
        });

        return updated.value();
    }

    public void invalidate(K key) {
        if (key != null) {
            map.remove(key);
        }
    }

    public void clear() {
        map.clear();
    }

    private record CacheEntry<V>(V value, long expiresAtMillis) {
        boolean isExpired(long nowMillis) {
            return nowMillis >= expiresAtMillis;
        }
    }
}

package com.aava.datastudio.service;

import com.aava.datastudio.api.dto.AppDto;
import com.aava.datastudio.domain.App;
import com.aava.datastudio.repo.AppRepository;
import com.aava.datastudio.service.cache.TtlCache;
import io.micrometer.core.instrument.MeterRegistry;
import io.micrometer.core.instrument.Timer;
import org.springframework.stereotype.Service;

import java.time.Clock;
import java.time.Duration;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;

/**
 * App business logic + latency improvements.
 */
@Service
public class AppService {

    private static final String CACHE_KEY_ALL_APPS = "all";

    private final AppRepository appRepository;
    private final TtlCache<String, List<AppDto>> appsCache;
    private final Timer appsTimer;

    public AppService(AppRepository appRepository, MeterRegistry meterRegistry) {
        this.appRepository = Objects.requireNonNull(appRepository, "appRepository must not be null");
        Objects.requireNonNull(meterRegistry, "meterRegistry must not be null");

        // TTL chosen as safe enterprise default to reduce repeated fetches without long staleness.
        this.appsCache = new TtlCache<>(Clock.systemUTC(), Duration.ofSeconds(30));
        this.appsTimer = Timer.builder("api.apps.latency")
                .description("Latency for fetching /api/apps")
                .register(meterRegistry);
    }

    public List<AppDto> getApps(int page, int size) {
        if (page < 0) {
            throw new IllegalArgumentException("page must be >= 0");
        }
        if (size < 1 || size > 200) {
            throw new IllegalArgumentException("size must be between 1 and 200");
        }

        List<AppDto> all = appsTimer.record(() -> appsCache.get(CACHE_KEY_ALL_APPS, this::loadAllApps));

        int from = Math.min(Math.multiplyExact(page, size), all.size());
        int to = Math.min(from + size, all.size());
        return all.subList(from, to);
    }

    private List<AppDto> loadAllApps() {
        // In real systems this would call DB/service; keep deterministic and safe.
        List<App> apps = appRepository.findAll();
        return apps.stream()
                .sorted(Comparator.comparing(App::getId))
                .map(a -> new AppDto(a.getId(), a.getName()))
                .toList();
    }
}

package com.aava.datastudio.service;

import com.aava.datastudio.api.dto.ProjectDetailsDto;
import com.aava.datastudio.domain.Project;
import com.aava.datastudio.exception.NotFoundException;
import com.aava.datastudio.repo.ProjectRepository;
import com.aava.datastudio.service.cache.TtlCache;
import io.micrometer.core.instrument.MeterRegistry;
import io.micrometer.core.instrument.Timer;
import org.springframework.stereotype.Service;

import java.time.Clock;
import java.time.Duration;
import java.util.Objects;

/**
 * Project business logic + caching.
 */
@Service
public class ProjectService {

    private final ProjectRepository projectRepository;
    private final TtlCache<Long, ProjectDetailsDto> projectDetailsCache;
    private final Timer projectDetailsTimer;

    public ProjectService(ProjectRepository projectRepository, MeterRegistry meterRegistry) {
        this.projectRepository = Objects.requireNonNull(projectRepository, "projectRepository must not be null");
        Objects.requireNonNull(meterRegistry, "meterRegistry must not be null");

        this.projectDetailsCache = new TtlCache<>(Clock.systemUTC(), Duration.ofSeconds(30));
        this.projectDetailsTimer = Timer.builder("api.project.details.latency")
                .description("Latency for fetching /api/projects/{id}/details")
                .register(meterRegistry);
    }

    public ProjectDetailsDto getProjectDetails(long id) {
        if (id <= 0) {
            throw new IllegalArgumentException("id must be positive");
        }

        return projectDetailsTimer.record(() -> projectDetailsCache.get(id, () -> loadProjectDetails(id)));
    }

    private ProjectDetailsDto loadProjectDetails(long id) {
        Project project = projectRepository.findById(id)
                .orElseThrow(() -> new NotFoundException("Project not found for id=" + id));
        return new ProjectDetailsDto(project.getId(), project.getName(), project.getDescription());
    }
}
