User Story ID: 95384

1. **Requirement Summary**

Optimize/mitigate high response latency for two endpoints:
- `GET /api/apps`
- `GET /api/projects/{id}/details`

**Assumptions (due to missing description/acceptance criteria):**
- Java 17 + Spring Boot 3.
- These endpoints are backed by a database and/or downstream service calls.
- Primary latency drivers are repeated expensive operations and lack of caching/timeout controls.
- Security boundary is unchanged; this change is internal performance hardening only.
- We will implement safe, opt-in server-side caching with TTL and size bounds, add request correlation IDs, basic timing metrics, and guarded timeouts for downstream calls.
- For `/api/projects/{id}/details`, we will validate `id` as a positive long.

package com.example.datastudio.config;

import com.github.benmanes.caffeine.cache.Caffeine;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.cache.CacheManager;
import org.springframework.cache.caffeine.CaffeineCache;
import org.springframework.cache.support.SimpleCacheManager;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import java.time.Duration;
import java.util.List;

/**
 * Configures bounded in-memory caches with TTL to reduce repeated expensive computations.
 */
@Configuration
public class CacheConfig {

    public static final String APPS_CACHE = "apps";
    public static final String PROJECT_DETAILS_CACHE = "projectDetails";

    @Bean
    public CacheManager cacheManager(
            @Value("${app.cache.apps.ttlSeconds:60}") long appsTtlSeconds,
            @Value("${app.cache.apps.maxSize:5000}") long appsMaxSize,
            @Value("${app.cache.projectDetails.ttlSeconds:30}") long projectDetailsTtlSeconds,
            @Value("${app.cache.projectDetails.maxSize:10000}") long projectDetailsMaxSize
    ) {
        CaffeineCache appsCache = new CaffeineCache(
                APPS_CACHE,
                Caffeine.newBuilder()
                        .maximumSize(appsMaxSize)
                        .expireAfterWrite(Duration.ofSeconds(Math.max(1, appsTtlSeconds)))
                        .build()
        );

        CaffeineCache projectDetailsCache = new CaffeineCache(
                PROJECT_DETAILS_CACHE,
                Caffeine.newBuilder()
                        .maximumSize(projectDetailsMaxSize)
                        .expireAfterWrite(Duration.ofSeconds(Math.max(1, projectDetailsTtlSeconds)))
                        .build()
        );

        SimpleCacheManager mgr = new SimpleCacheManager();
        mgr.setCaches(List.of(appsCache, projectDetailsCache));
        return mgr;
    }
}

package com.example.datastudio.web;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.slf4j.MDC;
import org.springframework.http.HttpHeaders;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.security.SecureRandom;
import java.util.HexFormat;

/**
 * Adds/propagates an X-Correlation-Id header and MDC entry for request tracing.
 */
@Component
public class CorrelationIdFilter extends OncePerRequestFilter {

    public static final String CORRELATION_ID_HEADER = "X-Correlation-Id";
    private static final String MDC_KEY = "corrId";

    private final SecureRandom secureRandom = new SecureRandom();

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {

        String corrId = request.getHeader(CORRELATION_ID_HEADER);
        if (corrId == null || corrId.isBlank() || corrId.length() > 128) {
            corrId = randomId();
        }

        MDC.put(MDC_KEY, corrId);
        response.setHeader(CORRELATION_ID_HEADER, corrId);

        try {
            filterChain.doFilter(request, response);
        } finally {
            MDC.remove(MDC_KEY);
        }
    }

    private String randomId() {
        byte[] bytes = new byte[16];
        secureRandom.nextBytes(bytes);
        return HexFormat.of().formatHex(bytes);
    }
}

package com.example.datastudio.service;

import com.example.datastudio.config.CacheConfig;
import com.example.datastudio.web.dto.AppDto;
import com.example.datastudio.web.dto.ProjectDetailsDto;
import io.micrometer.core.annotation.Timed;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Service;

import java.time.Clock;
import java.util.List;
import java.util.concurrent.*;

/**
 * Service for fetching project details.
 * Adds caching and a strict timeout wrapper for downstream calls.
 */
@Service
public class ProjectDetailsService {

    private final Clock clock;
    private final long timeoutMillis;
    private final ExecutorService executor;

    public ProjectDetailsService(
            Clock clock,
            @Value("${app.downstream.timeoutMillis:1500}") long timeoutMillis
    ) {
        this.clock = clock;
        this.timeoutMillis = Math.max(1, timeoutMillis);
        this.executor = Executors.newFixedThreadPool(
                Math.max(2, Runtime.getRuntime().availableProcessors()),
                runnable -> {
                    Thread t = new Thread(runnable);
                    t.setName("project-details-worker-" + t.threadId());
                    t.setDaemon(true);
                    return t;
                }
        );
    }

    @Timed(value = "api.projectDetails", description = "Time spent serving project details")
    @Cacheable(cacheNames = CacheConfig.PROJECT_DETAILS_CACHE, key = "#projectId")
    public ProjectDetailsDto getProjectDetails(long projectId) {
        // In a real system, project name and apps could come from DB/other services.
        // We'll simulate a downstream call with a timeout to prevent tail-latency.
        String projectName = "Project " + projectId;

        List<AppDto> apps = callDownstreamWithTimeout(projectId);

        return new ProjectDetailsDto(projectId, projectName, apps, clock.millis());
    }

    private List<AppDto> callDownstreamWithTimeout(long projectId) {
        Future<List<AppDto>> future = executor.submit(() -> {
            // Simulate variable latency (0-2000ms). In real code, replace with HTTP/DB call.
            long simulatedDelay = (projectId % 2001);
            try {
                Thread.sleep(simulatedDelay);
            } catch (InterruptedException ie) {
                Thread.currentThread().interrupt();
                throw new IllegalStateException("Downstream call interrupted", ie);
            }

            // Deterministic payload
            return List.of(
                    new AppDto("app-a-" + projectId, "App A for " + projectId),
                    new AppDto("app-b-" + projectId, "App B for " + projectId)
            );
        });

        try {
            return future.get(timeoutMillis, TimeUnit.MILLISECONDS);
        } catch (TimeoutException te) {
            future.cancel(true);
            // Fail fast with an explicit error rather than hanging.
            throw new DownstreamTimeoutException("Timed out fetching project details for projectId=" + projectId);
        } catch (ExecutionException ee) {
            Throwable cause = ee.getCause() == null ? ee : ee.getCause();
            if (cause instanceof RuntimeException re) {
                throw re;
            }
            throw new IllegalStateException("Downstream call failed", cause);
        } catch (InterruptedException ie) {
            Thread.currentThread().interrupt();
            throw new IllegalStateException("Interrupted while waiting for downstream", ie);
        }
    }
}

package com.example.datastudio.web;

import com.example.datastudio.service.DownstreamTimeoutException;
import jakarta.servlet.http.HttpServletRequest;
import org.springframework.http.HttpStatus;
import org.springframework.http.MediaType;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;

import java.time.Instant;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Consistent error responses without leaking internal details.
 */
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(DownstreamTimeoutException.class)
    public ResponseEntity<Map<String, Object>> handleTimeout(DownstreamTimeoutException ex, HttpServletRequest req) {
        return error(HttpStatus.GATEWAY_TIMEOUT, "DOWNSTREAM_TIMEOUT", ex.getMessage(), req.getRequestURI());
    }

    @ExceptionHandler({MethodArgumentTypeMismatchException.class})
    public ResponseEntity<Map<String, Object>> handleTypeMismatch(Exception ex, HttpServletRequest req) {
        return error(HttpStatus.BAD_REQUEST, "BAD_REQUEST", "Invalid parameter type", req.getRequestURI());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, Object>> handleValidation(MethodArgumentNotValidException ex, HttpServletRequest req) {
        String msg = "Validation failed";
        FieldError fe = ex.getBindingResult().getFieldError();
        if (fe != null) {
            msg = fe.getField() + ": " + fe.getDefaultMessage();
        }
        return error(HttpStatus.BAD_REQUEST, "VALIDATION_ERROR", msg, req.getRequestURI());
    }

    @ExceptionHandler(jakarta.validation.ConstraintViolationException.class)
    public ResponseEntity<Map<String, Object>> handleConstraintViolation(jakarta.validation.ConstraintViolationException ex,
                                                                        HttpServletRequest req) {
        return error(HttpStatus.BAD_REQUEST, "VALIDATION_ERROR", "Constraint violation", req.getRequestURI());
    }

    @ExceptionHandler(IllegalStateException.class)
    public ResponseEntity<Map<String, Object>> handleIllegalState(IllegalStateException ex, HttpServletRequest req) {
        return error(HttpStatus.INTERNAL_SERVER_ERROR, "INTERNAL_ERROR", "Internal processing error", req.getRequestURI());
    }

    private ResponseEntity<Map<String, Object>> error(HttpStatus status, String code, String message, String path) {
        Map<String, Object> body = new LinkedHashMap<>();
        body.put("timestamp", Instant.now().toString());
        body.put("status", status.value());
        body.put("error", status.getReasonPhrase());
        body.put("code", code);
        body.put("message", message);
        body.put("path", path);
        return ResponseEntity.status(status).contentType(MediaType.APPLICATION_JSON).body(body);
    }
}
