package com.aava.dms.naming;

import org.hibernate.boot.model.naming.Identifier;
import org.hibernate.boot.model.naming.PhysicalNamingStrategy;
import org.hibernate.engine.jdbc.env.spi.JdbcEnvironment;

import java.util.Locale;

/**
 * Enforces a strict physical naming convention for tables/columns.
 *
 * Assumption: the AAVA DataSource Library expects snake_case lowercase physical names.
 * Example: customerId -> customer_id
 */
public final class AavaPhysicalNamingStrategy implements PhysicalNamingStrategy {

    @Override
    public Identifier toPhysicalCatalogName(Identifier name, JdbcEnvironment jdbcEnvironment) {
        return apply(name);
    }

    @Override
    public Identifier toPhysicalSchemaName(Identifier name, JdbcEnvironment jdbcEnvironment) {
        return apply(name);
    }

    @Override
    public Identifier toPhysicalTableName(Identifier name, JdbcEnvironment jdbcEnvironment) {
        return apply(name);
    }

    @Override
    public Identifier toPhysicalSequenceName(Identifier name, JdbcEnvironment jdbcEnvironment) {
        return apply(name);
    }

    @Override
    public Identifier toPhysicalColumnName(Identifier name, JdbcEnvironment jdbcEnvironment) {
        return apply(name);
    }

    private Identifier apply(Identifier identifier) {
        if (identifier == null) {
            return null;
        }

        String text = identifier.getText();
        if (text == null || text.isBlank()) {
            return identifier;
        }

        String normalized = toSnakeCase(text).toLowerCase(Locale.ROOT);
        return Identifier.toIdentifier(normalized, identifier.isQuoted());
    }

    static String toSnakeCase(String input) {
        // Handles camelCase, PascalCase, and already_snake_case.
        String s = input.replaceAll("([a-z0-9])([A-Z])", "$1_$2");
        s = s.replaceAll("([A-Z]+)([A-Z][a-z0-9]+)", "$1_$2");
        s = s.replaceAll("__+", "_");
        return s;
    }
}

package com.aava.dms.validation;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EntityManagerFactory;
import jakarta.persistence.Table;
import jakarta.persistence.metamodel.Attribute;
import jakarta.persistence.metamodel.EntityType;
import org.springframework.boot.ApplicationArguments;
import org.springframework.boot.ApplicationRunner;
import org.springframework.dao.DataAccessResourceFailureException;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;

import javax.sql.DataSource;
import java.sql.DatabaseMetaData;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.*;

/**
 * Validates that entity attributes map to existing database columns.
 *
 * This provides a fail-fast guarantee to prevent runtime mapping issues.
 */
@Component
public class EntityColumnMappingValidator implements ApplicationRunner {

    private final EntityManagerFactory entityManagerFactory;
    private final DataSource dataSource;

    public EntityColumnMappingValidator(EntityManagerFactory entityManagerFactory,
                                        DataSource dataSource) {
        this.entityManagerFactory = Objects.requireNonNull(entityManagerFactory, "entityManagerFactory");
        this.dataSource = Objects.requireNonNull(dataSource, "dataSource");
    }

    @Override
    public void run(ApplicationArguments args) {
        try {
            validateAllEntities();
        } catch (SQLException e) {
            throw new DataAccessResourceFailureException("Failed to validate entity-to-column mappings", e);
        }
    }

    public void validateAllEntities() throws SQLException {
        Set<EntityType<?>> entities = entityManagerFactory.getMetamodel().getEntities();
        if (entities == null || entities.isEmpty()) {
            return;
        }

        try (var conn = dataSource.getConnection()) {
            DatabaseMetaData meta = conn.getMetaData();

            for (EntityType<?> entityType : entities) {
                Class<?> javaType = entityType.getJavaType();
                if (javaType.getAnnotation(Entity.class) == null) {
                    continue;
                }

                String tableName = resolveTableName(javaType);
                Set<String> actualColumns = readColumns(meta, tableName);

                List<String> missing = new ArrayList<>();
                for (Attribute<?, ?> attr : entityType.getAttributes()) {
                    // Skip collections/associations; validate basic + embedded columns.
                    if (attr.isAssociation() || attr.isCollection()) {
                        continue;
                    }

                    String columnName = resolveColumnName(javaType, attr.getName());
                    if (columnName == null) {
                        continue;
                    }

                    if (!actualColumns.contains(columnName.toLowerCase(Locale.ROOT))) {
                        missing.add(columnName);
                    }
                }

                if (!missing.isEmpty()) {
                    throw new IllegalStateException("Entity mapping validation failed for table '" + tableName +
                            "'. Missing columns: " + missing + ". Present columns: " + actualColumns);
                }
            }
        }
    }

    private static String resolveTableName(Class<?> entityClass) {
        Table table = entityClass.getAnnotation(Table.class);
        if (table != null && table.name() != null && !table.name().isBlank()) {
            return table.name();
        }
        // Default JPA table name is entity name. We enforce explicit @Table in our entities.
        return entityClass.getSimpleName();
    }

    private static String resolveColumnName(Class<?> entityClass, String fieldName) {
        try {
            var field = entityClass.getDeclaredField(fieldName);
            Column column = field.getAnnotation(Column.class);
            if (column != null && column.name() != null && !column.name().isBlank()) {
                return column.name();
            }
            // If no @Column provided, JPA naming strategy will compute it; we validate explicit mappings only.
            // Enterprise default: require explicit @Column to avoid ambiguous naming across libraries.
            return null;
        } catch (NoSuchFieldException e) {
            // Might be a property-based access or synthetic attribute; ignore.
            return null;
        }
    }

    private static Set<String> readColumns(DatabaseMetaData meta, String tableName) throws SQLException {
        Set<String> columns = new HashSet<>();
        // H2 stores uppercase by default; we normalize to lowercase.
        try (ResultSet rs = meta.getColumns(null, null, tableName.toUpperCase(Locale.ROOT), null)) {
            while (rs.next()) {
                String col = rs.getString("COLUMN_NAME");
                if (col != null) {
                    columns.add(col.toLowerCase(Locale.ROOT));
                }
            }
        }
        if (columns.isEmpty()) {
            // Try exact name (some DBs preserve case)
            try (ResultSet rs = meta.getColumns(null, null, tableName, null)) {
                while (rs.next()) {
                    String col = rs.getString("COLUMN_NAME");
                    if (col != null) {
                        columns.add(col.toLowerCase(Locale.ROOT));
                    }
                }
            }
        }
        return columns;
    }
}

package com.aava.dms.entity;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.Id;
import jakarta.persistence.Table;

import java.time.Instant;

@Entity
@Table(name = "customer")
public class CustomerEntity {

    @Id
    @Column(name = "customer_id", nullable = false, length = 64)
    private String customerId;

    @Column(name = "first_name", nullable = false, length = 100)
    private String firstName;

    @Column(name = "last_name", nullable = false, length = 100)
    private String lastName;

    @Column(name = "created_at", nullable = false)
    private Instant createdAt;

    protected CustomerEntity() {
        // for JPA
    }

    public CustomerEntity(String customerId, String firstName, String lastName, Instant createdAt) {
        this.customerId = customerId;
        this.firstName = firstName;
        this.lastName = lastName;
        this.createdAt = createdAt;
    }

    public String getCustomerId() {
        return customerId;
    }

    public String getFirstName() {
        return firstName;
    }

    public String getLastName() {
        return lastName;
    }

    public Instant getCreatedAt() {
        return createdAt;
    }
}

package com.aava.dms.entity;

@Table(name = "orders")

    @Id
    @Column(name = "order_id", nullable = false, length = 64)

    @Column(name = "customer_id", nullable = false, length = 64)

    @Column(name = "order_total", nullable = false, precision = 19, scale = 2)

    @Column(name = "created_at", nullable = false)

        // for JPA

        this.orderId = orderId;
        this.customerId = customerId;
        this.orderTotal = orderTotal;
        this.createdAt = createdAt;
    }

    public String getOrderId() {
        return orderId;
    }

    public String getCustomerId() {
        return customerId;
    }

    public BigDecimal getOrderTotal() {
        return orderTotal;
    }

    public Instant getCreatedAt() {
        return createdAt;
