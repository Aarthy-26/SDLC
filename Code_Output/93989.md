/**
 * CLI entry point that creates two Claude agents (BIM File Creator and Layout Metadata Creator),
 * runs them with sample prompts, and validates outputs.
 *
 * Usage:
 *   export CLAUDE_API_KEY="..."
 *   mvn -q test
 *   mvn -q exec:java -Dexec.mainClass=com.ascendion.aava.claudeims.App
 *
 * Notes:
 * - No secrets are hardcoded.
 * - Endpoints are configurable via environment variables.
 */
public final class App {

    public static void main(String[] args) {
        ClaudeApiClientConfig config = ClaudeApiClientConfig.fromEnv();
        ClaudeApiClient client = new ClaudeApiClient(config);

        AgentOutputValidator validator = new CompositeValidator(List.of(
                new MinimumFieldsValidator(),
                new NonEmptyOutputValidator()
        ));

        System.out.println("Starting IMS agent creation and testing at " + Instant.now());

        // 1) Create BIM File Creator agent
        AgentCreateRequest bimCreate = AgentCreateRequest.bimFileCreatorDefault();
        AgentCreateResponse bimAgent = client.createAgent(bimCreate);
        System.out.println("Created BIM agent: id=" + bimAgent.id());

        // 2) Create Layout Metadata Creator agent
        AgentCreateRequest layoutCreate = AgentCreateRequest.layoutMetadataCreatorDefault();
        AgentCreateResponse layoutAgent = client.createAgent(layoutCreate);
        System.out.println("Created Layout agent: id=" + layoutAgent.id());

        // 3) Execute both agents with smoke test prompts
        AgentRunResponse bimRun = client.runAgent(bimAgent.id(), AgentRunRequest.sampleBimRun());
        System.out.println("BIM run status=" + bimRun.status());

        AgentRunResponse layoutRun = client.runAgent(layoutAgent.id(), AgentRunRequest.sampleLayoutRun());
        System.out.println("Layout run status=" + layoutRun.status());

        // 4) Validate outputs
        validator.validateOrThrow("BIM File Creator", bimRun);
        validator.validateOrThrow("Layout Metadata Creator", layoutRun);

        System.out.println("All agent outputs validated successfully.");
    }

package com.ascendion.aava.claudeims.client;

import java.net.URI;
import java.util.Locale;
import java.util.Objects;

/**
 * Immutable configuration for Claude API client.
 */
public record ClaudeApiClientConfig(URI baseUri, String apiKey, int connectTimeoutSeconds, int requestTimeoutSeconds) {

    public static ClaudeApiClientConfig fromEnv() {
        String base = getenvOrDefault("CLAUDE_API_BASE_URL", "https://api.anthropic.com");
        String apiKey = getenvOrDefault("CLAUDE_API_KEY", "").trim();
        if (apiKey.isEmpty()) {
            throw new IllegalStateException("Missing required environment variable CLAUDE_API_KEY");
        }

        int connectTimeout = parseIntOrDefault(getenvOrDefault("CLAUDE_API_CONNECT_TIMEOUT_SECONDS", "10"), 10);
        int requestTimeout = parseIntOrDefault(getenvOrDefault("CLAUDE_API_REQUEST_TIMEOUT_SECONDS", "60"), 60);

        URI baseUri = URI.create(base);
        return new ClaudeApiClientConfig(baseUri, apiKey, connectTimeout, requestTimeout);
    }

    public ClaudeApiClientConfig {
        Objects.requireNonNull(baseUri, "baseUri");
        Objects.requireNonNull(apiKey, "apiKey");
        if (connectTimeoutSeconds <= 0) {
            throw new IllegalArgumentException("connectTimeoutSeconds must be > 0");
        }
        if (requestTimeoutSeconds <= 0) {
            throw new IllegalArgumentException("requestTimeoutSeconds must be > 0");
        }
    }

    private static String getenvOrDefault(String name, String defaultValue) {
        String value = System.getenv(name);
        return (value == null || value.isBlank()) ? defaultValue : value;
    }

    private static int parseIntOrDefault(String value, int defaultValue) {
        try {
            return Integer.parseInt(value.toLowerCase(Locale.ROOT).trim());
        } catch (RuntimeException ex) {
            return defaultValue;
        }
    }
}

// src/main/java/com/ascendion/aava/claudeims/client/ClaudeApiClient.java
package com.ascendion.aava.claudeims.client;

import com.ascendion.aava.claudeims.model.AgentCreateRequest;
import com.ascendion.aava.claudeims.model.AgentCreateResponse;
import com.ascendion.aava.claudeims.model.AgentRunRequest;
import com.ascendion.aava.claudeims.model.AgentRunResponse;
import com.ascendion.aava.claudeims.util.Json;

import java.io.IOException;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;
import java.util.Objects;

/**
 * Minimal REST client wrapper around Claude API to create and run agents.
 *
 * Endpoints used (defaults):
 * - POST {baseUrl}/v1/agents
 * - POST {baseUrl}/v1/agents/{id}/runs
 *
 * If your Claude tenant uses different endpoints, override via env vars:
 * - CLAUDE_API_AGENTS_PATH (default: /v1/agents)
 * - CLAUDE_API_AGENT_RUNS_SUFFIX (default: /runs)
 */
public final class ClaudeApiClient {

    private final ClaudeApiClientConfig config;
    private final HttpClient httpClient;
    private final String agentsPath;
    private final String agentRunsSuffix;

    public ClaudeApiClient(ClaudeApiClientConfig config) {
        this.config = Objects.requireNonNull(config, "config");
        this.httpClient = HttpClient.newBuilder()
                .connectTimeout(Duration.ofSeconds(config.connectTimeoutSeconds()))
                .build();

        this.agentsPath = getenvOrDefault("CLAUDE_API_AGENTS_PATH", "/v1/agents");
        this.agentRunsSuffix = getenvOrDefault("CLAUDE_API_AGENT_RUNS_SUFFIX", "/runs");
    }

    public AgentCreateResponse createAgent(AgentCreateRequest request) {
        Objects.requireNonNull(request, "request");
        URI uri = config.baseUri().resolve(agentsPath);
        String body = Json.toJson(request);

        HttpRequest httpRequest = baseRequest(uri)
                .POST(HttpRequest.BodyPublishers.ofString(body))
                .build();

        HttpResponse<String> response = send(httpRequest);
        ensure2xx(response, "createAgent");
        return Json.fromJson(response.body(), AgentCreateResponse.class);
    }

    public AgentRunResponse runAgent(String agentId, AgentRunRequest request) {
        if (agentId == null || agentId.isBlank()) {
            throw new IllegalArgumentException("agentId must be non-empty");
        }
        Objects.requireNonNull(request, "request");

        URI uri = config.baseUri().resolve(agentsPath + "/" + urlPathEscape(agentId) + agentRunsSuffix);
        String body = Json.toJson(request);

        HttpRequest httpRequest = baseRequest(uri)
                .POST(HttpRequest.BodyPublishers.ofString(body))
                .build();

        HttpResponse<String> response = send(httpRequest);
        ensure2xx(response, "runAgent");
        return Json.fromJson(response.body(), AgentRunResponse.class);
    }

    private HttpRequest.Builder baseRequest(URI uri) {
        return HttpRequest.newBuilder(uri)
                .timeout(Duration.ofSeconds(config.requestTimeoutSeconds()))
                // Headers are modeled after common Claude API usage; adjust if your API differs.
                .header("Content-Type", "application/json")
                .header("Accept", "application/json")
                .header("X-API-Key", config.apiKey());
    }

    private HttpResponse<String> send(HttpRequest request) {
        try {
            return httpClient.send(request, HttpResponse.BodyHandlers.ofString());
        } catch (IOException e) {
            throw new ClaudeClientException("I/O error calling Claude API: " + safeMsg(e), e);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new ClaudeClientException("Interrupted calling Claude API", e);
        }
    }

    private static void ensure2xx(HttpResponse<String> response, String operation) {
        int code = response.statusCode();
        if (code < 200 || code >= 300) {
            String payload = response.body() == null ? "" : response.body();
            throw new ClaudeApiException(operation + " failed with HTTP " + code + ": " + truncate(payload, 2000), code, payload);
        }
    }

    private static String truncate(String s, int max) {
        if (s == null) return "";
        if (s.length() <= max) return s;
        return s.substring(0, max) + "...";
    }

    private static String safeMsg(Throwable t) {
        String msg = t.getMessage();
        return msg == null ? t.getClass().getSimpleName() : msg;
    }

    private static String getenvOrDefault(String name, String defaultValue) {
        String value = System.getenv(name);
        return (value == null || value.isBlank()) ? defaultValue : value;
    }

    /**
     * Conservative path segment escape (not a full URL encoder) to avoid injection into URL paths.
     */
    private static String urlPathEscape(String segment) {
        String s = segment.trim();
        // Replace only path separators and control characters.
        s = s.replace("/", "%2F");
        s = s.replace("\\", "%5C");
        return s;
    }
}

        }
    }

import java.util.Objects;

/**
 * Runs multiple validators.
 */
public final class CompositeValidator implements AgentOutputValidator {

    private final List<AgentOutputValidator> validators;

    public CompositeValidator(List<AgentOutputValidator> validators) {
        this.validators = List.copyOf(Objects.requireNonNull(validators, "validators"));
        if (this.validators.isEmpty()) {
            throw new IllegalArgumentException("validators must not be empty");
        }
    }

    @Override
    public void validateOrThrow(String agentName, AgentRunResponse response) {
        for (AgentOutputValidator v : validators) {
            v.validateOrThrow(agentName, response);
        }
    }
}

package com.ascendion.aava.claudeims.validation;

import com.ascendion.aava.claudeims.model.AgentRunResponse;

import java.util.Collection;
import java.util.Map;

/**
 * Validates that output is not empty.
 */
public final class NonEmptyOutputValidator implements AgentOutputValidator {

    @Override
    public void validateOrThrow(String agentName, AgentRunResponse response) {
        Object output = response == null ? null : response.output();
        if (output == null) {
            throw new OutputValidationException(agentName + ": output is null");
        }
        if (output instanceof String s) {
            if (s.trim().isEmpty()) {
                throw new OutputValidationException(agentName + ": output string is empty");
            }
        } else if (output instanceof Collection<?> c) {
            if (c.isEmpty()) {
                throw new OutputValidationException(agentName + ": output collection is empty");
            }
        } else if (output instanceof Map<?, ?> m) {
            if (m.isEmpty()) {
                throw new OutputValidationException(agentName + ": output map is empty");
            }
        }
        // Otherwise accept arbitrary object types as non-empty.
    }
}
