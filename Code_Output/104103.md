User Story ID: 104103

1. Requirement Summary
- Integrate the AAVA Library into the DPAI application to manage DataSource connections, using AAVA-supported configuration and authentication mechanisms instead of handling DB connections/credentials directly in DPAI.
- Provide environment-specific configuration for database URL, username, authentication mode, and other required properties for all supported environments.
- Note: The actual AAVA Library SDK/API surface was not available in the work item content; this implementation provides a placeholder abstraction (AavaDataSourceFactory) with the same responsibility (create/configure DataSource using AAVA-style config & auth), to be swapped for the real AAVA artifact/classes.

Assumptions
- Java version: 17.
- Framework: Spring Boot 3.x (enterprise default for configuration profiles and DataSource wiring).
- Supported environments/profiles: dev, test, prod.
- Authentication modes supported by configuration: BASIC (username/password) and TOKEN (opaque token). Password/token are supplied via environment variables or external secret store and not committed.
- Target database driver is PostgreSQL for runnable example; can be swapped via configuration.
- If AAVA library becomes available, it will provide a way to build a DataSource (or connection provider) given URL, username, and auth mode; this code isolates that behind AavaDataSourceFactory.

import com.ascendion.aava.dpai.aava.AavaDataSourceFactory;

import org.springframework.context.annotation.Bean;

import javax.sql.DataSource;

/**
 * Creates the application DataSource via the AAVA abstraction.
 */
@Configuration
@EnableConfigurationProperties(AavaDataSourceProperties.class)
public class AavaDataSourceAutoConfiguration {

    @Bean
    public AavaDataSourceFactory aavaDataSourceFactory() {
        return new DefaultAavaDataSourceFactory();
    }

    @Bean
    public DataSource dataSource(AavaDataSourceFactory factory, @Valid AavaDataSourceProperties properties) {
        try {
            return factory.create(properties);
        } catch (DpaiConfigurationException e) {
            // Typed exception is handled centrally by ControllerAdvice.
            throw e;
        }
    }
}

package com.ascendion.aava.dpai.aava;

import com.ascendion.aava.dpai.config.AavaDataSourceProperties;
import com.ascendion.aava.dpai.error.DpaiConfigurationException;
import org.springframework.jdbc.datasource.DriverManagerDataSource;

import javax.sql.DataSource;

import static com.ascendion.aava.dpai.util.InputValidation.requireNonBlank;

/**
 * Placeholder abstraction — replace with the real AAVA Library SDK once available.
 *
 * This implementation does NOT call AAVA. It creates a simple {@link DriverManagerDataSource} using
 * configured URL/username and an auth-mode-specific secret (password or token).
 *
 * When the real AAVA Library is available, replace this class with an implementation backed by the
 * AAVA SDK that performs secure auth and pooling according to AAVA standards.
 */
public class DefaultAavaDataSourceFactory implements AavaDataSourceFactory {

    @Override
    public DataSource create(AavaDataSourceProperties properties) {
        if (properties == null) {
            throw new DpaiConfigurationException("DPAI_DATASOURCE_CONFIG_MISSING", "DataSource properties must be provided");
        }

        String url = requireNonBlank(properties.getUrl(), "dpai.datasource.url");
        String username = requireNonBlank(properties.getUsername(), "dpai.datasource.username");

        String secret;
        if (properties.getAuthMode() == AavaDataSourceProperties.AuthMode.TOKEN) {
            secret = requireNonBlank(properties.getToken(), "dpai.datasource.token");
        } else {
            secret = requireNonBlank(properties.getPassword(), "dpai.datasource.password");
        }

        DriverManagerDataSource ds = new DriverManagerDataSource();
        ds.setUrl(url);
        ds.setUsername(username);
        ds.setPassword(secret);

        if (properties.getDriverClassName() != null && !properties.getDriverClassName().isBlank()) {
            ds.setDriverClassName(properties.getDriverClassName().trim());
        }

        return ds;
    }
}

     * Require a string to be non-null and non-blank.

    public static String requireNonBlank(String value, String fieldName) {
        if (fieldName == null || fieldName.isBlank()) {
            fieldName = "<unknown>";
        }
        if (value == null || value.isBlank()) {
            throw new DpaiConfigurationException("DPAI_CONFIG_INVALID", "Missing/blank required property: " + fieldName);
        }
        return value.trim();
    }
}

package com.ascendion.aava.dpai.web;

import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RestController;

import javax.sql.DataSource;
import java.sql.Connection;
import java.sql.DatabaseMetaData;
import java.sql.SQLException;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Minimal endpoint to verify DataSource wiring.
 */
@RestController
public class HealthController {

    private final DataSource dataSource;

    public HealthController(DataSource dataSource) {
        this.dataSource = dataSource;
    }

    /**
     * Attempts to obtain a JDBC connection and returns basic metadata.
     */
    @GetMapping("/health/db")
    public Map<String, Object> dbHealth() {
        Map<String, Object> result = new LinkedHashMap<>();

        try (Connection c = dataSource.getConnection()) {
            DatabaseMetaData md = c.getMetaData();
            result.put("status", "UP");
            result.put("databaseProduct", md.getDatabaseProductName());
            result.put("databaseVersion", md.getDatabaseProductVersion());
            result.put("url", md.getURL());
            result.put("user", md.getUserName());
            return result;
        } catch (SQLException e) {
            // Convert to typed runtime exception to keep exception handling consistent.
            throw new com.ascendion.aava.dpai.error.DpaiConfigurationException(
                    "DPAI_DB_CONNECTION_FAILED",
                    "Database connectivity check failed: " + safeSqlMessage(e)
            );
        }
    }

    private static String safeSqlMessage(SQLException e) {
        // Avoid leaking sensitive details; keep message minimal.
        if (e == null) {
            return "unknown SQL error";
        }
        String msg = e.getMessage();
        if (msg == null || msg.isBlank()) {
            return "SQL error";
        }
        // Strip newlines to reduce log/response injection risk.
        return msg.replaceAll("[\\r\\n]+", " ").trim();
    }
}
