User Story ID: 104103

1. Requirement Summary — the clarified, confirmed requirement

Implement and integrate an AAVA-provided DataSource connection management library into the DPAI application so that DPAI obtains JDBC DataSource connections via the library (not via locally-managed credential/connection logic). Environment-specific configuration must supply the database URL, username, authentication mode, and other required properties.

Assumptions (enterprise defaults applied to fill gaps safely):
- Java 17, Spring Boot 3.3.x, Maven build.
- AAVA Library is represented here as an interface (`AavaDataSourceFactory`) with a default implementation that reads environment-specific configuration and creates a pooled `HikariDataSource`. In your real project, replace the `aava-starter` dependency and the factory implementation with the actual AAVA library artifact/classes.
- Supported environments are handled via Spring profiles (`dev`, `test`, `prod`) with `application-<profile>.yml`.
- Authentication modes supported: `BASIC` (username/password) and `IAM_TOKEN` (token-based; token supplied by config or external secret manager). Token/password are never logged.
- A minimal health-check endpoint is included to validate that a connection can be acquired.

package com.ascendion.dpai.aava.core;

import com.ascendion.dpai.aava.config.AavaDataSourceProperties;
import com.ascendion.dpai.aava.config.AavaDataSourceProperties.AuthenticationMode;
import com.ascendion.dpai.aava.exception.AavaConfigurationException;
import com.zaxxer.hikari.HikariConfig;
import com.zaxxer.hikari.HikariDataSource;

import javax.sql.DataSource;
import java.util.Objects;

/**
 * Default implementation that mimics an AAVA library behavior by:
 * - reading environment-specific config
 * - applying the required authentication mechanism
 * - returning a pooled Hikari DataSource
 *
 * Replace this implementation with the real AAVA library integration.
 */
public class DefaultAavaDataSourceFactory implements AavaDataSourceFactory {

    @Override
    public DataSource create(AavaDataSourceProperties properties) {
        Objects.requireNonNull(properties, "properties must not be null");

        AuthenticationMode mode = Objects.requireNonNull(properties.getAuthenticationMode(), "authenticationMode must not be null");

        HikariConfig cfg = new HikariConfig();
        cfg.setJdbcUrl(properties.getJdbcUrl());
        cfg.setUsername(properties.getUsername());
        cfg.setDriverClassName(properties.getDriverClassName());

        // Pool settings
        cfg.setMaximumPoolSize(properties.getPool().getMaximumPoolSize());
        cfg.setMinimumIdle(properties.getPool().getMinimumIdle());
        cfg.setConnectionTimeout(properties.getPool().getConnectionTimeoutMs());
        cfg.setIdleTimeout(properties.getPool().getIdleTimeoutMs());
        cfg.setMaxLifetime(properties.getPool().getMaxLifetimeMs());

        // Auth handling (do not log secrets)
        if (mode == AuthenticationMode.BASIC) {
            if (properties.getPassword() == null || properties.getPassword().isBlank()) {
                throw new AavaConfigurationException("AAVA BASIC auth requires a non-empty password (aava.datasource.password)");
            }
            cfg.setPassword(properties.getPassword());
        } else if (mode == AuthenticationMode.IAM_TOKEN) {
            if (properties.getToken() == null || properties.getToken().isBlank()) {
                throw new AavaConfigurationException("AAVA IAM_TOKEN auth requires a non-empty token (aava.datasource.token)");
            }
            // Token-based DB auth varies by vendor/driver.
            // Common approaches include placing the token in the password field or a driver property.
            // We do BOTH defensively; driver will use what it supports.
            cfg.setPassword(properties.getToken());
            cfg.addDataSourceProperty("accessToken", properties.getToken());
        } else {
            throw new AavaConfigurationException("Unsupported authentication mode: " + mode);
        }

        // Security-hardening defaults
        cfg.setLeakDetectionThreshold(0L); // disabled by default; enable only when troubleshooting
        cfg.setAutoCommit(true);

        return new HikariDataSource(cfg);
    }
}

import org.springframework.http.HttpStatus;

import java.util.Map;

/**
 * Maps typed exceptions to safe HTTP responses without leaking secrets.
 */
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(AavaConfigurationException.class)
    public ResponseEntity<Map<String, Object>> handleAavaConfiguration(AavaConfigurationException ex) {
        Map<String, Object> body = new LinkedHashMap<>();
        body.put("timestamp", OffsetDateTime.now().toString());
        body.put("error", "CONFIGURATION_ERROR");
        body.put("message", ex.getMessage());
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }
}

package com.ascendion.dpai.health;

import org.springframework.http.MediaType;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RestController;

import javax.sql.DataSource;
import java.sql.Connection;
import java.time.OffsetDateTime;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Minimal endpoint to validate that DPAI can obtain a connection via the AAVA-managed DataSource.
 */
@RestController
public class DataSourceHealthController {

    private final DataSource dataSource;

    public DataSourceHealthController(DataSource dataSource) {
        this.dataSource = dataSource;
    }

    @GetMapping(path = "/internal/datasource/health", produces = MediaType.APPLICATION_JSON_VALUE)
    public Map<String, Object> health() throws Exception {
        Map<String, Object> out = new LinkedHashMap<>();
        out.put("timestamp", OffsetDateTime.now().toString());

        try (Connection c = dataSource.getConnection()) {
            out.put("ok", Boolean.TRUE);
            out.put("databaseProduct", c.getMetaData().getDatabaseProductName());
            out.put("databaseVersion", c.getMetaData().getDatabaseProductVersion());
        }

        return out;
    }
}

- `AavaDataSourceFactory`: Abstraction representing the AAVA library’s contract to obtain a `DataSource`.
