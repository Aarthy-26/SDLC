User Story ID: 97420

1. **Requirement Summary**
Create a test-case artifact generator for validating a Power BI to Looker migration/conversion. The solution shall produce a comprehensive set of test cases covering:
- Functional validation of converted reports/dashboards/tiles/pages
- Data accuracy validation (counts, totals, aggregations, key points)
- Visualization correctness (titles, labels, legends, axes, sorting, values)
- Filter/slicer behavior (apply/clear/change)
- Calculated fields/measures equivalence
- UI/interaction behaviors (navigation, drill-down/through, sorting, interactions)
- Negative/error scenarios (unsupported configs, invalid inputs, missing data, conversion failures)
- Traceability: each test case includes preconditions, steps, test data, expected results, postconditions, and requirement link

**Assumptions** (enterprise defaults)
- Output is a machine-readable test case document in JSON and a human-readable Markdown file.
- Test cases are created from a configurable template catalog (no Power BI/Looker APIs required).
- The tool runs as a standalone CLI Java app (Java 17) and reads optional input metadata from a JSON file.
- No secrets are stored; no network calls are performed.

        static void writeMarkdown(Path path, GenerationContext ctx, List<TestCase> testCases) throws IOException {
            Objects.requireNonNull(path, "path");
            Objects.requireNonNull(ctx, "ctx");
            Objects.requireNonNull(testCases, "testCases");

            safeCreateParentDirs(path);

            try (BufferedWriter w = Files.newBufferedWriter(path, StandardCharsets.UTF_8)) {
                w.write("# Test Cases: Power BI to Looker Migration\n\n");
                w.write("- Story ID: **" + escapeMd(ctx.storyId) + "**\n");
                w.write("- Story Title: **" + escapeMd(ctx.storyTitle) + "**\n");
                w.write("- Requirement Link: " + escapeMd(ctx.requirementLink) + "\n");
                if (!ctx.projectName.isBlank()) w.write("- Project: " + escapeMd(ctx.projectName) + "\n");
                if (!ctx.areaPath.isBlank()) w.write("- Area Path: " + escapeMd(ctx.areaPath) + "\n");
                if (!ctx.iterationPath.isBlank()) w.write("- Iteration Path: " + escapeMd(ctx.iterationPath) + "\n");
                if (!ctx.datasetName.isBlank()) w.write("- Dataset: " + escapeMd(ctx.datasetName) + "\n");
                if (!ctx.reportName.isBlank()) w.write("- Report: " + escapeMd(ctx.reportName) + "\n");
                w.write("- Generated: " + Instant.now().toString() + "\n\n");

                w.write("## Test Case Index\n\n");
                for (TestCase tc : testCases) {
                    w.write("- [" + escapeMd(tc.id) + ": " + escapeMd(tc.title) + "](#" + toAnchor(tc.id) + ")\n");
                }
                w.write("\n---\n\n");

                for (TestCase tc : testCases) {
                    w.write("## " + escapeMd(tc.id) + ": " + escapeMd(tc.title) + "\n\n");
                    w.write("- Category: " + escapeMd(tc.category) + "\n");
                    w.write("- Priority: " + escapeMd(tc.priority) + "\n");
                    w.write("- Tags: " + escapeMd(String.join(", ", tc.tags)) + "\n");
                    w.write("- Requirement Link: " + escapeMd(tc.requirementLink) + "\n\n");

                    writeSection(w, "Preconditions", tc.preconditions);
                    writeSection(w, "Test Steps", tc.steps);
                    writeSection(w, "Test Data", tc.testData);
                    writeSection(w, "Expected Results", tc.expectedResults);
                    writeSection(w, "Postconditions", tc.postconditions);

                    w.write("---\n\n");
                }
            }
        }

        static void writeJson(Path path, GenerationContext ctx, List<TestCase> testCases) throws IOException {
            Objects.requireNonNull(path, "path");

            safeCreateParentDirs(path);

            try (BufferedWriter w = Files.newBufferedWriter(path, StandardCharsets.UTF_8)) {
                StringBuilder sb = new StringBuilder(64 * 1024);
                sb.append("{");
                appendJsonField(sb, "storyId", ctx.storyId).append(",");
                appendJsonField(sb, "storyTitle", ctx.storyTitle).append(",");
                appendJsonField(sb, "requirementLink", ctx.requirementLink).append(",");
                appendJsonField(sb, "projectName", ctx.projectName).append(",");

                appendJsonField(sb, "iterationPath", ctx.iterationPath).append(",");
                appendJsonField(sb, "sourceSystem", ctx.sourceSystem).append(",");
                appendJsonField(sb, "targetSystem", ctx.targetSystem).append(",");
                appendJsonField(sb, "datasetName", ctx.datasetName).append(",");
                appendJsonField(sb, "reportName", ctx.reportName).append(",");
                appendJsonField(sb, "generatedAt", Instant.now().toString()).append(",");
                sb.append("\"testCases\":[");
                for (int i = 0; i < testCases.size(); i++) {
                    TestCase tc = testCases.get(i);
                    sb.append("{");
                    appendJsonField(sb, "id", tc.id).append(",");
                    appendJsonField(sb, "title", tc.title).append(",");
                    appendJsonField(sb, "category", tc.category).append(",");
                    appendJsonField(sb, "priority", tc.priority).append(",");
                    appendJsonArray(sb, "tags", tc.tags).append(",");
                    appendJsonArray(sb, "preconditions", tc.preconditions).append(",");

        private static void writeSection(BufferedWriter w, String title, List<String> items) throws IOException {
            w.write("### " + title + "\n\n");
            if (items == null || items.isEmpty()) {
                w.write("- (none)\n\n");
                return;
            }
            for (String s : items) {
                w.write("1. " + escapeMd(s) + "\n");
            }
            w.write("\n");
        }

        private static void safeCreateParentDirs(Path path) throws IOException {
            Path parent = path.getParent();
            if (parent != null) {
                Files.createDirectories(parent);
            }
        }

        private static String escapeMd(String s) {
            if (s == null) return "";
            return s.replace("\\", "\\\\")
                    .replace("`", "\\`")
                    .replace("*", "\\*")
                    .replace("_", "\\_")
                    .replace("[", "\\[")
                    .replace("]", "\\]")
                    .replace("<", "&lt;")
                    .replace(">", "&gt;");
        }

        private static String toAnchor(String s) {
            // GitHub-style anchor approximation.
            return s.toLowerCase().replaceAll("[^a-z0-9]+", "-").replaceAll("(^-|-$)", "");
        }

        private static StringBuilder appendJsonField(StringBuilder sb, String key, String value) {
            sb.append("\"").append(jsonEscape(key)).append("\":\"").append(jsonEscape(value)).append("\"");
            return sb;
        }

        private static StringBuilder appendJsonArray(StringBuilder sb, String key, List<String> values) {
            sb.append("\"").append(jsonEscape(key)).append("\":[");
            if (values != null) {
                for (int i = 0; i < values.size(); i++) {
                    sb.append("\"").append(jsonEscape(values.get(i))).append("\"");
                    if (i < values.size() - 1) {
                        sb.append(",");
                    }
                }
            }
            sb.append("]");
            return sb;
        }

        private static String jsonEscape(String s) {
            if (s == null) {
                return "";
            }
            StringBuilder out = new StringBuilder(s.length() + 16);
            for (int i = 0; i < s.length(); i++) {
                char c = s.charAt(i);
                switch (c) {
                    case '"' -> out.append("\\\"");
                    case '\\' -> out.append("\\\\");
                    case '\b' -> out.append("\\b");
                    case '\f' -> out.append("\\f");
                    case '\n' -> out.append("\\n");
                    case '\r' -> out.append("\\r");
                    case '\t' -> out.append("\\t");
                    default -> {
                        if (c < 0x20) {
                            out.append(String.format("\\u%04x", (int) c));
                        } else {
                            out.append(c);
                        }
                    }
                }
            }
            return out.toString();
        }
    }

    /**
     * Minimal JSON parser for a flat JSON object (string keys and string values only).
     *
     * <p>Example:
     * <pre>
     * {"datasetName":"Sales","reportName":"Executive Overview"}
     * </pre>
     *
     * <p>This intentionally does not support nested objects/arrays to keep the tool dependency-free.
     */
    static final class SimpleJsonFlatObjectParser {

        static Map<String, String> parse(Path path) throws IOException {
            Objects.requireNonNull(path, "path");
            if (!Files.exists(path) || !Files.isRegularFile(path)) {
                throw new ValidationException("Metadata file does not exist: " + path);
            }
            String json;
            try (BufferedReader r = Files.newBufferedReader(path, StandardCharsets.UTF_8)) {
                StringBuilder sb = new StringBuilder();
                String line;
                while ((line = r.readLine()) != null) {
                    sb.append(line);
                }
                json = sb.toString().trim();
            }

            if (json.isBlank()) {
                return Map.of();
            }

            Tokenizer t = new Tokenizer(json);
            t.skipWs();
            t.expect('{');
            t.skipWs();

            Map<String, String> map = new LinkedHashMap<>();
            if (t.peek() == '}') {
                t.expect('}');
                return map;
            }

            while (true) {
                t.skipWs();
                String key = t.readString();
                t.skipWs();
                t.expect(':');
                t.skipWs();
                String value = t.readString();
                map.put(key, value);
                t.skipWs();
                char p = t.peek();
                if (p == ',') {
                    t.expect(',');
                    continue;
                }
                if (p == '}') {
                    t.expect('}');
                    break;
                }
                throw new ValidationException("Invalid JSON: expected ',' or '}' at position " + t.pos);
            }
            return map;
        }

        static final class Tokenizer {
            final String s;
            int pos;

            Tokenizer(String s) {
                this.s = Objects.requireNonNull(s, "s");
            }

            void skipWs() {
                while (pos < s.length()) {
                    char c = s.charAt(pos);
                    if (c == ' ' || c == '\n' || c == '\r' || c == '\t') {
                        pos++;
                    } else {
                        return;
                    }
                }
            }

            char peek() {
                if (pos >= s.length()) {
                    return (char) 0;
                }
                return s.charAt(pos);
            }

            void expect(char c) {
                if (peek() != c) {
                    throw new ValidationException("Invalid JSON: expected '" + c + "' at position " + pos);
                }
                pos++;
            }

            String readString() {
                if (peek() != '"') {
                    throw new ValidationException("Invalid JSON: expected string at position " + pos);
                }
                pos++; // opening quote
                StringBuilder out = new StringBuilder();
                while (pos < s.length()) {
                    char c = s.charAt(pos++);
                    if (c == '"') {
                        return out.toString();
                    }
                    if (c == '\\') {
                        if (pos >= s.length()) {
                            throw new ValidationException("Invalid JSON: bad escape at end of input");
                        }
                        char e = s.charAt(pos++);
                        switch (e) {
                            case '"' -> out.append('"');
                            case '\\' -> out.append('\\');
                            case '/' -> out.append('/');
                            case 'b' -> out.append('\b');
                            case 'f' -> out.append('\f');
                            case 'n' -> out.append('\n');
                            case 'r' -> out.append('\r');
                            case 't' -> out.append('\t');
                            case 'u' -> {
                                if (pos + 4 > s.length()) {
                                    throw new ValidationException("Invalid JSON: incomplete unicode escape at position " + (pos - 2));
                                }
                                String hex = s.substring(pos, pos + 4);
                                pos += 4;
                                try {
                                    out.append((char) Integer.parseInt(hex, 16));
                                } catch (NumberFormatException nfe) {
                                    throw new ValidationException("Invalid JSON: invalid unicode escape: \\u" + hex);
                                }
                            }
                            default -> throw new ValidationException("Invalid JSON: unsupported escape: \\" + e);
                        }
                    } else {
                        out.append(c);
                    }
                }
                throw new ValidationException("Invalid JSON: unterminated string");
            }
        }
    }

    /** Typed validation exception (no blanket catch) for clearer error handling. */
    static final class ValidationException extends RuntimeException {
        ValidationException(String message) {
            super(message);
        }
    }
}
