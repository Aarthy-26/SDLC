package com.aava.datastudio.importapi.dto;

import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.annotation.JsonProperty;

import java.util.Objects;

/**
 * DTO returned/consumed by the App Import API describing a column.
 *
 * <p>Fix for Jackson deserialization: Jackson requires either a no-args constructor
 * (plus setters/fields) or an explicitly annotated creator constructor.
 * This class provides a public no-args constructor and standard setters.</p>
 */
@JsonIgnoreProperties(ignoreUnknown = true)
public class ColumnResponse {

    @JsonProperty("name")
    private String name;

    @JsonProperty("type")
    private String type;

    @JsonProperty("nullable")
    private boolean nullable;

    @JsonProperty("length")
    private Integer length;

    /**
     * Public no-args constructor required by Jackson for default deserialization.
     */
    public ColumnResponse() {
        // intentionally empty
    }

    public ColumnResponse(String name, String type, boolean nullable, Integer length) {
        this.name = name;
        this.type = type;
        this.nullable = nullable;
        this.length = length;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getType() {
        return type;
    }

    public void setType(String type) {
        this.type = type;
    }

    public boolean isNullable() {
        return nullable;
    }

    public void setNullable(boolean nullable) {
        this.nullable = nullable;
    }

    public Integer getLength() {
        return length;
    }

    public void setLength(Integer length) {
        this.length = length;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        ColumnResponse that = (ColumnResponse) o;
        return nullable == that.nullable
                && Objects.equals(name, that.name)
                && Objects.equals(type, that.type)
                && Objects.equals(length, that.length);
    }

    @Override
    public int hashCode() {
        return Objects.hash(name, type, nullable, length);
    }

    @Override
    public String toString() {
        return "ColumnResponse{" +
                "name='" + name + '\'' +
                ", type='" + type + '\'' +
                ", nullable=" + nullable +
                ", length=" + length +
                '}';
    }
}

package com.aava.datastudio.importapi.dto;

import com.fasterxml.jackson.databind.DeserializationFeature;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;

import static org.junit.jupiter.api.Assertions.*;

/**
 * Regression test ensuring ColumnResponse can be deserialized by Jackson.
 */
public class ColumnResponseJacksonTest {

    @Test
    void shouldDeserializeColumnResponse() throws Exception {
        String json = "{" +
                "\"name\":\"customer_id\"," +
                "\"type\":\"STRING\"," +
                "\"nullable\":false," +
                "\"length\":64" +
                "}";

        ObjectMapper mapper = new ObjectMapper()
                .configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);

        ColumnResponse result = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(result);
        assertEquals("customer_id", result.getName());
        assertEquals("STRING", result.getType());
        assertFalse(result.isNullable());
        assertEquals(64, result.getLength());
    }

    @Test
    void shouldIgnoreUnknownFields() throws Exception {
        String json = "{" +
                "\"name\":\"created_at\"," +
                "\"type\":\"TIMESTAMP\"," +
                "\"nullable\":true," +
                "\"extraField\":\"ignored\"" +
                "}";

        ObjectMapper mapper = new ObjectMapper();
        ColumnResponse result = mapper.readValue(json, ColumnResponse.class);

        assertEquals("created_at", result.getName());
        assertEquals("TIMESTAMP", result.getType());
        assertTrue(result.isNullable());
        assertNull(result.getLength());
    }
}
