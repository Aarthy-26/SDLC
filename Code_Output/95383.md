User Story ID: 95383

1. **Requirement Summary**

Fix the Jackson JSON deserialization failure for `ColumnResponse` used by the App Import API by ensuring the DTO can be constructed during deserialization.

Assumptions:
- The failure occurs because `ColumnResponse` is a POJO/DTO without a public no-args constructor and without a valid Jackson `@JsonCreator` path.
- The API uses Jackson (Spring Boot default) and needs to deserialize JSON into `ColumnResponse` (or a container that includes it).
- We will implement an immutable DTO with explicit Jackson annotations (safer than relying on a no-args constructor) and also provide a no-args constructor for maximum compatibility.
- Java 17, Spring Boot 3.2.x, Maven build.
- Any unknown JSON fields should not break deserialization (forward compatibility).

        return args -> {
            String json = "{\"columns\":[{\"name\":\"id\",\"type\":\"integer\",\"nullable\":false}]}";
            ImportRequest request = mapper.readValue(json, ImportRequest.class);
            ColumnResponse first = request.columns().get(0);

        };
    }

package com.aava.datastudio.appimport.dto;

import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.annotation.JsonProperty;
import jakarta.validation.constraints.NotBlank;

/**
 * DTO representing a column definition/response element used by the App Import API.
 *
 * This class is intentionally designed to be safely deserializable by Jackson:
 * - Provides a public no-args constructor (legacy / reflective frameworks)
 * - Provides an explicit @JsonCreator constructor for immutable-style creation
 * - Ignores unknown properties for forward compatibility
 */
@JsonIgnoreProperties(ignoreUnknown = true)
public final class ColumnResponse {

    @NotBlank
    private String name;

    private String type;

    private Boolean nullable;

    /**
     * Public no-args constructor required by certain Jackson configurations and frameworks.
     */
    public ColumnResponse() {
        // Intentionally empty.
    }

    /**
     * Jackson creator constructor. Prefer this for reliable deserialization.
     */
    @JsonCreator
    public ColumnResponse(
            @JsonProperty(value = "name", required = true) String name,
            @JsonProperty("type") String type,
            @JsonProperty("nullable") Boolean nullable) {
        this.name = name;
        this.type = type;
        this.nullable = nullable;
    }

    public String name() {
        return name;
    }

    public String type() {
        return type;
    }

    public Boolean nullable() {
        return nullable;
    }

    public void setName(String name) {
        this.name = name;
    }

    public void setType(String type) {
        this.type = type;
    }

    public void setNullable(Boolean nullable) {
        this.nullable = nullable;
    }

    @Override
    public String toString() {
        return "ColumnResponse{" +
                "name='" + name + '\'' +
                ", type='" + type + '\'' +
                ", nullable=" + nullable +
                '}';
    }
}

        @JsonProperty("columns")

    public ImportRequest {
        // Defensive copy + avoid null list surprises.

            columns = new ArrayList<>();
        } else {
            columns = List.copyOf(columns);

package com.aava.datastudio.appimport.web;

import com.fasterxml.jackson.databind.exc.InvalidDefinitionException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.time.OffsetDateTime;
import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Centralized error responses to make Jackson/validation failures explicit.
 */
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, Object>> handleValidation(MethodArgumentNotValidException ex) {
        Map<String, Object> body = new LinkedHashMap<>();
        body.put("timestamp", OffsetDateTime.now().toString());
        body.put("error", "validation_error");
        body.put("status", HttpStatus.BAD_REQUEST.value());

        Map<String, String> fields = new LinkedHashMap<>();
        for (FieldError fe : ex.getBindingResult().getFieldErrors()) {
            fields.put(fe.getField(), fe.getDefaultMessage());
        }
        body.put("fieldErrors", fields);

        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ResponseEntity<Map<String, Object>> handleNotReadable(HttpMessageNotReadableException ex) {
        Map<String, Object> body = new LinkedHashMap<>();
        body.put("timestamp", OffsetDateTime.now().toString());
        body.put("error", "invalid_json");
        body.put("status", HttpStatus.BAD_REQUEST.value());

        Throwable cause = ex.getCause();
        if (cause instanceof InvalidDefinitionException ide) {
            body.put("message", "Jackson cannot construct one of the request DTOs: " + ide.getType());
        } else {
            body.put("message", "Request body is not readable JSON.");
        }

        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }
}
