User Story ID: 95383

1. **Requirement Summary**

Fix the App Import API failure caused by a Jackson deserialization error when mapping JSON into `ColumnResponse` due to a missing no-args constructor.

Assumptions:
- The API uses Jackson for JSON (de)serialization.
- `ColumnResponse` is a DTO returned/consumed by the App Import API and must be deserializable from JSON.
- Enterprise default is to support Jackson by providing either:
  - an explicit public no-args constructor, and
  - explicit Jackson annotations when needed to avoid ambiguity.
- Java version: 17.
- Build tool: Maven.

 * Demonstration controller showing Jackson deserialization into ColumnResponse.
 * In a real system, ColumnResponse may be nested in other structures.
 */
@RestController
@RequestMapping(value = "/api/import", produces = MediaType.APPLICATION_JSON_VALUE)
public class AppImportController {

    /**
     * Echo endpoint:
     * - Accepts an ImportRequest that includes a list of ColumnResponse items.
     * - Returns the same list.
     *
     * If ColumnResponse lacks a no-args constructor (or a proper @JsonCreator), Jackson will fail.
     */
    @PostMapping(consumes = MediaType.APPLICATION_JSON_VALUE)
    public List<ColumnResponse> importApp(@Valid @RequestBody ImportRequest request) {
        // Defensively copy

            result.addAll(request.columns());
        }
        return result;
    }

### `src/main/java/com/ascendion/aava/appimport/dto/ColumnResponse.java`
```java
package com.ascendion.aava.appimport.dto;

import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.annotation.JsonProperty;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;

/**
 * DTO representing a column returned by/used in the App Import API.
 *
 * Key fix:
 * - Provide a public no-args constructor so Jackson can instantiate the object.
 * - Provide getters/setters for property binding.
 * - Ignore unknown properties to reduce brittleness across API versions.
 */
@JsonIgnoreProperties(ignoreUnknown = true)
public class ColumnResponse {

    @NotBlank(message = "name is required")
    @Size(max = 200, message = "name must be <= 200 characters")
    private String name;

    @Size(max = 100, message = "type must be <= 100 characters")
    private String type;

    private boolean nullable;

    /**
     * No-args constructor required by Jackson for default POJO deserialization.
     */
    public ColumnResponse() {
        // intentionally empty
    }

    /**
     * Convenience constructor for code usage; also safe for Jackson when used with @JsonCreator.
     */
    @JsonCreator
    public ColumnResponse(
            @JsonProperty("name") String name,
            @JsonProperty("type") String type,
            @JsonProperty("nullable") boolean nullable
    ) {
        this.name = name;
        this.type = type;
        this.nullable = nullable;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getType() {
        return type;
    }

    public void setType(String type) {
        this.type = type;
    }

    public boolean isNullable() {
        return nullable;
    }

    public void setNullable(boolean nullable) {
        this.nullable = nullable;
    }
}
```

package com.ascendion.aava.appimport;

import com.ascendion.aava.appimport.dto.ColumnResponse;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertNotNull;

/**
 * Ensures ColumnResponse can be deserialized (regression test for missing no-args constructor).
 */
public class JacksonDeserializationTest {

    private final ObjectMapper objectMapper = new ObjectMapper();

    @Test
    void shouldDeserializeColumnResponse() throws Exception {
        String json = "{\"name\":\"colA\",\"type\":\"STRING\",\"nullable\":true}";
        ColumnResponse dto = objectMapper.readValue(json, ColumnResponse.class);

        assertNotNull(dto);
        assertEquals("colA", dto.getName());
        assertEquals("STRING", dto.getType());
        assertEquals(true, dto.isNullable());
    }
}
