User Story ID: 95383

1. **Requirement Summary** — the clarified, confirmed requirement

Fix the App Import API failure caused by a Jackson deserialization error when mapping JSON into `ColumnResponse` because the class lacks an accessible no-args constructor (or equivalent Jackson creator).

Assumptions:
- The API uses Jackson (via Spring Boot or a similar stack) for JSON (de)serialization.
- `ColumnResponse` is a DTO/POJO used in request/response payloads for the App Import API.
- The issue can be resolved by adding a public no-args constructor and ensuring Jackson can bind properties via standard getters/setters (or by using `@JsonCreator`/`@JsonProperty`).
- Maven is the build tool.

package com.example.appimport.dto;

import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.annotation.JsonProperty;

import java.util.Objects;

/**
 * DTO representing a column returned by the App Import API.
 *
 * <p>Fixes Jackson deserialization errors by providing:
 * <ul>
 *   <li>a public no-args constructor (required by many Jackson configurations)</li>
 *   <li>standard getters/setters for property binding</li>
 *   <li>an optional {@link JsonCreator} constructor for immutable-style creation</li>
 * </ul>
 */
@JsonIgnoreProperties(ignoreUnknown = true)
public class ColumnResponse {

    private String name;
    private String type;
    private Boolean nullable;

    /**
     * Public no-args constructor required for Jackson default POJO deserialization.
     */
    public ColumnResponse() {
        // Intentionally empty.
    }

    /**
     * Optional creator constructor enabling constructor-based deserialization.
     * This is safe to keep alongside the no-args constructor.
     */
    @JsonCreator
    public ColumnResponse(
            @JsonProperty("name") String name,
            @JsonProperty("type") String type,
            @JsonProperty("nullable") Boolean nullable) {
        this.name = name;
        this.type = type;
        this.nullable = nullable;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getType() {
        return type;
    }

    public void setType(String type) {
        this.type = type;
    }

    public Boolean getNullable() {
        return nullable;
    }

    public void setNullable(Boolean nullable) {
        this.nullable = nullable;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        ColumnResponse that = (ColumnResponse) o;
        return Objects.equals(name, that.name)
                && Objects.equals(type, that.type)
                && Objects.equals(nullable, that.nullable);
    }

    @Override
    public int hashCode() {
        return Objects.hash(name, type, nullable);
    }

    @Override
    public String toString() {
        return "ColumnResponse{" +
                "name='" + name + '\'' +
                ", type='" + type + '\'' +
                ", nullable=" + nullable +
                '}';
    }
}

package com.example.appimport.dto;

import com.fasterxml.jackson.databind.DeserializationFeature;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertNotNull;

/**
 * Verifies that {@link ColumnResponse} can be deserialized by Jackson without errors.
 */
public class ColumnResponseJacksonTest {

    @Test
    void shouldDeserializeWithNoArgsConstructorAndSetters() throws Exception {
        ObjectMapper mapper = new ObjectMapper()
                .configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);

        String json = "{\"name\":\"colA\",\"type\":\"STRING\",\"nullable\":true}";

        ColumnResponse response = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(response);
        assertEquals("colA", response.getName());
        assertEquals("STRING", response.getType());
        assertEquals(Boolean.TRUE, response.getNullable());
    }

    @Test
    void shouldIgnoreUnknownFields() throws Exception {
        ObjectMapper mapper = new ObjectMapper();

        String json = "{\"name\":\"colB\",\"type\":\"INT\",\"nullable\":false,\"unknown\":123}";

        ColumnResponse response = mapper.readValue(json, ColumnResponse.class);

        assertNotNull(response);
        assertEquals("colB", response.getName());
        assertEquals("INT", response.getType());
        assertEquals(Boolean.FALSE, response.getNullable());
    }
}
