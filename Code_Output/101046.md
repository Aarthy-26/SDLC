package com.aava.secrets;

import com.aava.secrets.vault.VaultClient;
import com.aava.secrets.vault.VaultClientException;
import com.aava.secrets.vault.VaultConfig;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.nio.charset.StandardCharsets;
import java.util.Base64;
import java.util.Map;
import java.util.Objects;

/**
 * CLI utility that securely writes a Databricks access token into AAVA Secret Management (Vault) for INT.
 *
 * <p>Security notes:
 * <ul>
 *   <li>No secrets are hardcoded.</li>
 *   <li>The Databricks token is never logged.</li>
 *   <li>Vault authentication uses AppRole (roleId/secretId) passed via environment variables.</li>
 * </ul>
 *
 * <p>Usage:
 * <pre>
 * export VAULT_ADDR=https://vault.int.example.com
 * export VAULT_NAMESPACE=aava   # optional, if Vault Enterprise namespaces are used
 * export VAULT_APPROLE_ROLE_ID=...   # do not print
 * export VAULT_APPROLE_SECRET_ID=... # do not print
 * export DATABRICKS_ACCESS_TOKEN=... # do not print
 *
 * # optional overrides
 * export VAULT_SECRET_PATH=secret/aava/int/databricks/agent-access-token
 * export VAULT_KV_VERSION=2
 *
 * mvn -q -DskipTests package
 * java -jar target/aava-databricks-token-secret-1.0.0.jar
 * </pre>
 */
public final class DatabricksTokenSecretUpdater {

    private static final Logger LOGGER = LoggerFactory.getLogger(DatabricksTokenSecretUpdater.class);

    public static void main(String[] args) {
        try {
            VaultConfig config = VaultConfig.fromEnvironment();
            String databricksToken = readRequiredEnv("DATABRICKS_ACCESS_TOKEN");

            // Basic validation to prevent accidental empty writes.
            if (databricksToken.isBlank()) {
                throw new IllegalArgumentException("DATABRICKS_ACCESS_TOKEN must not be blank.");
            }

            // Never log token; only log a deterministic mask.
            String masked = mask(databricksToken);
            LOGGER.info("Preparing to write Databricks access token to Vault path '{}' (token={})",
                    config.secretPath(), masked);

            try (VaultClient vault = VaultClient.create(config)) {
                vault.loginWithAppRole(config.approleRoleId(), config.approleSecretId());

                // Store token under a clear key while not revealing it.
                vault.writeSecret(config.secretPath(), Map.of(
                        "databricks_access_token", databricksToken
                ));

                // Read back metadata (not the secret value) by verifying that secret exists.
                // For KV v2, a read will return value; we avoid printing it.
                // Instead, we just attempt a read and ensure key exists.
                Map<String, Object> read = vault.readSecret(config.secretPath());
                if (!read.containsKey("databricks_access_token")) {
                    throw new IllegalStateException("Secret write verification failed: key not found after write.");
                }

                LOGGER.info("Secret stored successfully at '{}' and verified (value not displayed).", config.secretPath());
            }
        } catch (Exception e) {
            // Fail fast with a clear message, no token exposure.
            LOGGER.error("Failed to store Databricks token in secret management: {}", safeMessage(e));
            System.exit(1);
        }
    }

    private static String readRequiredEnv(String name) {
        String v = System.getenv(name);
        if (v == null) {
            throw new IllegalArgumentException("Missing required environment variable: " + name);
        }
        return v;
    }

    /**
     * Masks sensitive values while keeping a small prefix/suffix for troubleshooting.
     */
    static String mask(String token) {
        Objects.requireNonNull(token, "token");
        String t = token.trim();
        if (t.length() <= 8) {
            return "***";
        }
        String prefix = t.substring(0, 3);
        String suffix = t.substring(t.length() - 3);
        // Include length and a short hash-like fingerprint without revealing the token.
        String fingerprint = Base64.getUrlEncoder().withoutPadding()
                .encodeToString(("fp:" + t.length()).getBytes(StandardCharsets.UTF_8));
        return prefix + "***" + suffix + "(len=" + t.length() + "," + fingerprint + ")";
    }

    private static String safeMessage(Exception e) {
        if (e instanceof VaultClientException) {
            // Vault errors can include request path/status; ensure no secret value included.
            return e.getMessage();
        }
        String m = e.getMessage();
        return (m == null || m.isBlank()) ? e.getClass().getSimpleName() : m;
    }
}

package com.aava.secrets.vault;

import java.net.URI;
import java.util.Locale;

/**
 * Immutable configuration for Vault connectivity and secret location.
 */
public record VaultConfig(
        URI vaultAddress,
        String namespace,
        int kvVersion,
        String secretPath,
        String approleRoleId,
        String approleSecretId
) {

    public static VaultConfig fromEnvironment() {
        String addr = requireEnv("VAULT_ADDR");
        String ns = System.getenv("VAULT_NAMESPACE");

        String kvVersionRaw = System.getenv().getOrDefault("VAULT_KV_VERSION", "2").trim();
        int kv;
        try {
            kv = Integer.parseInt(kvVersionRaw);
        } catch (NumberFormatException e) {
            throw new IllegalArgumentException("VAULT_KV_VERSION must be 1 or 2.");
        }
        if (kv != 1 && kv != 2) {
            throw new IllegalArgumentException("VAULT_KV_VERSION must be 1 or 2.");
        }

        String defaultPath = "secret/aava/int/databricks/agent-access-token";
        String path = System.getenv().getOrDefault("VAULT_SECRET_PATH", defaultPath).trim();
        if (path.isEmpty()) {
            throw new IllegalArgumentException("VAULT_SECRET_PATH must not be empty.");
        }

        String roleId = requireEnv("VAULT_APPROLE_ROLE_ID");
        String secretId = requireEnv("VAULT_APPROLE_SECRET_ID");

        URI uri;
        try {
            uri = URI.create(addr);
        } catch (IllegalArgumentException e) {
            throw new IllegalArgumentException("VAULT_ADDR must be a valid URI.");
        }
        String scheme = uri.getScheme();
        if (scheme == null || !(scheme.toLowerCase(Locale.ROOT).equals("http") || scheme.toLowerCase(Locale.ROOT).equals("https"))) {
            throw new IllegalArgumentException("VAULT_ADDR must start with http:// or https://");
        }

        return new VaultConfig(uri, (ns == null || ns.isBlank()) ? null : ns.trim(), kv, path, roleId.trim(), secretId.trim());
    }

    private static String requireEnv(String name) {
        String v = System.getenv(name);
        if (v == null || v.isBlank()) {
            throw new IllegalArgumentException("Missing required environment variable: " + name);
        }
        return v;
    }
}

package com.aava.secrets.vault;

import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.apache.hc.client5.http.classic.CloseableHttpClient;
import org.apache.hc.client5.http.classic.methods.HttpGet;
import org.apache.hc.client5.http.classic.methods.HttpPost;
import org.apache.hc.client5.http.config.RequestConfig;
import org.apache.hc.client5.http.impl.classic.CloseableHttpClientBuilder;
import org.apache.hc.client5.http.impl.classic.HttpClients;
import org.apache.hc.core5.http.ClassicHttpResponse;
import org.apache.hc.core5.http.ContentType;
import org.apache.hc.core5.http.HttpHeaders;
import org.apache.hc.core5.http.HttpStatus;
import org.apache.hc.core5.http.io.entity.EntityUtils;
import org.apache.hc.core5.http.io.entity.StringEntity;
import org.apache.hc.core5.util.Timeout;

import java.io.Closeable;
import java.io.IOException;
import java.net.URI;
import java.nio.charset.StandardCharsets;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Objects;

/**
 * Minimal Vault HTTP client supporting:
 * <ul>
 *     <li>AppRole login</li>
 *     <li>KV v1/v2 write and read</li>
 * </ul>
 */
public final class VaultClient implements Closeable {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    private final VaultConfig config;
    private final CloseableHttpClient http;
    private volatile String clientToken;

    private VaultClient(VaultConfig config) {
        this.config = Objects.requireNonNull(config, "config");

        RequestConfig requestConfig = RequestConfig.custom()
                .setConnectionRequestTimeout(Timeout.ofSeconds(10))
                .setConnectTimeout(Timeout.ofSeconds(10))
                .setResponseTimeout(Timeout.ofSeconds(20))
                .build();

        CloseableHttpClientBuilder builder = HttpClients.custom()
                .setDefaultRequestConfig(requestConfig);

        this.http = builder.build();
    }

    public static VaultClient create(VaultConfig config) {
        return new VaultClient(config);
    }

    /**
     * Authenticates against Vault using AppRole and stores the client token in-memory.
     */
    public void loginWithAppRole(String roleId, String secretId) {
        Objects.requireNonNull(roleId, "roleId");
        Objects.requireNonNull(secretId, "secretId");

        URI uri = config.vaultAddress().resolve("/v1/auth/approle/login");

        Map<String, Object> payload = new LinkedHashMap<>();
        payload.put("role_id", roleId);
        payload.put("secret_id", secretId);

        Map<String, Object> resp = postJson(uri, payload, false);
        Object authObj = resp.get("auth");
        if (!(authObj instanceof Map<?, ?> authMap)) {
            throw new VaultClientException("Vault AppRole login failed: missing 'auth' in response");
        }
        Object tokenObj = authMap.get("client_token");
        if (!(tokenObj instanceof String token) || token.isBlank()) {
            throw new VaultClientException("Vault AppRole login failed: missing 'client_token' in response");
        }
        this.clientToken = token;
    }

    /**
     * Writes secret data to the configured KV path.
     */
    public void writeSecret(String secretPath, Map<String, ?> data) {
        ensureLoggedIn();
        Objects.requireNonNull(secretPath, "secretPath");
        Objects.requireNonNull(data, "data");

        if (data.isEmpty()) {
            throw new IllegalArgumentException("Secret data must not be empty.");
        }

        URI uri = config.vaultAddress().resolve("/v1/" + normalizePathForWrite(secretPath));

        Map<String, Object> payload;
        if (config.kvVersion() == 2) {
            payload = Map.of("data", new LinkedHashMap<>(data));
        } else {
            payload = new LinkedHashMap<>(data);
        }

        postJson(uri, payload, true);
    }

    /**
     * Reads secret data from the configured KV path.
     * Returns a map of key/value pairs in the secret.
     */
    public Map<String, Object> readSecret(String secretPath) {
        ensureLoggedIn();
        Objects.requireNonNull(secretPath, "secretPath");

        URI uri = config.vaultAddress().resolve("/v1/" + normalizePathForRead(secretPath));

        Map<String, Object> resp = getJson(uri, true);
        Object dataObj = resp.get("data");
        if (!(dataObj instanceof Map<?, ?> dataMap)) {
            throw new VaultClientException("Vault read failed: missing 'data' in response");
        }

        if (config.kvVersion() == 2) {
            Object nested = dataMap.get("data");
            if (!(nested instanceof Map<?, ?> nestedMap)) {
                throw new VaultClientException("Vault read failed: missing 'data.data' in KV v2 response");
            }
            Map<String, Object> out = new LinkedHashMap<>();
            for (Map.Entry<?, ?> e : nestedMap.entrySet()) {
                if (e.getKey() != null) {
                    out.put(String.valueOf(e.getKey()), e.getValue());
                }
            }
            return out;
        }

        Map<String, Object> out = new LinkedHashMap<>();
        for (Map.Entry<?, ?> e : dataMap.entrySet()) {
            if (e.getKey() != null) {
                out.put(String.valueOf(e.getKey()), e.getValue());
            }
        }
        return out;
    }

    private String normalizePathForWrite(String secretPath) {
        String p = trimSlashes(secretPath);
        if (config.kvVersion() == 2) {
            // KV v2 write path is /v1/<mount>/data/<path>
            // If user provides secret/<path>, assume mount=secret
            return toKv2DataPath(p);
        }
        return p;
    }

    private String normalizePathForRead(String secretPath) {
        String p = trimSlashes(secretPath);
        if (config.kvVersion() == 2) {
            // KV v2 read path is /v1/<mount>/data/<path>
            return toKv2DataPath(p);
        }
        return p;
    }

    private static String toKv2DataPath(String path) {
        // If path already contains "/data/" after the mount, keep as-is.
        // Otherwise, convert "<mount>/<rest>" -> "<mount>/data/<rest>".
        String p = trimSlashes(path);
        int firstSlash = p.indexOf('/');
        if (firstSlash < 0) {
            // Only mount provided; treat secret as a key at mount root.
            return p + "/data";
        }
        String mount = p.substring(0, firstSlash);
        String rest = p.substring(firstSlash + 1);
        if (rest.startsWith("data/")) {
            return p;
        }
        return mount + "/data/" + rest;
    }

    private static String trimSlashes(String s) {
        String p = Objects.requireNonNull(s, "s").trim();
        while (p.startsWith("/")) {
            p = p.substring(1);
        }
        while (p.endsWith("/")) {
            p = p.substring(0, p.length() - 1);
        }
        if (p.isEmpty()) {
            throw new IllegalArgumentException("Secret path must not be empty.");
        }
        return p;
    }

    private void ensureLoggedIn() {
        if (clientToken == null || clientToken.isBlank()) {
            throw new VaultClientException("Not authenticated to Vault. Call loginWithAppRole() first.");
        }
    }

    private Map<String, Object> postJson(URI uri, Map<String, ?> payload, boolean includeAuth) {
        HttpPost post = new HttpPost(uri);
        post.setHeader(HttpHeaders.ACCEPT, ContentType.APPLICATION_JSON.getMimeType());
        post.setHeader(HttpHeaders.CONTENT_TYPE, ContentType.APPLICATION_JSON.getMimeType());
        if (config.namespace() != null) {
            post.setHeader("X-Vault-Namespace", config.namespace());
        }
        if (includeAuth) {
            post.setHeader("X-Vault-Token", clientToken);
        }

        String body;
        try {
            body = MAPPER.writeValueAsString(payload);
        } catch (IOException e) {
            throw new VaultClientException("Failed to serialize JSON payload", e);
        }
        post.setEntity(new StringEntity(body, ContentType.APPLICATION_JSON));

        try (ClassicHttpResponse resp = http.executeOpen(null, post, null)) {
            int status = resp.getCode();
            String respBody = resp.getEntity() == null ? "" : EntityUtils.toString(resp.getEntity(), StandardCharsets.UTF_8);

            if (status < 200 || status >= 300) {
                throw new VaultClientException("Vault HTTP " + status + " for POST " + uri.getPath() + ": " + sanitizeVaultError(respBody));
            }

            if (respBody.isBlank()) {
                return Map.of();
            }
            return MAPPER.readValue(respBody, new TypeReference<>() {});
        } catch (IOException e) {
            throw new VaultClientException("Vault POST request failed: " + uri, e);
        }
    }

    private Map<String, Object> getJson(URI uri, boolean includeAuth) {
        HttpGet get = new HttpGet(uri);
        get.setHeader(HttpHeaders.ACCEPT, ContentType.APPLICATION_JSON.getMimeType());
        if (config.namespace() != null) {
            get.setHeader("X-Vault-Namespace", config.namespace());
        }
        if (includeAuth) {
            get.setHeader("X-Vault-Token", clientToken);
        }

        try (ClassicHttpResponse resp = http.executeOpen(null, get, null)) {
            int status = resp.getCode();
            String respBody = resp.getEntity() == null ? "" : EntityUtils.toString(resp.getEntity(), StandardCharsets.UTF_8);

            if (status == HttpStatus.SC_NOT_FOUND) {
                throw new VaultClientException("Secret not found at path: " + uri.getPath());
            }
            if (status < 200 || status >= 300) {
                throw new VaultClientException("Vault HTTP " + status + " for GET " + uri.getPath() + ": " + sanitizeVaultError(respBody));
            }

            if (respBody.isBlank()) {
                return Map.of();
            }
            return MAPPER.readValue(respBody, new TypeReference<>() {});
        } catch (IOException e) {
            throw new VaultClientException("Vault GET request failed: " + uri, e);
        }
    }

    /**
     * Sanitizes Vault error response body to reduce risk of echoing sensitive data.
     * Vault should not include secrets in error responses, but we defensively truncate.
     */
    private static String sanitizeVaultError(String body) {
        if (body == null) {
            return "";
        }
        String trimmed = body.trim();
        int max = 512;
        if (trimmed.length() > max) {
            return trimmed.substring(0, max) + "...";
        }
        return trimmed;
    }

    @Override
    public void close() throws IOException {
        http.close();
    }
}
