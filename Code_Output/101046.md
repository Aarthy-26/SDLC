1. **Requirement Summary** — Store a provided Databricks access token securely in AAVA Secret Management and ensure it is retrievable only by authorized applications in the INT environment.

**Assumptions**:
- AAVA Secret Management is backed by Azure Key Vault for the INT environment.
- Authentication to Key Vault is performed via Azure AD DefaultAzureCredential (Managed Identity/service principal) and not via hardcoded secrets.
- The secret name is configurable and defaults to `databricks-access-token`.
- Authorized applications are controlled by Key Vault access policies/RBAC and are out of scope for code, but the code enforces no secret value is logged.
- Update semantics: setting the secret with the same name overwrites the existing value (Key Vault versioning applies).

// src/main/java/com/aava/secrets/DatabricksTokenSecretUploader.java
package com.aava.secrets;

import com.azure.identity.DefaultAzureCredential;
import com.azure.identity.DefaultAzureCredentialBuilder;
import com.azure.security.keyvault.secrets.SecretClient;
import com.azure.security.keyvault.secrets.SecretClientBuilder;
import com.azure.security.keyvault.secrets.models.KeyVaultSecret;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import java.nio.charset.StandardCharsets;
import java.util.Base64;
import java.util.Locale;
import java.util.Objects;

/**
 * Uploads (creates/updates) a Databricks access token into AAVA Secret Management.
 *
 * <p>Enterprise security defaults:
 * <ul>
 *   <li>No secret value is ever logged.</li>
 *   <li>Uses Azure AD authentication via {@link DefaultAzureCredential} (Managed Identity/service principal).</li>
 *   <li>Validates inputs and fails fast with explicit error messages.</li>
 * </ul>
 *
 * <h3>How to run (INT environment)</h3>
 * <pre>
 * mvn -q -DskipTests package
 * java -jar target/databricks-token-secret-uploader-1.0.0.jar \
 *   --vaultUrl=https://&lt;your-int-keyvault-name&gt;.vault.azure.net/ \
 *   --secretName=databricks-access-token \
 *   --tokenEnv=DATABRICKS_ACCESS_TOKEN
 * </pre>
 *
 * The token is read from an environment variable by default (recommended) and can also be supplied as base64.
 */
public final class DatabricksTokenSecretUploader {

    private static final Logger LOGGER = LoggerFactory.getLogger(DatabricksTokenSecretUploader.class);

    private DatabricksTokenSecretUploader() {
        // utility
    }

    /**
     * Entry point.
     *
     * <p>Supported arguments:
     * <ul>
     *   <li>--vaultUrl (required): Key Vault URL for INT.</li>
     *   <li>--secretName (optional): defaults to databricks-access-token.</li>
     *   <li>--tokenEnv (optional): env var name containing token, defaults to DATABRICKS_ACCESS_TOKEN.</li>
     *   <li>--tokenBase64 (optional): token as base64 string (prefer env var).</li>
     *   <li>--contentType (optional): defaults to text/plain.</li>
     * </ul>
     */
    public static void main(String[] args) {
        CliOptions options = CliOptions.parse(args);

        String vaultUrl = requireNonBlank(options.vaultUrl(), "vaultUrl");
        String secretName = defaultIfBlank(options.secretName(), "databricks-access-token");
        String tokenEnv = defaultIfBlank(options.tokenEnv(), "DATABRICKS_ACCESS_TOKEN");
        String contentType = defaultIfBlank(options.contentType(), "text/plain");

        String token = resolveToken(options, tokenEnv);
        validateToken(token);
        validateSecretName(secretName);

        DefaultAzureCredential credential = new DefaultAzureCredentialBuilder().build();
        SecretClient client = new SecretClientBuilder()
                .vaultUrl(vaultUrl)
                .credential(credential)
                .buildClient();

        try {
            KeyVaultSecret secret = new KeyVaultSecret(secretName, token);
            secret.getProperties().setContentType(contentType);

            // Set secret (creates new version if exists)
            KeyVaultSecret result = client.setSecret(secret);

            // Never log secret value
            LOGGER.info("Secret stored successfully. name='{}', version='{}', vault='{}'",
                    secretName,
                    safe(result.getProperties().getVersion()),
                    vaultUrl);

            // Optional check: attempt retrieval metadata only (still do not log value)
            KeyVaultSecret fetched = client.getSecret(secretName);
            LOGGER.info("Secret is retrievable for this identity. name='{}', enabled={}, createdOn={}, updatedOn={}",
                    secretName,
                    fetched.getProperties().isEnabled(),
                    fetched.getProperties().getCreatedOn(),
                    fetched.getProperties().getUpdatedOn());

        } catch (com.azure.core.exception.ClientAuthenticationException e) {
            // Authn failure: identity not configured
            System.err.println("Authentication to Secret Management failed. Ensure Managed Identity/Service Principal is configured for INT. "
                    + "Details=" + sanitizeForStderr(e.getMessage()));
            System.exit(2);
        } catch (com.azure.core.exception.ResourceNotFoundException e) {
            System.err.println("Key Vault not found or secret not accessible. Check vaultUrl and network/firewall settings for INT. "
                    + "Details=" + sanitizeForStderr(e.getMessage()));
            System.exit(3);
        } catch (com.azure.core.exception.HttpResponseException e) {
            // Includes authorization issues (403)
            System.err.println("Secret Management request failed. Ensure this identity has 'set' permissions for the secret and INT Key Vault access. "
                    + "Status=" + e.getResponse().getStatusCode() + ", Details=" + sanitizeForStderr(e.getMessage()));
            System.exit(4);
        } catch (IllegalArgumentException e) {
            System.err.println("Invalid input: " + e.getMessage());
            System.exit(5);
        }
    }

    private static String resolveToken(CliOptions options, String tokenEnv) {
        if (isNonBlank(options.tokenBase64())) {
            return decodeBase64(options.tokenBase64().trim());
        }
        String fromEnv = System.getenv(tokenEnv);
        return fromEnv == null ? "" : fromEnv.trim();
    }

    private static void validateToken(String token) {
        if (!isNonBlank(token)) {
            throw new IllegalArgumentException("Databricks token is required. Provide via env var or --tokenBase64.");
        }
        if (token.length() < 10) {
            throw new IllegalArgumentException("Databricks token looks too short; refusing to upload.");
        }
    }

    private static void validateSecretName(String secretName) {
        // Azure Key Vault secret naming: alphanumeric and '-' (simple policy)
        // We keep it strict to avoid surprises.
        if (!secretName.matches("^[0-9A-Za-z-]{1,127}$")) {
            throw new IllegalArgumentException("secretName must match ^[0-9A-Za-z-]{1,127}$");
        }
    }

    private static String decodeBase64(String base64) {
        try {
            byte[] decoded = Base64.getDecoder().decode(base64);
            return new String(decoded, StandardCharsets.UTF_8).trim();
        } catch (IllegalArgumentException e) {
            throw new IllegalArgumentException("tokenBase64 is not valid base64.");
        }
    }

    private static String safe(String value) {
        return value == null ? "" : value;
    }

    private static String sanitizeForStderr(String message) {
        if (message == null) {
            return "";
        }
        // Defensive: avoid echoing token-like strings if they appear in exception messages.
        // This is a best-effort redaction.
        return message.replaceAll("(?i)(token|secret|access[_-]?token)=[^\\s,;]+", "$1=<redacted>");
    }

    private static boolean isNonBlank(String s) {
        return s != null && !s.trim().isEmpty();
    }

    private static String requireNonBlank(String s, String fieldName) {
        if (!isNonBlank(s)) {
            throw new IllegalArgumentException(fieldName + " is required and must be non-blank.");
        }
        return s.trim();
    }

    private static String defaultIfBlank(String s, String defaultValue) {
        return isNonBlank(s) ? s.trim() : Objects.requireNonNull(defaultValue, "defaultValue");
    }

    /**
     * Minimal CLI options parser (no external CLI dependency).
     */
    private record CliOptions(String vaultUrl, String secretName, String tokenEnv, String tokenBase64, String contentType) {

        static CliOptions parse(String[] args) {
            String vaultUrl = null;
            String secretName = null;
            String tokenEnv = null;
            String tokenBase64 = null;
            String contentType = null;

            if (args != null) {
                for (String arg : args) {
                    if (arg == null) {
                        continue;
                    }
                    String a = arg.trim();
                    if (a.isEmpty()) {
                        continue;
                    }
                    if (!a.startsWith("--") || !a.contains("=")) {
                        throw new IllegalArgumentException("Invalid argument format: '" + a + "'. Expected --key=value");
                    }
                    String[] kv = a.substring(2).split("=", 2);
                    String key = kv[0].trim().toLowerCase(Locale.ROOT);
                    String value = kv.length > 1 ? kv[1].trim() : "";

                    switch (key) {
                        case "vaulturl" -> vaultUrl = value;
                        case "secretname" -> secretName = value;
                        case "tokenenv" -> tokenEnv = value;
                        case "tokenbase64" -> tokenBase64 = value;
                        case "contenttype" -> contentType = value;
                        default -> throw new IllegalArgumentException("Unknown argument: --" + key);
                    }
                }
            }

            return new CliOptions(vaultUrl, secretName, tokenEnv, tokenBase64, contentType);
        }
    }
}
