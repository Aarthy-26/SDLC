1. **Requirement Summary** — the clarified, confirmed requirement

Build a Java program (standalone, single-file by default) that analyzes historical test execution results across multiple days and identifies:
- **Recurring Failures**: test cases that have failed continuously across multiple consecutive days, including the number of consecutive failed days.
- **Inconsistent/Flaky Failures**: test cases that show pass/fail fluctuations (e.g., Pass → Fail → Pass → Fail) and test cases that fail for a period and later return to a passing state.
- Provide **historical evidence** for each identified test case (a dated sequence of outcomes) to support investigation.

**Assumptions** (enterprise-safe defaults):
- Input is a UTF-8 CSV file with header: `date,testCaseId,status` where:
  - `date` is `yyyy-MM-dd`
  - `status` is `PASS` or `FAIL` (case-insensitive)
- Multiple executions for the same test on the same date are allowed; the **worst** status for that date is used (FAIL overrides PASS).
- Missing dates are ignored (analysis uses observed execution days per test case, ordered by date).
- A test is considered a **Recurring Failure** if it has **>= 2** consecutive observed execution days ending in FAIL.
- A test is considered **Inconsistent** if either:
  - it contains an alternating pattern `P,F,P,F` or `F,P,F,P` in any 4 consecutive observed execution days, OR
  - it has at least one `FAIL → PASS` recovery after having a run of **>= 2** consecutive FAILs.
- Output is printed to console as a readable report; no external systems, authN/authZ needed.

    enum Status {

        static Status parse(String raw) {

            }

                return FAIL;

            Objects.requireNonNull(a, "a");
            Objects.requireNonNull(b, "b");
            return (a == FAIL || b == FAIL) ? FAIL : PASS;
        }

        char shortCode() {
            return this == PASS ? 'P' : 'F';

    static final class FailurePatternAnalyzer {

        AnalysisReport analyze(List<ExecutionRecord> records) {
            if (records == null) {
                throw new InvalidInputException("Records list is required.");
            }
            if (records.isEmpty()) {
                return new AnalysisReport(List.of(), List.of(), 0);
            }

            // Aggregate to per-testCase per-day using worst status for the day.
            Map<String, Map<LocalDate, Status>> aggregated = new HashMap<>();
            for (ExecutionRecord r : records) {
                if (r == null) {
                    continue;
                }
                aggregated
                        .computeIfAbsent(r.testCaseId(), k -> new HashMap<>())
                        .merge(r.date(), r.status(), Status::worstOf);
            }

            // Build timelines ordered by date.
            Map<String, List<DailyOutcome>> timelines = new TreeMap<>();
            for (Map.Entry<String, Map<LocalDate, Status>> e : aggregated.entrySet()) {
                String testCaseId = e.getKey();
                Map<LocalDate, Status> byDate = e.getValue();
                List<DailyOutcome> t = byDate.entrySet()
                        .stream()
                        .sorted(Map.Entry.comparingByKey())
                        .map(x -> new DailyOutcome(x.getKey(), x.getValue()))
                        .collect(Collectors.toList());
                timelines.put(testCaseId, t);
            }

            List<TestCaseFinding> recurring = new ArrayList<>();
            List<TestCaseFinding> inconsistent = new ArrayList<>();

            for (Map.Entry<String, List<DailyOutcome>> e : timelines.entrySet()) {
                String testCaseId = e.getKey();
                List<DailyOutcome> timeline = e.getValue();

                Optional<RecurringFailureEvidence> recurringEv = detectRecurringFailure(timeline);
                List<InconsistentFailureEvidence> inconsistentEv = detectInconsistentPatterns(timeline);

                TestCaseFinding finding = new TestCaseFinding(testCaseId, timeline, recurringEv, inconsistentEv);

                if (finding.isRecurring()) {
                    recurring.add(finding);
                }
                if (finding.isInconsistent()) {
                    inconsistent.add(finding);
                }
            }

            // Sort recurring by longest consecutive failures desc
            recurring.sort(Comparator
                    .comparingInt((TestCaseFinding f) -> f.recurringFailure().map(RecurringFailureEvidence::consecutiveFailedDays).orElse(0))
                    .reversed()
                    .thenComparing(TestCaseFinding::testCaseId));

            // Sort inconsistent by testCaseId
            inconsistent.sort(Comparator.comparing(TestCaseFinding::testCaseId));

            return new AnalysisReport(recurring, inconsistent, timelines.size());
        }

        private Optional<RecurringFailureEvidence> detectRecurringFailure(List<DailyOutcome> timeline) {
            if (timeline == null || timeline.size() < 2) {
                return Optional.empty();
            }

            // Count trailing consecutive FAILs (ending at last observed date)
            int count = 0;
            List<DailyOutcome> failingRun = new LinkedList<>();
            for (int i = timeline.size() - 1; i >= 0; i--) {
                DailyOutcome d = timeline.get(i);
                if (d.status() == Status.FAIL) {
                    count++;
                    failingRun.add(0, d);
                } else {
                    break;
                }
            }

            if (count >= 2) {
                return Optional.of(new RecurringFailureEvidence(count, failingRun));
            }
            return Optional.empty();
        }

        private List<InconsistentFailureEvidence> detectInconsistentPatterns(List<DailyOutcome> timeline) {
            if (timeline == null || timeline.size() < 3) {
                return List.of();
            }

            List<InconsistentFailureEvidence> evidence = new ArrayList<>();

            // 1) Detect alternating patterns in any 4-day window.
            if (timeline.size() >= 4) {
                for (int i = 0; i <= timeline.size() - 4; i++) {
                    List<DailyOutcome> window = timeline.subList(i, i + 4);
                    String pattern = window.stream().map(x -> String.valueOf(x.status().shortCode())).collect(Collectors.joining());
                    if (pattern.equals("PFPF") || pattern.equals("FPFP")) {
                        evidence.add(new InconsistentFailureEvidence(
                                InconsistentType.ALTERNATING_PATTERN,
                                "Alternating pattern detected: " + pattern,
                                new ArrayList<>(window)
                        ));
                    }
                }
            }

            // 2) Detect failure for a period then return to passing state:
            // find any run of >=2 FAILs followed later by PASS.
            int runStart = -1;
            int runLength = 0;
            for (int i = 0; i < timeline.size(); i++) {
                Status s = timeline.get(i).status();
                if (s == Status.FAIL) {
                    if (runStart < 0) {
                        runStart = i;
                        runLength = 1;
                    } else {
                        runLength++;
                    }
                } else { // PASS
                    if (runStart >= 0 && runLength >= 2) {
                        int evStart = Math.max(0, runStart - 1);
                        int evEndExclusive = Math.min(timeline.size(), i + 1); // include the recovery PASS day
                        List<DailyOutcome> window = new ArrayList<>(timeline.subList(evStart, evEndExclusive));
                        evidence.add(new InconsistentFailureEvidence(
                                InconsistentType.FAILURE_THEN_RECOVERY,
                                "Failure run of " + runLength + " day(s) recovered to PASS",
                                window
                        ));
                    }
                    // reset run
                    runStart = -1;
                    runLength = 0;
                }
            }

            return evidence;
        }
    }

    static final class CsvExecutionRecordReader {

        static List<ExecutionRecord> read(Path path) throws IOException {
            if (path == null) {
                throw new InvalidInputException("CSV path is required.");
            }
            if (!Files.exists(path) || !Files.isRegularFile(path)) {
                throw new InvalidInputException("CSV path does not exist or is not a file: " + path);
            }

            try (InputStream in = Files.newInputStream(path)) {
                return read(in);
            }
        }

        static List<ExecutionRecord> read(InputStream in) throws IOException {
            Objects.requireNonNull(in, "in");
            List<ExecutionRecord> out = new ArrayList<>();

            try (BufferedReader br = new BufferedReader(new InputStreamReader(in, StandardCharsets.UTF_8))) {
                String header = br.readLine();
                if (header == null) {
                    throw new InvalidInputException("CSV is empty. Expected header: date,testCaseId,status");
                }

                String[] cols = splitCsvLine(header);
                if (cols.length < 3
                        || !"date".equalsIgnoreCase(cols[0].trim())
                        || !"testCaseId".equalsIgnoreCase(cols[1].trim())
                        || !"status".equalsIgnoreCase(cols[2].trim())) {
                    throw new InvalidInputException("Invalid CSV header. Expected: date,testCaseId,status");
                }

                String line;
                int lineNo = 1;
                while ((line = br.readLine()) != null) {
                    lineNo++;
                    if (line.trim().isEmpty()) {
                        continue;
                    }
                    String[] parts = splitCsvLine(line);
                    if (parts.length < 3) {
                        throw new InvalidInputException("Invalid CSV line " + lineNo + ": expected 3 columns but got " + parts.length);
                    }

                    LocalDate date = parseDate(parts[0], lineNo);
                    String testCaseId = requireNonBlank(parts[1], "testCaseId (line " + lineNo + ")");
                    Status status = Status.parse(parts[2]);

                    out.add(new ExecutionRecord(date, testCaseId, status));
                }
            }

            return out;
        }

        private static LocalDate parseDate(String raw, int lineNo) {
            if (raw == null || raw.trim().isEmpty()) {
                throw new InvalidInputException("Missing date at line " + lineNo);
            }
            try {
                return LocalDate.parse(raw.trim(), DATE_FMT);
            } catch (DateTimeParseException e) {
                throw new InvalidInputException("Invalid date at line " + lineNo + ": '" + raw + "' (expected yyyy-MM-dd)");
            }
        }

        // Minimal CSV splitting supporting quoted values with commas.
        private static String[] splitCsvLine(String line) {
            List<String> tokens = new ArrayList<>();
            StringBuilder sb = new StringBuilder();
            boolean inQuotes = false;
            for (int i = 0; i < line.length(); i++) {
                char c = line.charAt(i);
                if (c == '"') {
                    inQuotes = !inQuotes;
                    continue;
                }
                if (c == ',' && !inQuotes) {
                    tokens.add(sb.toString());
                    sb.setLength(0);
                } else {
                    sb.append(c);
                }
            }
            tokens.add(sb.toString());
            return tokens.toArray(new String[0]);
        }
    }

        private static String formatTimeline(List<DailyOutcome> timeline) {
            if (timeline == null || timeline.isEmpty()) {
                return "(no data)";
            }
            return timeline.stream()
                    .map(d -> d.date() + "=" + d.status().shortCode())
