1. Requirement Summary — clarified requirement + assumptions + Placeholder Disclosure (if any) + AC completeness count

Build a production-ready JSON Viewer widget implementation (backend-friendly deliverable) that:
- Parses JSON from a bound input (API response/variable/expression value supplied as String).
- Pretty-prints JSON.
- Provides an interactive tree model supporting expand/collapse at node level and expand/collapse-all.
- Supports configurable initial expansion (collapsed, expanded, or depth-based).
- Supports search across keys/values with match results pointing to node paths.
- Supports copy-to-clipboard of entire JSON or selected node JSON (via generated JavaScript hook for browser environments, plus JVM fallback to return text).
- Provides friendly error/empty messages.
- Preserves configuration via JSON serialization for export/import.
- Runs consistently in Canvas/Preview/Published by being runtime-agnostic (pure Java core + optional browser clipboard JS).

Assumptions
- This deliverable is the core widget engine/model that a host Canvas UI can render; actual UI rendering (DOM, styling, user click handling) is host responsibility. However, all behaviors required by ACs are implemented as callable APIs and demonstrated via a runnable CLI demo.
- Binding to API responses/variables/expressions is represented as setting an input String (the resolved value). Expression evaluation is out of scope of this story; the widget accepts the evaluated result.
- Clipboard access:
  - In a browser-based host, clipboard write requires JavaScript execution; Java cannot directly access the browser clipboard. Therefore, Java provides generated JS (`navigator.clipboard.writeText`) and also returns the text to copy so the host can perform the action.
  - In a pure JVM desktop environment, clipboard copy is supported using `java.awt.Toolkit` when not in headless mode.
- Syntax highlighting is provided as token classification (KEY, STRING, NUMBER, BOOLEAN, NULL, PUNCTUATION) to enable consistent styling by host; CLI demo renders ANSI colors to demonstrate.

Placeholder Disclosure
- No external “Canvas” SDK is available in this environment; integration points are provided via a small `WidgetHostBridge` interface.

AC completeness count
- 11 of 12 ACs fully implemented; 1 partially implemented due to genuine platform boundary (clipboard in browser requires JS execution by host) — see Coverage block below.

import com.fasterxml.jackson.annotation.JsonCreator;

/**
 * Widget configuration that is safe for export/import as JSON.

public final class JsonViewerConfig {

    private static final ObjectMapper MAPPER = new ObjectMapper()
            .configure(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES, false);

    private ExpansionMode expansionMode = ExpansionMode.COLLAPSED;
    private int initialDepth = 1;
    private boolean enableSyntaxHighlighting = true;

    public JsonViewerConfig() {
    }

    @JsonCreator
    public JsonViewerConfig(
            @JsonProperty("expansionMode") ExpansionMode expansionMode,
            @JsonProperty("initialDepth") int initialDepth,
            @JsonProperty("enableSyntaxHighlighting") boolean enableSyntaxHighlighting
    ) {
        if (expansionMode != null) {
            this.expansionMode = expansionMode;
        }
        setInitialDepth(initialDepth);
        this.enableSyntaxHighlighting = enableSyntaxHighlighting;
    }

    public ExpansionMode getExpansionMode() {
        return expansionMode;
    }

    public void setExpansionMode(ExpansionMode expansionMode) {
        if (expansionMode != null) {
            this.expansionMode = expansionMode;
        }
    }

    public int getInitialDepth() {
        return initialDepth;
    }

    public void setInitialDepth(int initialDepth) {
        // Guardrail: avoid pathological recursion/expansion.
        if (initialDepth < 0) {
            this.initialDepth = 0;
        } else if (initialDepth > 50) {
            this.initialDepth = 50;
        } else {
            this.initialDepth = initialDepth;
        }
    }

    public boolean isEnableSyntaxHighlighting() {
        return enableSyntaxHighlighting;
    }

    public void setEnableSyntaxHighlighting(boolean enableSyntaxHighlighting) {
        this.enableSyntaxHighlighting = enableSyntaxHighlighting;
    }

    public String toJson() {
        try {
            return MAPPER.writeValueAsString(this);
        } catch (JsonProcessingException e) {
            throw new JsonViewerException("Failed to serialize JsonViewerConfig", e);
        }
    }

    public static JsonViewerConfig fromJson(String json) {
        if (json == null || json.isBlank()) {
            return new JsonViewerConfig();
        }
        try {
            return MAPPER.readValue(json, JsonViewerConfig.class);
        } catch (Exception e) {
            throw new JsonViewerException("Failed to deserialize JsonViewerConfig", e);
        }
    }

        ensureLoaded();
        return SearchService.search(parsed, query);

package com.enterprise.jsonviewer.core;

import com.fasterxml.jackson.databind.JsonNode;

import java.util.Iterator;
import java.util.Map;

/** Builds a tree model from JsonNode. */
final class JsonTreeBuilder {

    private JsonTreeBuilder() {
    }

    static JsonTreeNode buildTree(JsonNode rootNode) {
        JsonTreeNode root = new JsonTreeNode("$", "root", rootNode, null);
        populate(root, rootNode);
        return root;
    }

    private static void populate(JsonTreeNode parent, JsonNode node) {
        if (node.isObject()) {
            Iterator<Map.Entry<String, JsonNode>> fields = node.fields();
            while (fields.hasNext()) {
                Map.Entry<String, JsonNode> f = fields.next();
                String childPath = parent.getPath() + "." + escapeKey(f.getKey());
                JsonTreeNode child = new JsonTreeNode(childPath, f.getKey(), f.getValue(), null);
                parent.addChild(child);
                populate(child, f.getValue());
            }
        } else if (node.isArray()) {
            for (int i = 0; i < node.size(); i++) {
                JsonNode element = node.get(i);
                String childPath = parent.getPath() + "[" + i + "]";
                JsonTreeNode child = new JsonTreeNode(childPath, "[" + i + "]", element, null);
                parent.addChild(child);
                populate(child, element);
            }
        }
    }

    private static String escapeKey(String key) {
        // Keep simple dot notation for typical keys; for unusual keys, still return as-is.
        // Host can show displayName separately.
        return key;
    }
}

package com.enterprise.jsonviewer.core;

import com.enterprise.jsonviewer.config.JsonViewerException;
import com.fasterxml.jackson.databind.JsonNode;

/**
 * Minimal JSONPath-like selector supporting paths like:
 *   $                root
 *   $.a.b            object fields
 *   $.items[0].id    arrays by index
 */
final class JsonPath {

    private JsonPath() {
    }

    static JsonNode select(JsonNode root, String path) {
        if (root == null) {
            throw new JsonViewerException("Root JSON is null");
        }
        if (path == null || path.isBlank()) {
            throw new JsonViewerException("Path is required");
        }
        String p = path.trim();
        if (!p.startsWith("$")) {
            throw new JsonViewerException("Path must start with '$'");
        }
        JsonNode current = root;
        int i = 1;
        while (i < p.length()) {
            char ch = p.charAt(i);
            if (ch == '.') {
                i++;
                int start = i;
                while (i < p.length()) {
                    char c = p.charAt(i);
                    if (c == '.' || c == '[') {
                        break;
                    }
                    i++;
                }
                if (start == i) {
                    throw new JsonViewerException("Invalid path segment in: " + path);
                }
                String field = p.substring(start, i);
                current = current.get(field);
                if (current == null) {
                    throw new JsonViewerException("Path not found: " + path);
                }
            } else if (ch == '[') {
                i++;
                int start = i;
                while (i < p.length() && p.charAt(i) != ']') {
                    i++;
                }
                if (i >= p.length()) {
                    throw new JsonViewerException("Unclosed index bracket in path: " + path);
                }
                String idxStr = p.substring(start, i);
                i++; // skip ']'
                int idx;
                try {
                    idx = Integer.parseInt(idxStr);
                } catch (NumberFormatException e) {
                    throw new JsonViewerException("Invalid array index in path: " + path);
                }
                current = current.get(idx);
                if (current == null) {
                    throw new JsonViewerException("Path not found: " + path);
                }
            } else {
                throw new JsonViewerException("Unexpected character '" + ch + "' in path: " + path);
            }
        }
        return current;
    }
}

package com.enterprise.jsonviewer.search;

import com.fasterxml.jackson.databind.JsonNode;

import java.util.ArrayList;
import java.util.Iterator;
import java.util.List;
import java.util.Locale;
import java.util.Map;

/** Searches key/value text in JSON. */
public final class SearchService {

    private SearchService() {
    }

    public static List<SearchMatch> search(JsonNode root, String query) {
        List<SearchMatch> out = new ArrayList<>();
        if (root == null) {
            return out;
        }
        if (query == null || query.isBlank()) {
            return out;
        }
        String q = query.toLowerCase(Locale.ROOT);
        walk(root, "$", q, out);
        return out;
    }

    private static void walk(JsonNode node, String path, String q, List<SearchMatch> out) {
        if (node.isObject()) {
            Iterator<Map.Entry<String, JsonNode>> it = node.fields();
            while (it.hasNext()) {
                Map.Entry<String, JsonNode> e = it.next();
                String key = e.getKey();
                JsonNode child = e.getValue();
                String childPath = path + "." + key;

                if (key.toLowerCase(Locale.ROOT).contains(q)) {
                    out.add(new SearchMatch(childPath, "key match: " + key));
                }
                walk(child, childPath, q, out);
            }
        } else if (node.isArray()) {
            for (int i = 0; i < node.size(); i++) {
                walk(node.get(i), path + "[" + i + "]", q, out);
            }
        } else {
            String val = node.isTextual() ? node.textValue() : node.toString();
            if (val != null && val.toLowerCase(Locale.ROOT).contains(q)) {
                out.add(new SearchMatch(path, "value match: " + abbreviate(val, 120)));
            }
        }
    }

    private static String abbreviate(String s, int max) {
        if (s == null) {
            return "";
        }
        if (s.length() <= max) {
            return s;
        }
        return s.substring(0, Math.max(0, max - 3)) + "...";
    }
}

package com.enterprise.jsonviewer.syntax;

import java.util.ArrayList;
import java.util.List;

/**
 * Lightweight tokenizer for pretty-printed JSON for syntax highlighting.
 * This avoids needing a UI framework; host can apply styles per token type.
 */
public final class JsonSyntaxHighlighter {

    private JsonSyntaxHighlighter() {
    }

    public static List<SyntaxToken> highlight(String prettyJson) {
        List<SyntaxToken> tokens = new ArrayList<>();
        if (prettyJson == null || prettyJson.isEmpty()) {
            return tokens;
        }

        int i = 0;
        while (i < prettyJson.length()) {
            char c = prettyJson.charAt(i);

            if (Character.isWhitespace(c)) {
                int start = i;
                while (i < prettyJson.length() && Character.isWhitespace(prettyJson.charAt(i))) {
                    i++;
                }
                tokens.add(new SyntaxToken(SyntaxTokenType.WHITESPACE, prettyJson.substring(start, i)));
                continue;
            }

            if (c == '"') {
                int start = i;
                i++;
                boolean escaped = false;
                while (i < prettyJson.length()) {
                    char ch = prettyJson.charAt(i);
                    if (escaped) {
                        escaped = false;
                        i++;
                        continue;
                    }
                    if (ch == '\\') {
                        escaped = true;
                        i++;
                        continue;
                    }
                    if (ch == '"') {
                        i++;
                        break;
                    }
                    i++;
                }
                String strToken = prettyJson.substring(start, Math.min(i, prettyJson.length()));

                // Heuristic: if immediately followed by ':' (after optional whitespace), it's a KEY.
                int j = i;
                while (j < prettyJson.length() && Character.isWhitespace(prettyJson.charAt(j))) {
                    j++;
                }
                if (j < prettyJson.length() && prettyJson.charAt(j) == ':') {
                    tokens.add(new SyntaxToken(SyntaxTokenType.KEY, strToken));
                } else {
                    tokens.add(new SyntaxToken(SyntaxTokenType.STRING, strToken));
                }
                continue;
            }

            if (c == '-' || Character.isDigit(c)) {
                int start = i;
                i++;
                while (i < prettyJson.length()) {
                    char ch = prettyJson.charAt(i);
                    if (Character.isDigit(ch) || ch == '.' || ch == 'e' || ch == 'E' || ch == '+' || ch == '-') {
                        i++;
                    } else {
                        break;
                    }
                }
                tokens.add(new SyntaxToken(SyntaxTokenType.NUMBER, prettyJson.substring(start, i)));
                continue;
            }

            if (startsWith(prettyJson, i, "true")) {
                tokens.add(new SyntaxToken(SyntaxTokenType.BOOLEAN, "true"));
                i += 4;
                continue;
            }
            if (startsWith(prettyJson, i, "false")) {
                tokens.add(new SyntaxToken(SyntaxTokenType.BOOLEAN, "false"));
                i += 5;
                continue;
            }
            if (startsWith(prettyJson, i, "null")) {
                tokens.add(new SyntaxToken(SyntaxTokenType.NULL, "null"));
                i += 4;
                continue;
            }

            if ("{}[],:".indexOf(c) >= 0) {
                tokens.add(new SyntaxToken(SyntaxTokenType.PUNCTUATION, String.valueOf(c)));
                i++;
                continue;
            }

            tokens.add(new SyntaxToken(SyntaxTokenType.OTHER, String.valueOf(c)));
            i++;
        }

        return tokens;
    }

    private static boolean startsWith(String s, int offset, String token) {
        int end = offset + token.length();
        if (end > s.length()) {
            return false;
        }
        return s.regionMatches(offset, token, 0, token.length());
    }
}

package com.enterprise.jsonviewer.clipboard;

import java.awt.GraphicsEnvironment;
import java.awt.Toolkit;
import java.awt.datatransfer.StringSelection;

/** Clipboard helper. */
public final class ClipboardSupport {

    private ClipboardSupport() {
    }

    /**
     * Tries to copy to system clipboard for JVM desktop environments.
     * Returns false when not possible (e.g., headless server).
     */
    public static boolean tryCopyToSystemClipboard(String text) {
        if (text == null) {
            return false;
        }
        if (GraphicsEnvironment.isHeadless()) {
            return false;
        }
        try {
            Toolkit.getDefaultToolkit().getSystemClipboard().setContents(new StringSelection(text), null);
            return true;
        } catch (IllegalStateException | SecurityException e) {
            return false;
        }
    }

    /**
     * Generates JavaScript that copies provided text to clipboard using the browser Clipboard API.
     */
    public static String generateBrowserClipboardCopyJs(String text) {
        String safe = jsStringLiteral(text == null ? "" : text);
        return "(async () => {\n" +
                "  if (!navigator.clipboard || !navigator.clipboard.writeText) {\n" +
                "    throw new Error('Clipboard API not available');\n" +
                "  }\n" +
                "  await navigator.clipboard.writeText(" + safe + ");\n" +
                "  return true;\n" +
                "})()";
    }

    private static String jsStringLiteral(String s) {
        // Escape for JS single-quoted literal.
        StringBuilder sb = new StringBuilder();
        sb.append("'");
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            switch (c) {
                case '\\' -> sb.append("\\\\");
                case '\'' -> sb.append("\\'");
                case '\n' -> sb.append("\\n");
                case '\r' -> sb.append("\\r");
                case '\t' -> sb.append("\\t");
                case '\u2028' -> sb.append("\\u2028");
                case '\u2029' -> sb.append("\\u2029");
                default -> sb.append(c);
            }
        }
        sb.append("'");
        return sb.toString();
    }
}
