1. Requirement Summary
- Build a JSON Viewer widget backend component that renders arbitrary JSON (from API responses/variables/expressions) into a pretty-printed, syntax-highlighted, searchable, expandable/collapsible tree representation.
- Support expand/collapse per node and expand/collapse all.
- Support copy-to-clipboard of the whole JSON or a selected sub-tree.
- Provide friendly messages for empty input or invalid JSON.
- Support configurable initial expansion behavior: fully collapsed, fully expanded, or expanded up to a max depth.
- Preserve widget configuration via a serializable configuration object suitable for export/import.
- Assumptions:
  - This deliverable provides a production-ready Java library (core engine) for a JSON Viewer widget; UI integration (Canvas/Preview/Published) is assumed to call this library to render HTML/CSS and wire client-side clipboard actions.
  - Copy-to-clipboard is provided as generated JavaScript snippets (safe, minimal) to be executed by the hosting UI; server-side Java cannot access a user clipboard directly.
  - Search returns JSON-Pointer-like paths (RFC 6901 style) to matching nodes; host UI can use these paths to focus/highlight nodes.
  - JSON input may be a JSON object/array/value; when input is a primitive, it is displayed as a single node.

package com.aava.datastudio.jsonviewer.internal;

/**
 * Minimal HTML escaper to prevent XSS when rendering JSON content.
 */
public final class HtmlEscaper {

    private HtmlEscaper() {
    }

    public static String escape(String input) {
        if (input == null) {
            return "";
        }
        StringBuilder sb = new StringBuilder(input.length() + 16);
        for (int i = 0; i < input.length(); i++) {
            char c = input.charAt(i);
            switch (c) {
                case '&' -> sb.append("&amp;");
                case '<' -> sb.append("&lt;");
                case '>' -> sb.append("&gt;");
                case '"' -> sb.append("&quot;");
                case '\'' -> sb.append("&#x27;");
                default -> sb.append(c);
            }
        }
        return sb.toString();
    }

    public static String escapeJsonPointerToken(String token) {
        // For rendering tokens in HTML context, not for pointer semantics.
        return escape(token);
    }
}

    }

    public static String append(String basePointer, String token) {
        if (basePointer == null || basePointer.isBlank()) {
            basePointer = "";
        }
        if (token == null) {
            token = "null";
        }
        return basePointer + "/" + escapeToken(token);

        return token.replace("~", "~0").replace("/", "~1");

package com.aava.datastudio.jsonviewer;

import com.aava.datastudio.jsonviewer.internal.HtmlEscaper;
import com.aava.datastudio.jsonviewer.internal.JsonPointerUtil;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.SerializationFeature;

import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.Base64;
import java.util.Iterator;
import java.util.List;
import java.util.Map;
import java.util.Objects;

/**
 * Renders JSON into a pretty-printed, syntax-highlighted, expandable HTML tree.
 * <p>
 * Security: All rendered values are HTML-escaped to prevent XSS.
 */
public final class JsonViewerRenderer {

    private final ObjectMapper mapper;

    public JsonViewerRenderer() {
        this.mapper = new ObjectMapper().enable(SerializationFeature.INDENT_OUTPUT);
    }

    /**
     * Validates input as JSON.
     */
    public JsonValidationResult validate(String json) {
        if (json == null || json.isBlank()) {
            return JsonValidationResult.invalid("No JSON to display.");
        }
        try {
            mapper.readTree(json);
            return JsonValidationResult.ok();
        } catch (JsonProcessingException e) {
            return JsonValidationResult.invalid("Invalid JSON: " + safeMessage(e.getOriginalMessage()));
        }
    }

    /**
     * Render the viewer model.
     *
     * @param json   JSON string (object/array/primitive). Must not be null/blank.
     * @param config widget config
     * @return model including HTML/CSS and pretty JSON
     */
    public JsonViewerModel render(String json, JsonViewerConfig config) {
        Objects.requireNonNull(config, "config");

        if (json == null || json.isBlank()) {
            return emptyModel("No JSON to display.", config);
        }

        final JsonNode root;
        try {
            root = mapper.readTree(json);
        } catch (JsonProcessingException e) {
            return emptyModel("Invalid JSON.", config);
        }

        final String pretty;
        try {
            pretty = mapper.writeValueAsString(root);
        } catch (JsonProcessingException e) {
            throw new JsonViewerException("Failed to pretty print JSON", e);
        }

        List<SearchMatch> searchIndex = new ArrayList<>();
        StringBuilder html = new StringBuilder(8_192);
        html.append("<div class=\"jv-root\">\n");
        html.append(renderNode(root, "", 0, config, searchIndex));
        html.append("</div>\n");

        String css = defaultCss();
        String copyAllJs = config.isEnableCopy() ? buildCopyAllJs(pretty) : "";

        return new JsonViewerModel(pretty, html.toString(), css, copyAllJs, searchIndex);
    }

    private JsonViewerModel emptyModel(String message, JsonViewerConfig config) {
        String css = defaultCss();
        String safe = HtmlEscaper.escape(message);
        String html = "<div class=\"jv-root\"><div class=\"jv-empty\">" + safe + "</div></div>";
        String copyAllJs = config.isEnableCopy() ? buildCopyAllJs("") : "";
        return new JsonViewerModel("", html, css, copyAllJs, List.of());
    }

    private String renderNode(JsonNode node,
                              String pointer,
                              int depth,
                              JsonViewerConfig config,
                              List<SearchMatch> searchIndex) {

        boolean expanded = isInitiallyExpanded(depth, config);
        StringBuilder sb = new StringBuilder(1_024);

        if (node.isObject()) {
            sb.append(containerStart("object", pointer, expanded));
            Iterator<Map.Entry<String, JsonNode>> fields = node.fields();
            int count = 0;
            while (fields.hasNext()) {
                Map.Entry<String, JsonNode> entry = fields.next();
                String key = entry.getKey();
                JsonNode child = entry.getValue();

                String childPointer = JsonPointerUtil.append(pointer, key);
                sb.append("<div class=\"jv-row\">\n");
                sb.append("<span class=\"jv-key\">\"")
                        .append(HtmlEscaper.escape(key))
                        .append("\"</span>");
                sb.append("<span class=\"jv-colon\">: </span>");
                sb.append(renderNode(child, childPointer, depth + 1, config, searchIndex));
                sb.append("</div>\n");
                count++;
            }
            sb.append(containerEnd(count, "{", "}"));
        } else if (node.isArray()) {
            sb.append(containerStart("array", pointer, expanded));
            int idx = 0;
            for (JsonNode child : node) {
                String childPointer = JsonPointerUtil.append(pointer, String.valueOf(idx));
                sb.append("<div class=\"jv-row\">\n");
                sb.append("<span class=\"jv-index\">[").append(idx).append("]</span>");
                sb.append("<span class=\"jv-colon\"> </span>");
                sb.append(renderNode(child, childPointer, depth + 1, config, searchIndex));
                sb.append("</div>\n");
                idx++;
            }
            sb.append(containerEnd(idx, "[", "]"));
        } else {
            // leaf value
            String rendered = renderPrimitive(node);
            // index for search
            String preview = node.isTextual() ? node.asText() : node.toString();
            preview = preview.length() > 80 ? preview.substring(0, 77) + "..." : preview;
            searchIndex.add(new SearchMatch(pointer.isEmpty() ? "/" : pointer, preview));

            sb.append("<span class=\"jv-value\" data-pointer=\"")
                    .append(HtmlEscaper.escape(pointer.isEmpty() ? "/" : pointer))
                    .append("\">")
                    .append(rendered)
                    .append("</span>");
        }
        return sb.toString();
    }

    private String containerStart(String type, String pointer, boolean expanded) {
        String safePointer = HtmlEscaper.escape(pointer.isEmpty() ? "/" : pointer);
        String open = "<details class=\"jv-node jv-" + type + "\" data-pointer=\"" + safePointer + "\"" + (expanded ? " open" : "") + ">\n";
        open += "<summary class=\"jv-summary\">";
        open += "<span class=\"jv-bracket\"></span>";
        open += "<span class=\"jv-spacer\"></span>";
        open += "</summary>\n";
        open += "<div class=\"jv-children\">\n";
        return open;
    }

    private String containerEnd(int count, String openBracket, String closeBracket) {
        return "</div>\n" +
                "<div class=\"jv-meta\"><span class=\"jv-bracket\">" + openBracket + "</span>" +
                "<span class=\"jv-count\">" + count + "</span>" +
                "<span class=\"jv-bracket\">" + closeBracket + "</span></div>\n" +
                "</details>\n";
    }

    private boolean isInitiallyExpanded(int depth, JsonViewerConfig config) {
        return switch (config.getInitialExpansionMode()) {
            case COLLAPSED -> false;
            case EXPANDED -> true;
            case DEPTH_BASED -> depth < config.getInitialExpansionDepth();
        };
    }

    private String renderPrimitive(JsonNode node) {
        if (node.isTextual()) {
            return "<span class=\"jv-string\">\"" + HtmlEscaper.escape(node.asText()) + "\"</span>";
        }
        if (node.isNumber()) {
            return "<span class=\"jv-number\">" + HtmlEscaper.escape(node.numberValue().toString()) + "</span>";
        }
        if (node.isBoolean()) {
            return "<span class=\"jv-boolean\">" + node.asBoolean() + "</span>";
        }
        if (node.isNull()) {
            return "<span class=\"jv-null\">null</span>";
        }
        // fallback for other scalar types
        return "<span class=\"jv-other\">" + HtmlEscaper.escape(node.toString()) + "</span>";
    }

    private String defaultCss() {
        return "/* JSON Viewer Widget styles */\n" +
                ".jv-root{font-family:ui-monospace,SFMono-Regular,Menlo,Monaco,Consolas,\"Liberation Mono\",\"Courier New\",monospace;font-size:12px;line-height:1.4;color:#111;}\n" +
                ".jv-empty{padding:10px;color:#555;background:#fafafa;border:1px solid #eee;border-radius:6px;}\n" +
                ".jv-node{margin-left:14px;}\n" +
                ".jv-summary{cursor:pointer;user-select:none;list-style:none;}\n" +
                ".jv-children{padding-left:10px;border-left:1px dashed #ddd;margin:4px 0 6px 0;}\n" +
                ".jv-row{margin:2px 0;}\n" +
                ".jv-key{color:#922;}\n" +
                ".jv-index{color:#555;}\n" +
                ".jv-colon{color:#333;}\n" +
                ".jv-string{color:#0b7285;}\n" +
                ".jv-number{color:#1c7ed6;}\n" +
                ".jv-boolean{color:#d6336c;}\n" +
                ".jv-null{color:#868e96;}\n" +
                ".jv-other{color:#343a40;}\n" +
                ".jv-meta{color:#777;margin-left:22px;}\n" +
                ".jv-count{padding:0 6px;}\n";
    }

    private String buildCopyAllJs(String prettyJson) {
        // Host UI should execute this in browser to copy.
        String base64 = Base64.getEncoder().encodeToString(prettyJson.getBytes(StandardCharsets.UTF_8));
        return "(async function(){const text=atob('" + base64 + "');" +
                "if(navigator.clipboard&&navigator.clipboard.writeText){await navigator.clipboard.writeText(text);}" +
                "else{const ta=document.createElement('textarea');ta.value=text;document.body.appendChild(ta);ta.select();document.execCommand('copy');ta.remove();}})();";
    }

    private String safeMessage(String msg) {
        if (msg == null) {
            return "";
        }
        String trimmed = msg.trim();
        return trimmed.length() > 200 ? trimmed.substring(0, 200) : trimmed;
    }
}

package com.aava.datastudio.jsonviewer;

import com.aava.datastudio.jsonviewer.internal.JsonPointerUtil;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;

import java.util.ArrayList;
import java.util.Iterator;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.Objects;

/**
 * Search service for JSON content.
 * Returns matches as JSON Pointers to nodes whose key or value contains the query.
 */
public final class JsonSearchService {

    private final ObjectMapper mapper;

    public JsonSearchService() {
        this.mapper = new ObjectMapper();
    }

    /**
     * Search within JSON.
     *
     * @param json  json string
     * @param query query string (case-insensitive). If blank, returns empty list.
     */
    public List<SearchMatch> search(String json, String query) {
        if (json == null || json.isBlank()) {
            return List.of();
        }
        if (query == null || query.isBlank()) {
            return List.of();
        }

        final JsonNode root;
        try {
            root = mapper.readTree(json);
        } catch (JsonProcessingException e) {
            return List.of();
        }

        String q = query.toLowerCase(Locale.ROOT);
        List<SearchMatch> matches = new ArrayList<>();
        traverse(root, "", q, matches);
        return matches;
    }

    private void traverse(JsonNode node, String pointer, String q, List<SearchMatch> out) {
        Objects.requireNonNull(node, "node");
        Objects.requireNonNull(q, "q");
        Objects.requireNonNull(out, "out");

        if (node.isObject()) {
            Iterator<Map.Entry<String, JsonNode>> it = node.fields();
            while (it.hasNext()) {
                Map.Entry<String, JsonNode> e = it.next();
                String key = e.getKey();
                JsonNode child = e.getValue();
                String childPtr = JsonPointerUtil.append(pointer, key);

                if (key != null && key.toLowerCase(Locale.ROOT).contains(q)) {
                    out.add(new SearchMatch(childPtr, "key: " + key));
                }
                traverse(child, childPtr, q, out);
            }
        } else if (node.isArray()) {
            int idx = 0;
            for (JsonNode child : node) {
                String childPtr = JsonPointerUtil.append(pointer, String.valueOf(idx));
                traverse(child, childPtr, q, out);
                idx++;
            }
        } else {
            String valuePreview = node.isTextual() ? node.asText() : node.toString();
            if (valuePreview != null && valuePreview.toLowerCase(Locale.ROOT).contains(q)) {
                out.add(new SearchMatch(pointer.isEmpty() ? "/" : pointer, valuePreview));
            }
        }
    }
}

3. Code Documentation
- JsonViewerRenderer
  - validate(json): returns JsonValidationResult with friendly error message for blank/invalid JSON.
  - render(json, config): parses JSON using Jackson, pretty-prints it, renders an HTML <details>/<summary> tree with expand/collapse support, and returns CSS + copy-to-clipboard JS snippet.
  - All JSON-derived content is HTML-escaped (HtmlEscaper) to prevent XSS.
- JsonSearchService
  - search(json, query): case-insensitive search across keys and scalar values, returning JSON Pointer paths to matching nodes.
- JsonViewerConfig
  - Serializable config object to preserve widget settings for export/import; supports initial expansion modes.
- JsonViewerCliApp
  - Runnable main to render a JSON file locally and print outputs.
