# Requirement Summary
Connect an existing JSON-based app project to a VS Code plugin so that natural-language (NLP) user requirements can be interpreted into concrete JSON/code changes, applied safely to the project, validated, and then packaged for upload to and execution within DataPowerAI.

## Assumptions
- The "JSON app" is represented as a directory containing one or more `.json` files (and optional supporting assets) and the enhancement workflow is change-by-change.
- The VS Code plugin communicates with a local HTTP service running on the developer machine.
- The NLP step is implemented as deterministic, rules-based intent parsing (enterprise-safe default) rather than calling an external LLM service.
- DataPowerAI upload is represented as generating a deployable ZIP artifact plus a manifest; actual platform authentication/upload APIs are not specified, so the code produces a ready-to-upload bundle.
- Java version: 17. Build tool: Maven.

# Generated Java Code
Below is a complete, runnable Spring Boot service that the VS Code plugin can call.
It:
1) Accepts a workspace path + natural language requirement.
2) Parses requirement into a structured change plan.
3) Applies changes to targeted JSON files (deep merge + add/update fields) with JSON-schema-like validation.
4) Runs validation (JSON parse + required-field checks + optional semantic checks).
5) Produces a deployable ZIP bundle with a manifest for DataPowerAI.

```java
 * Centralized exception handling for safe error responses.

@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(EnhancerException.class)
    public ResponseEntity<Map<String, Object>> handleEnhancer(EnhancerException ex) {
        Map<String, Object> body = new HashMap<>();
        body.put("error", ex.getMessage());
        body.put("code", ex.getCode());
        return ResponseEntity.status(ex.getHttpStatus()).body(body);
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, Object>> handleInvalid(MethodArgumentNotValidException ex) {
        Map<String, Object> body = new HashMap<>();
        body.put("error", "Validation failed");
        body.put("details", ex.getBindingResult().getFieldErrors().stream()
                .map(fe -> fe.getField() + ": " + fe.getDefaultMessage())
                .toList());
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(ConstraintViolationException.class)
    public ResponseEntity<Map<String, Object>> handleConstraint(ConstraintViolationException ex) {
        Map<String, Object> body = new HashMap<>();
        body.put("error", "Validation failed");
        body.put("details", ex.getConstraintViolations().stream()
                .map(v -> v.getPropertyPath() + ": " + v.getMessage())
                .toList());
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<Map<String, Object>> handleGeneric(Exception ex) {
        Map<String, Object> body = new HashMap<>();
        body.put("error", "Internal server error");
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(body);
    }
}
```

```java
package com.ascendion.dpai.enhancer.core;

import com.ascendion.dpai.enhancer.core.nlp.RequirementInterpreter;
import com.ascendion.dpai.enhancer.core.plan.ChangePlan;
import com.ascendion.dpai.enhancer.core.plan.ChangeStep;
import com.ascendion.dpai.enhancer.core.validation.ProjectValidator;
import com.ascendion.dpai.enhancer.core.zip.BundlePackager;
import org.springframework.stereotype.Service;

import java.nio.file.Path;
import java.time.Instant;
import java.util.ArrayList;
import java.util.List;

/**
 * Orchestrates the enhancement workflow.
 */
@Service
public class EnhancerService {

    private final RequirementInterpreter interpreter;
    private final JsonChangeApplier applier;
    private final ProjectValidator validator;
    private final BundlePackager packager;

    public EnhancerService(RequirementInterpreter interpreter,
                           JsonChangeApplier applier,
                           ProjectValidator validator,
                           BundlePackager packager) {
        this.interpreter = interpreter;
        this.applier = applier;
        this.validator = validator;
        this.packager = packager;
    }

    /**
     * Enhances a JSON app workspace based on a natural-language requirement.
     */
    public EnhancementResult enhance(String workspacePath, String requirementText) {
        Path workspace = SafePaths.requireSafeWorkspace(workspacePath);
        String normalized = RequirementInterpreter.normalize(requirementText);

        ChangePlan plan = interpreter.interpret(normalized);
        if (plan.steps().isEmpty()) {
            throw EnhancerException.badRequest("NO_CHANGES", "No actionable changes could be inferred from requirement.");
        }

        List<String> applied = new ArrayList<>();
        for (ChangeStep step : plan.steps()) {
            applied.add(applier.applyStep(workspace, step));
        }

        List<String> validations = validator.validate(workspace);
        boolean success = validations.stream().noneMatch(m -> m.startsWith("ERROR:"));

        String bundle = packager.createBundle(workspace, "dpai-bundle-" + Instant.now().toEpochMilli());

        return new EnhancementResult(
                workspace.toAbsolutePath().normalize().toString(),
                normalized,
                applied,
                validations,
                success,
                bundle
        );
    }
}
```

```java
package com.ascendion.dpai.enhancer.core;

import java.io.IOException;
import java.nio.file.*;

/**
 * Provides safe path handling to reduce path traversal and unintended writes.
 */
public final class SafePaths {

    private SafePaths() {
    }

    /**
     * Validates workspace path and ensures it's a directory.
     */
    public static Path requireSafeWorkspace(String workspacePath) {
        if (workspacePath == null || workspacePath.isBlank()) {
            throw EnhancerException.badRequest("INVALID_WORKSPACE", "workspacePath must be provided");
        }

        Path workspace = Paths.get(workspacePath).toAbsolutePath().normalize();
        if (!Files.exists(workspace)) {
            throw EnhancerException.notFound("WORKSPACE_NOT_FOUND", "Workspace path does not exist: " + workspace);
        }
        if (!Files.isDirectory(workspace)) {
            throw EnhancerException.badRequest("WORKSPACE_NOT_DIR", "Workspace path is not a directory: " + workspace);
        }

        // Ensure workspace is readable/writable.
        try {
            if (!Files.isReadable(workspace) || !Files.isWritable(workspace)) {
                throw EnhancerException.forbidden("WORKSPACE_NO_ACCESS", "Workspace must be readable and writable: " + workspace);
            }
        } catch (SecurityException se) {
            throw EnhancerException.forbidden("WORKSPACE_SECURITY", "No access to workspace: " + workspace);
        }

        return workspace;
    }

    /**
     * Resolves a relative path under the workspace safely.
     */
    public static Path resolveUnderWorkspace(Path workspace, String relativePath) {
        if (relativePath == null || relativePath.isBlank()) {
            throw EnhancerException.badRequest("INVALID_PATH", "Target path must be provided");
        }
        Path resolved = workspace.resolve(relativePath).normalize();
        if (!resolved.startsWith(workspace)) {
            throw EnhancerException.forbidden("PATH_TRAVERSAL", "Refusing to access path outside workspace");
        }
        return resolved;
    }

    public static void ensureParentDirs(Path file) {
        Path parent = file.getParent();
        if (parent == null) {
            return;
        }
        try {
            Files.createDirectories(parent);
        } catch (IOException e) {
            throw EnhancerException.serverError("IO_ERROR", "Failed to create directories: " + parent);
        }
    }
}
```

## src/main/java/com/ascendion/dpai/enhancer/core/JsonChangeApplier.java
```java
package com.ascendion.dpai.enhancer.core;

import com.ascendion.dpai.enhancer.core.plan.ChangeStep;
import com.ascendion.dpai.enhancer.core.plan.Operation;
import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;

/**
 * Applies JSON changes to files (deep merge of object nodes).
 */
@Component
public class JsonChangeApplier {

    private final ObjectMapper mapper = new ObjectMapper();

    public String applyStep(Path workspace, ChangeStep step) {
        if (step == null) {
            throw EnhancerException.badRequest("INVALID_STEP", "ChangeStep is null");
        }

        Path target = SafePaths.resolveUnderWorkspace(workspace, step.targetFile());
        if (step.operation() == Operation.ADD_OR_UPDATE_JSON_FIELDS) {
            applyJsonMerge(target, step.jsonPayload());
            return "Updated JSON fields in " + step.targetFile();
        }

        throw EnhancerException.badRequest("UNSUPPORTED_OPERATION", "Unsupported operation: " + step.operation());
    }

    private void applyJsonMerge(Path targetFile, String jsonPayload) {
        if (jsonPayload == null || jsonPayload.isBlank()) {
            throw EnhancerException.badRequest("EMPTY_PAYLOAD", "jsonPayload must not be empty");
        }

        JsonNode patch;
        try {
            patch = mapper.readTree(jsonPayload);
        } catch (JsonProcessingException e) {
            throw EnhancerException.badRequest("INVALID_JSON_PAYLOAD", "jsonPayload is not valid JSON");
        }

        if (!patch.isObject()) {
            throw EnhancerException.badRequest("PAYLOAD_NOT_OBJECT", "jsonPayload must be a JSON object");
        }

        ObjectNode base;
        if (Files.exists(targetFile)) {
            try {
                JsonNode existing = mapper.readTree(Files.readString(targetFile, StandardCharsets.UTF_8));
                if (existing == null || existing.isNull()) {
                    base = mapper.createObjectNode();
                } else if (!existing.isObject()) {
                    throw EnhancerException.badRequest("TARGET_NOT_OBJECT", "Target JSON must be an object: " + targetFile);
                } else {
                    base = (ObjectNode) existing;
                }
            } catch (IOException e) {
                throw EnhancerException.serverError("READ_FAILED", "Failed reading: " + targetFile);
            }
        } else {
            base = mapper.createObjectNode();
        }

        ObjectNode merged = deepMerge(base, (ObjectNode) patch);

        SafePaths.ensureParentDirs(targetFile);
        try {
            String pretty = mapper.writerWithDefaultPrettyPrinter().writeValueAsString(merged);
            Files.writeString(targetFile, pretty + System.lineSeparator(), StandardCharsets.UTF_8);
        } catch (IOException e) {
            throw EnhancerException.serverError("WRITE_FAILED", "Failed writing: " + targetFile);
        }
    }

    private ObjectNode deepMerge(ObjectNode base, ObjectNode patch) {
        patch.fields().forEachRemaining(entry -> {
            String key = entry.getKey();
            JsonNode patchValue = entry.getValue();

            JsonNode baseValue = base.get(key);
            if (baseValue != null && baseValue.isObject() && patchValue.isObject()) {
                deepMerge((ObjectNode) baseValue, (ObjectNode) patchValue);
            } else {
                base.set(key, patchValue);
            }
        });
        return base;
    }
}
```

```java
package com.ascendion.dpai.enhancer.core.nlp;

import com.ascendion.dpai.enhancer.core.plan.ChangePlan;
import com.ascendion.dpai.enhancer.core.plan.ChangeStep;
import com.ascendion.dpai.enhancer.core.plan.Operation;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Enterprise-safe default NLP: deterministic rule-based interpreter.
 *
 * Supported patterns (examples):
 * - "update app.json set ui.theme=dark"
 * - "set app.json ui.title=My App"
 * - "add field to config/settings.json logging.level=INFO"
 *
 * Values are treated as strings unless they look like JSON literals (true/false/null/number/object/array).
 */
@Component
public class RuleBasedRequirementInterpreter implements RequirementInterpreter {

    private static final Pattern COMMAND = Pattern.compile(
            "(?i)^(?:update|set|add)(?:\\s+field\\s+to)?\\s+(?<file>[^\\s]+)\\s+(?:set\\s+)?(?<path>[A-Za-z0-9_.-]+)\\s*=\\s*(?<value>.+)$");

    @Override
    public ChangePlan interpret(String normalizedRequirement) {
        List<ChangeStep> steps = new ArrayList<>();

        Matcher m = COMMAND.matcher(normalizedRequirement);
        if (m.matches()) {
            String file = m.group("file");
            String path = m.group("path");
            String valueRaw = m.group("value").trim();

            String jsonPatch = JsonPatchBuilder.buildObjectPatch(path, valueRaw);
            steps.add(new ChangeStep(Operation.ADD_OR_UPDATE_JSON_FIELDS, file, jsonPatch));
        }

        return new ChangePlan(steps);
    }
}
```

## src/main/java/com/ascendion/dpai/enhancer/core/nlp/JsonPatchBuilder.java
```java
package com.ascendion.dpai.enhancer.core.nlp;

import com.ascendion.dpai.enhancer.core.EnhancerException;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;

import java.math.BigDecimal;

/**
 * Builds a JSON object patch from dot-notated paths.
 */
public final class JsonPatchBuilder {

    private static final ObjectMapper MAPPER = new ObjectMapper();

    private JsonPatchBuilder() {
    }

    public static String buildObjectPatch(String dotPath, String rawValue) {
        if (dotPath == null || dotPath.isBlank()) {
            throw EnhancerException.badRequest("INVALID_DOT_PATH", "Path must be provided");
        }
        if (rawValue == null) {
            rawValue = "null";
        }

        String[] parts = dotPath.split("\\.");
        ObjectNode root = MAPPER.createObjectNode();
        ObjectNode current = root;

        for (int i = 0; i < parts.length; i++) {
            String p = parts[i];
            if (p.isBlank()) {
                throw EnhancerException.badRequest("INVALID_DOT_PATH", "Dot path contains empty segment");
            }

            if (i == parts.length - 1) {
                current.set(p, parseValue(rawValue));
            } else {
                ObjectNode child = MAPPER.createObjectNode();
                current.set(p, child);
                current = child;
            }
        }

        try {
            return MAPPER.writeValueAsString(root);
        } catch (Exception e) {
            throw EnhancerException.serverError("PATCH_BUILD_FAILED", "Failed to build JSON patch");
        }
    }

    private static JsonNode parseValue(String rawValue) {
        String v = rawValue.trim();

        // If it's a JSON object/array literal, parse it.
        if ((v.startsWith("{") && v.endsWith("}")) || (v.startsWith("[") && v.endsWith("]"))) {
            try {
                return MAPPER.readTree(v);
            } catch (Exception e) {
                throw EnhancerException.badRequest("INVALID_JSON_LITERAL", "Value is not valid JSON: " + rawValue);
            }
        }

        if ("true".equalsIgnoreCase(v)) return MAPPER.getNodeFactory().booleanNode(true);
        if ("false".equalsIgnoreCase(v)) return MAPPER.getNodeFactory().booleanNode(false);
        if ("null".equalsIgnoreCase(v)) return MAPPER.getNodeFactory().nullNode();

        // Number?
        try {
            BigDecimal bd = new BigDecimal(v);
            return MAPPER.getNodeFactory().numberNode(bd);
        } catch (NumberFormatException ignored) {
            // string
        }

        // Strip surrounding quotes if provided.
        if ((v.startsWith("\"") && v.endsWith("\"")) || (v.startsWith("'") && v.endsWith("'"))) {
            v = v.substring(1, v.length() - 1);
        }

        return MAPPER.getNodeFactory().textNode(v);
    }
}
```

```java
 * - If an app.json exists, ensures it has minimal expected keys
 */
@Component
public class DefaultProjectValidator implements ProjectValidator {

    private final ObjectMapper mapper = new ObjectMapper();

    @Override
    public List<String> validate(Path workspace) {
        if (workspace == null) {
            throw EnhancerException.badRequest("INVALID_WORKSPACE", "workspace is null");
        }

        List<String> messages = new ArrayList<>();

        // Validate all JSON files parse.
        try {
            Files.walk(workspace)
                    .filter(p -> Files.isRegularFile(p) && p.getFileName().toString().toLowerCase().endsWith(".json"))
                    .forEach(p -> {
                        try {
                            mapper.readTree(Files.readString(p, StandardCharsets.UTF_8));
                            messages.add("OK: JSON parse " + workspace.relativize(p));
                        } catch (IOException e) {
                            messages.add("ERROR: Invalid JSON " + workspace.relativize(p) + " (" + e.getMessage() + ")");
                        }
                    });
        } catch (IOException e) {
            messages.add("ERROR: Failed walking workspace (" + e.getMessage() + ")");
        }

        // Minimal semantic check.
        Path appJson = workspace.resolve("app.json");
        if (Files.exists(appJson) && Files.isRegularFile(appJson)) {
            try {
                var node = mapper.readTree(Files.readString(appJson, StandardCharsets.UTF_8));
                if (!node.isObject()) {
                    messages.add("ERROR: app.json must be a JSON object");
                } else {
                    if (node.get("ui") == null) {
                        messages.add("WARN: app.json missing 'ui' object");
                    }
                    if (node.get("version") == null) {
                        messages.add("WARN: app.json missing 'version'");
                    }
                }
            } catch (IOException e) {
                messages.add("ERROR: Failed reading app.json (" + e.getMessage() + ")");
            }
        } else {
            messages.add("WARN: app.json not found; skipping semantic checks");
        }

        return messages;
    }
}
```

## src/main/java/com/ascendion/dpai/enhancer/core/zip/ZipBundlePackager.java
```java
package com.ascendion.dpai.enhancer.core.zip;

import com.ascendion.dpai.enhancer.core.EnhancerException;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.io.OutputStream;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.time.Instant;
import java.util.zip.ZipEntry;
import java.util.zip.ZipOutputStream;

/**
 * Packages the workspace into a zip, adding a manifest.json for DataPowerAI.
 */
@Component
public class ZipBundlePackager implements BundlePackager {

    @Override
    public String createBundle(Path workspace, String bundleBaseName) {
        if (workspace == null) {
            throw EnhancerException.badRequest("INVALID_WORKSPACE", "workspace is null");
        }
        if (bundleBaseName == null || bundleBaseName.isBlank()) {
            bundleBaseName = "dpai-bundle";
        }

        Path outDir = workspace.resolve(".dpai").normalize();
        try {
            Files.createDirectories(outDir);
        } catch (IOException e) {
            throw EnhancerException.serverError("BUNDLE_DIR_FAILED", "Unable to create bundle output directory");
        }

        Path zipPath = outDir.resolve(bundleBaseName + ".zip");

        try (OutputStream os = Files.newOutputStream(zipPath, StandardOpenOption.CREATE, StandardOpenOption.TRUNCATE_EXISTING);
             ZipOutputStream zos = new ZipOutputStream(os, StandardCharsets.UTF_8)) {

            // Add manifest.json
            String manifest = "{" +
                    "\"bundleVersion\":\"1\"," +
                    "\"createdAt\":\"" + Instant.now().toString() + "\"" +
                    "}";
            addBytes(zos, "manifest.json", manifest.getBytes(StandardCharsets.UTF_8));

            // Add workspace files (excluding .dpai output itself)
            Files.walk(workspace)
                    .filter(p -> Files.isRegularFile(p))
                    .filter(p -> !p.normalize().startsWith(outDir))
                    .forEach(p -> {
                        String entryName = workspace.relativize(p).toString().replace('\\', '/');
                        try {
                            ZipEntry entry = new ZipEntry(entryName);
                            zos.putNextEntry(entry);
                            Files.copy(p, zos);
                            zos.closeEntry();
                        } catch (IOException e) {
                            throw EnhancerException.serverError("ZIP_FAILED", "Failed zipping: " + entryName);
                        }
                    });

        } catch (EnhancerException e) {
            throw e;
        } catch (IOException e) {
            throw EnhancerException.serverError("ZIP_IO_ERROR", "Failed creating zip bundle");
        }

        return zipPath.toAbsolutePath().normalize().toString();
    }

    private void addBytes(ZipOutputStream zos, String name, byte[] bytes) throws IOException {
        ZipEntry entry = new ZipEntry(name);
        zos.putNextEntry(entry);
        zos.write(bytes);
        zos.closeEntry();
    }
}
```
