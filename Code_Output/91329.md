User Story ID: 91329

1. **Requirement Summary**

Build a secure, maintainable Java-based “app enhancement pipeline” that can be invoked by a VS Code plugin to:
- Load an existing JSON-defined app (JSON app code) from disk.
- Accept a natural-language change request.
- Use basic NLP (rule/keyword-based by default) to infer intended modifications (e.g., update app name/description, add/update UI fields, and add simple validation rules) without breaking existing structure.
- Apply changes to the JSON safely with schema-like validation.
- Validate the updated JSON for errors.
- Package the updated app as a deployable artifact (ZIP) suitable for uploading to a platform (DataPowerAI) and produce a manifest.

Assumptions (due to limited story detail):
- The “JSON app” is represented as a single JSON file with top-level fields: `name`, `description`, `version`, and `ui.fields[]`.
- NLP is implemented as deterministic, auditable parsing (no external LLM calls) to keep the example self-contained and secure.
- “Connect to VS Code plugin” is achieved via a local CLI interface that a VS Code extension can invoke (common enterprise pattern). A REST/Spring Boot variant can be produced if required.
- “Upload to DataPowerAI” is represented as generating a ZIP artifact and manifest; actual platform API integration is not specified in the user story, so it is not implemented.
- Java 17 is used.

        public static List<String> validate(AppModel app) {
            List<String> errors = new ArrayList<>();
            if (safeTrimToNull(app.name) == null) {
                errors.add("name is required");
            }
            if (safeTrimToNull(app.version) == null) {
                errors.add("version is required");
            }
            if (app.ui == null) {
                errors.add("ui is required");
                return errors;
            }

            for (Field f : app.ui.fields) {
                if (safeTrimToNull(f.id) == null) {
                    errors.add("ui.fields[].id is required");
                    continue;
                }
                if (!ids.add(f.id)) {
                    errors.add("Duplicate field id: " + f.id);
                }
                if (safeTrimToNull(f.label) == null) {
                    errors.add("Field label is required for id: " + f.id);
                }
                if (f.minLength != null && f.minLength < 0) {
                    errors.add("minLength must be >= 0 for id: " + f.id);
                }
                if (f.maxLength != null && f.maxLength < 0) {
                    errors.add("maxLength must be >= 0 for id: " + f.id);
                }
                if (f.minLength != null && f.maxLength != null && f.minLength > f.maxLength) {
                    errors.add("minLength must be <= maxLength for id: " + f.id);
                }
            }
            return errors;
        }

    /**
     * A deterministic, rule-based NLP interpreter.
     *
     * Supported commands (separated by ';'):
     * - set name to <value>
     * - set description to <value>
     * - add field <Id> type <text|number|boolean> <required|optional>
     * - remove field <Id>
     * - set field <Id> required|optional
     * - set field <Id> minLength <n>
     * - set field <Id> maxLength <n>
     */
    public static final class ChangeRequestNlp {

        public List<AppChange> interpret(String request) {
            String normalized = Objects.requireNonNull(request, "request").trim();
            if (normalized.isEmpty()) {
                return List.of();
            }
            String[] parts = normalized.split(";");
            List<AppChange> changes = new ArrayList<>();
            for (String p : parts) {
                String cmd = p.trim();
                if (cmd.isEmpty()) {
                    continue;
                }
                interpretSingle(cmd).ifPresent(changes::add);
            }
            return changes;
        }

        private Optional<AppChange> interpretSingle(String cmd) {
            String lower = cmd.toLowerCase(Locale.ROOT).trim();

            if (lower.startsWith("set name to ")) {
                String val = cmd.substring(indexAfter(cmd, "set name to "));
                return Optional.of(new SetNameChange(val));
            }
            if (lower.startsWith("set description to ")) {
                String val = cmd.substring(indexAfter(cmd, "set description to "));
                return Optional.of(new SetDescriptionChange(val));
            }
            if (lower.startsWith("add field ")) {
                return Optional.of(parseAddField(cmd));
            }
            if (lower.startsWith("remove field ")) {
                String id = cmd.substring(indexAfter(cmd, "remove field ")).trim();
                return Optional.of(new RemoveFieldChange(id));
            }
            if (lower.startsWith("set field ")) {
                return Optional.of(parseSetField(cmd));
            }
            // Unknown command: ignore for safety, but could be surfaced if desired.
            return Optional.empty();
        }

        private AppChange parseAddField(String cmd) {
            // Expected: add field <Id> type <Type> required|optional
            String[] tokens = cmd.trim().split("\\s+");
            if (tokens.length < 6) {
                throw new ValidationException("Invalid add field command: " + cmd);
            }
            // tokens[0]=add tokens[1]=field tokens[2]=Id tokens[3]=type tokens[4]=Type tokens[5]=required|optional
            if (!"add".equalsIgnoreCase(tokens[0]) || !"field".equalsIgnoreCase(tokens[1])) {
                throw new ValidationException("Invalid add field command: " + cmd);
            }
            String id = tokens[2];
            if (!"type".equalsIgnoreCase(tokens[3])) {
                throw new ValidationException("Invalid add field command (missing 'type'): " + cmd);
            }
            FieldType type = FieldType.from(tokens[4]);
            boolean required = "required".equalsIgnoreCase(tokens[5]);
            String label = id; // default label
            return new UpsertFieldChange(new Field(id, label, type, required, null, null));
        }

        private AppChange parseSetField(String cmd) {
            // Supported:
            // set field <Id> required|optional
            // set field <Id> minLength <n>
            // set field <Id> maxLength <n>
            String[] tokens = cmd.trim().split("\\s+");
            if (tokens.length < 4) {
                throw new ValidationException("Invalid set field command: " + cmd);
            }
            // tokens[0]=set tokens[1]=field tokens[2]=Id ...
            if (!"set".equalsIgnoreCase(tokens[0]) || !"field".equalsIgnoreCase(tokens[1])) {
                throw new ValidationException("Invalid set field command: " + cmd);
            }
            String id = tokens[2];
            String prop = tokens[3];

            if ("required".equalsIgnoreCase(prop) || "optional".equalsIgnoreCase(prop)) {
                boolean req = "required".equalsIgnoreCase(prop);
                return new UpdateFieldPropertyChange(id, UpdateFieldPropertyChange.Kind.REQUIRED, req, null);
            }
            if ("minlength".equalsIgnoreCase(prop) && tokens.length >= 5) {
                int v = parseNonNegativeInt(tokens[4], "minLength");
                return new UpdateFieldPropertyChange(id, UpdateFieldPropertyChange.Kind.MIN_LENGTH, null, v);
            }
            if ("maxlength".equalsIgnoreCase(prop) && tokens.length >= 5) {
                int v = parseNonNegativeInt(tokens[4], "maxLength");
                return new UpdateFieldPropertyChange(id, UpdateFieldPropertyChange.Kind.MAX_LENGTH, null, v);
            }

            throw new ValidationException("Unsupported set field command: " + cmd);
        }

        private int parseNonNegativeInt(String raw, String name) {
            try {
                int v = Integer.parseInt(raw);
                if (v < 0) {
                    throw new ValidationException(name + " must be >= 0");
                }
                return v;
            } catch (NumberFormatException e) {
                throw new ValidationException("Invalid " + name + ": " + raw);
            }
        }

        private int indexAfter(String full, String prefixLowercase) {
            // Find prefix in a case-insensitive way by using lowercase lengths.
            return prefixLowercase.length();
        }
    }

    public static final class UpsertFieldChange implements AppChange {
        private final Field field;

        public UpsertFieldChange(Field field) {
            this.field = Objects.requireNonNull(field, "field");
        }

        @Override
        public AppModel apply(AppModel app) {
            String id = safeTrimToNull(field.id);
            if (id == null) {
                throw new ValidationException("Field id cannot be empty");
            }

            List<Field> newFields = new ArrayList<>(app.ui.fields);
            boolean updated = false;
            for (int i = 0; i < newFields.size(); i++) {
                if (id.equals(newFields.get(i).id)) {
                    newFields.set(i, field);
                    updated = true;
                    break;
                }
            }
            if (!updated) {
                newFields.add(field);
                newFields.sort(Comparator.comparing(f -> f.id == null ? "" : f.id));
            }
            return app.withUi(app.ui.withFields(newFields));
        }
    }

    public static final class RemoveFieldChange implements AppChange {
        private final String id;

        public RemoveFieldChange(String id) {
            this.id = safeTrimToNull(id);
        }

        @Override
        public AppModel apply(AppModel app) {
            if (id == null) {
                return app;
            }
            List<Field> newFields = new ArrayList<>();
            for (Field f : app.ui.fields) {
                if (!id.equals(f.id)) {
                    newFields.add(f);
                }
            }
            return app.withUi(app.ui.withFields(newFields));
        }
    }

    public static final class UpdateFieldPropertyChange implements AppChange {
        enum Kind { REQUIRED, MIN_LENGTH, MAX_LENGTH }

        private final String id;
        private final Kind kind;
        private final Boolean required;
        private final Integer intValue;

        public UpdateFieldPropertyChange(String id, Kind kind, Boolean required, Integer intValue) {
            this.id = safeTrimToNull(id);
            this.kind = Objects.requireNonNull(kind, "kind");
            this.required = required;
            this.intValue = intValue;
        }

        @Override
        public AppModel apply(AppModel app) {
            if (id == null) {
                throw new ValidationException("Field id cannot be empty");
            }

            List<Field> newFields = new ArrayList<>();
            boolean found = false;
            for (Field f : app.ui.fields) {
                if (id.equals(f.id)) {
                    found = true;
                    Field updated = switch (kind) {
                        case REQUIRED -> f.withRequired(Boolean.TRUE.equals(required));
                        case MIN_LENGTH -> f.withMinLength(intValue);
                        case MAX_LENGTH -> f.withMaxLength(intValue);
                    };
                    newFields.add(updated);
                } else {
                    newFields.add(f);
                }
            }
            if (!found) {
                throw new ValidationException("Field not found: " + id);
            }
            return app.withUi(app.ui.withFields(newFields));
        }
    }

        public Manifest(String manifestVersion,
                        String generatedAt,

                        List<String> appliedChanges) {

            this.generatedAt = generatedAt;
            this.sourceAppJson = sourceAppJson;
            this.updatedAppJson = updatedAppJson;
            this.updatedSha256 = updatedSha256;

        public JsonObject toJson() {
            Map<String, JsonValue> map = new LinkedHashMap<>();
            map.put("manifestVersion", Json.ofString(manifestVersion));

    public static final class DeploymentPackager {

        public Manifest buildManifest(Path originalJson, Path updatedJson, List<AppChange> changes) throws IOException {
            String sha = sha256Base64(updatedJson);
            List<String> changeNames = new ArrayList<>();
            for (AppChange c : changes) {
                changeNames.add(c.getClass().getSimpleName());
            }
            return new Manifest(
                    "1",
                    Instant.now().toString(),
                    originalJson.toAbsolutePath().toString(),
                    updatedJson.toAbsolutePath().toString(),
                    sha,
                    changeNames
            );
        }

        public void packageZip(Path zipPath, Path updatedJson, Path manifestJson) throws IOException {
            Objects.requireNonNull(zipPath, "zipPath");
            Objects.requireNonNull(updatedJson, "updatedJson");
            Objects.requireNonNull(manifestJson, "manifestJson");

            Path dir = Optional.ofNullable(zipPath.getParent()).orElse(Path.of("."));
            Files.createDirectories(dir);

            // Ensure we only zip these specific files.
            try (OutputStream fos = new BufferedOutputStream(Files.newOutputStream(zipPath));
                 ZipOutputStream zos = new ZipOutputStream(fos, StandardCharsets.UTF_8)) {

                addFile(zos, updatedJson, "app/app.json");
                addFile(zos, manifestJson, "manifest.json");
            }
        }

        private void addFile(ZipOutputStream zos, Path file, String entryName) throws IOException {
            ZipEntry entry = new ZipEntry(entryName);
            entry.setTime(System.currentTimeMillis());
            zos.putNextEntry(entry);
            try (InputStream in = new BufferedInputStream(Files.newInputStream(file))) {
                in.transferTo(zos);
            }
            zos.closeEntry();
        }

        private String sha256Base64(Path file) throws IOException {
            MessageDigest md;
            try {
                md = MessageDigest.getInstance("SHA-256");
            } catch (NoSuchAlgorithmException e) {
                throw new IllegalStateException("SHA-256 not supported", e);
            }
            try (InputStream in = new BufferedInputStream(Files.newInputStream(file))) {
                byte[] buf = new byte[8192];
                int r;
                while ((r = in.read(buf)) >= 0) {
                    if (r > 0) {
                        md.update(buf, 0, r);
                    }
                }
            }
            return Base64.getEncoder().encodeToString(md.digest());
        }
    }

    public sealed interface JsonValue permits JsonObject, JsonArray, JsonString, JsonNumber, JsonBoolean, JsonNull {
        String toCompactString();
    }

    public static final class JsonObject implements JsonValue {
        private final Map<String, JsonValue> map;

        public JsonObject(Map<String, JsonValue> map) {
            this.map = Collections.unmodifiableMap(new LinkedHashMap<>(Objects.requireNonNull(map, "map")));
        }

        public Optional<JsonValue> get(String key) {
            return Optional.ofNullable(map.get(key));
        }

        public Optional<String> getString(String key) {
            JsonValue v = map.get(key);
            if (v instanceof JsonString s) {
                return Optional.ofNullable(s.value);
            }
            return Optional.empty();
        }

        public Optional<Integer> getInt(String key) {
            JsonValue v = map.get(key);
            if (v instanceof JsonNumber n) {
                try {
                    return Optional.of(Integer.parseInt(n.value));
                } catch (NumberFormatException e) {
                    return Optional.empty();
                }
            }
            return Optional.empty();
        }

        public Optional<Boolean> getBoolean(String key) {
            JsonValue v = map.get(key);
            if (v instanceof JsonBoolean b) {
                return Optional.of(b.value);
            }
            return Optional.empty();
        }

        public Optional<JsonObject> getObject(String key) {
            JsonValue v = map.get(key);
            if (v instanceof JsonObject o) {
                return Optional.of(o);
            }
            return Optional.empty();
        }

        public Optional<JsonArray> getArray(String key) {
            JsonValue v = map.get(key);
            if (v instanceof JsonArray a) {
                return Optional.of(a);
            }
            return Optional.empty();
        }

        @Override
        public String toCompactString() {
            StringBuilder sb = new StringBuilder();
            sb.append('{');
            boolean first = true;
            for (Map.Entry<String, JsonValue> e : map.entrySet()) {
                if (!first) {
                    sb.append(',');
                }
                first = false;
                sb.append('"').append(Json.escape(e.getKey())).append('"').append(':');
                sb.append(e.getValue().toCompactString());
            }
            sb.append('}');
            return sb.toString();
        }
    }

        public static String escape(String s) {
            if (s == null) {
                return "";
            }
            StringBuilder sb = new StringBuilder(s.length() + 16);
            for (int i = 0; i < s.length(); i++) {
                char c = s.charAt(i);
                switch (c) {
                    case '"' -> sb.append("\\\"");
                    case '\\' -> sb.append("\\\\");
                    case '\b' -> sb.append("\\b");
                    case '\f' -> sb.append("\\f");
                    case '\n' -> sb.append("\\n");
                    case '\r' -> sb.append("\\r");
                    case '\t' -> sb.append("\\t");
                    default -> {
                        if (c < 0x20) {
                            sb.append(String.format("\\u%04x", (int) c));
                        } else {
                            sb.append(c);
                        }
                    }
                }
            }
            return sb.toString();
        }

    public static final class JsonParseException extends RuntimeException {
        public JsonParseException(String message) {
            super(message);
        }
    }

    private static final class JsonParser {
        private final String s;
        private int i;

        JsonParser(String s) {
            this.s = Objects.requireNonNull(s, "json");
            this.i = 0;
        }

        JsonValue parse() {
            skipWs();
            JsonValue v = parseValue();
            skipWs();
            if (i != s.length()) {
                throw error("Trailing content");
            }
            return v;
        }

        private JsonValue parseValue() {
            skipWs();
            if (i >= s.length()) {
                throw error("Unexpected end");
            }
            char c = s.charAt(i);
            return switch (c) {
                case '{' -> parseObject();
                case '[' -> parseArray();
                case '"' -> new JsonString(parseString());
                case 't' -> parseLiteral("true", new JsonBoolean(true));
                case 'f' -> parseLiteral("false", new JsonBoolean(false));
                case 'n' -> parseLiteral("null", new JsonNull());
                default -> {
                    if (c == '-' || (c >= '0' && c <= '9')) {
                        yield new JsonNumber(parseNumber());
                    }
                    throw error("Unexpected char: " + c);
                }
            };
        }

        private JsonValue parseLiteral(String lit, JsonValue v) {
            if (s.startsWith(lit, i)) {
                i += lit.length();
                return v;
            }
            throw error("Expected literal: " + lit);
        }

        private JsonObject parseObject() {
            expect('{');
            skipWs();
            Map<String, JsonValue> map = new LinkedHashMap<>();
            if (peek('}')) {
                expect('}');
                return new JsonObject(map);
            }
            while (true) {
                skipWs();
                String key = parseString();
                skipWs();
                expect(':');
                skipWs();
                JsonValue val = parseValue();
                map.put(key, val);
                skipWs();
                if (peek('}')) {
                    expect('}');
                    break;
                }
                expect(',');
            }
            return new JsonObject(map);
        }

        private JsonArray parseArray() {
            expect('[');
            skipWs();
            List<JsonValue> list = new ArrayList<>();
            if (peek(']')) {
                expect(']');
                return new JsonArray(list);
            }
            while (true) {
                skipWs();
                JsonValue v = parseValue();
                list.add(v);
                skipWs();
                if (peek(']')) {
                    expect(']');
                    break;
                }
                expect(',');
            }
            return new JsonArray(list);
        }

        private String parseString() {
            expect('"');
            StringBuilder sb = new StringBuilder();
            while (i < s.length()) {
                char c = s.charAt(i++);
                if (c == '"') {
                    return sb.toString();
                }
                if (c == '\\') {
                    if (i >= s.length()) {
                        throw error("Unterminated escape");
                    }
                    char e = s.charAt(i++);
                    switch (e) {
                        case '"' -> sb.append('"');
                        case '\\' -> sb.append('\\');
                        case '/' -> sb.append('/');
                        case 'b' -> sb.append('\b');
                        case 'f' -> sb.append('\f');
                        case 'n' -> sb.append('\n');
                        case 'r' -> sb.append('\r');
                        case 't' -> sb.append('\t');
                        case 'u' -> {
                            if (i + 4 > s.length()) {
                                throw error("Invalid unicode escape");
                            }
                            String hex = s.substring(i, i + 4);
                            i += 4;
                            try {
                                sb.append((char) Integer.parseInt(hex, 16));
                            } catch (NumberFormatException ex) {
                                throw error("Invalid unicode hex: " + hex);
                            }
                        }
                        default -> throw error("Invalid escape: \\" + e);
                    }
                } else {
                    sb.append(c);
                }
            }
            throw error("Unterminated string");
        }

        private String parseNumber() {
            int start = i;
            if (peek('-')) {
                i++;
            }
            if (i >= s.length()) {
                throw error("Invalid number");
            }
            if (s.charAt(i) == '0') {
                i++;
            } else {
                if (!isDigit(s.charAt(i))) {
                    throw error("Invalid number");
                }
                while (i < s.length() && isDigit(s.charAt(i))) {
                    i++;
                }
            }
            if (i < s.length() && s.charAt(i) == '.') {
                i++;
                if (i >= s.length() || !isDigit(s.charAt(i))) {
                    throw error("Invalid number fraction");
                }
                while (i < s.length() && isDigit(s.charAt(i))) {
                    i++;
                }
            }
            if (i < s.length()) {
                char c = s.charAt(i);
                if (c == 'e' || c == 'E') {
                    i++;
                    if (i < s.length() && (s.charAt(i) == '+' || s.charAt(i) == '-')) {
                        i++;
                    }
                    if (i >= s.length() || !isDigit(s.charAt(i))) {
                        throw error("Invalid exponent");
                    }
                    while (i < s.length() && isDigit(s.charAt(i))) {
                        i++;
                    }
                }
            }
            return s.substring(start, i);
        }

        private boolean isDigit(char c) {
            return c >= '0' && c <= '9';
        }

        private void skipWs() {
            while (i < s.length()) {
                char c = s.charAt(i);
                if (c == ' ' || c == '\n' || c == '\r' || c == '\t') {
                    i++;
                } else {
                    break;
                }
            }
        }

        private boolean peek(char c) {
            return i < s.length() && s.charAt(i) == c;
        }

        private void expect(char c) {
            if (i >= s.length() || s.charAt(i) != c) {
                throw error("Expected '" + c + "'");
            }
            i++;
        }

        private JsonParseException error(String msg) {
            return new JsonParseException(msg + " at position " + i);
        }
    }

    private static String safeTrimToNull(String s) {
        if (s == null) {
            return null;
        }
        String t = s.trim();
        return t.isEmpty() ? null : t;
    }
