1. **Requirement Summary**

Build a secure, maintainable Java implementation that simulates connecting an existing “JSON app” project to a VS Code plugin, uses NLP to understand user change requests, applies safe JSON transformations to the app definition, validates the updated app for schema/consistency errors, packages the result, and “uploads/deploys” it to a DataPowerAI-like endpoint.

**Assumptions** (defaults applied due to missing specifics)
- The “JSON app” is represented as a single JSON document with `app`, `ui`, and `features` sections.
- The VS Code plugin interaction is represented as a local boundary: read an input JSON file and a natural-language change request string.
- NLP is implemented deterministically (rules/regex + keyword parsing) to keep the solution self-contained and secure without external ML services.
- “Deploy to DataPowerAI” is represented as an HTTPS POST upload to a configured URL (no hardcoded secrets); if no URL is provided, the tool writes a deployable ZIP locally.
- Authentication is via optional Bearer token read from environment variable `DATAPOWERAI_TOKEN`.
- Java 17 is used.
- No real PII is processed; sample data uses placeholders.

     * Deterministic "NLP" interpreter:
     * extracts intent using keyword patterns and quoted text.

    private static final class NlpInterpreter {
        static ChangePlan interpret(String changeRequest) {
            if (changeRequest == null || changeRequest.trim().isEmpty()) {
                throw new ValidationException("Change request must not be empty.");
            }

            String raw = changeRequest.trim();
            String normalized = raw.toLowerCase(Locale.ROOT);

            List<UiChange> ui = new ArrayList<>();
            Set<String> enable = new TreeSet<>();
            Set<String> disable = new TreeSet<>();

            // Title update: "set title to 'X'" or "change title to \"X\""
            String title = extractQuotedValueAfter(normalized, raw, List.of("set title to", "change title to", "update title to"));
            if (title != null) {
                ui.add(new SetTitleChange(title));
            }

            // Add button: "add a button labeled 'X'" optionally with id: "id 'submit'"
            if (normalized.contains("add") && normalized.contains("button")) {
                String label = extractQuotedValueAfter(normalized, raw, List.of("labeled", "labelled", "label"));
                if (label != null) {
                    String id = Optional.ofNullable(extractQuotedValueAfter(normalized, raw, List.of("id")))
                            .orElseGet(() -> "btn_" + slug(label));
                    ui.add(new AddButtonChange(id, label));
                }
            }

            // Enable/disable features: "enable feature analytics" / "disable feature x"
            enable.addAll(extractFeatureNames(normalized, raw, "enable"));
            disable.addAll(extractFeatureNames(normalized, raw, "disable"));

            // If both enable and disable contain same feature, prefer last instruction in text:
            for (String f : Set.copyOf(enable)) {
                if (disable.contains(f)) {
                    int lastEnable = normalized.lastIndexOf("enable");
                    int lastDisable = normalized.lastIndexOf("disable");
                    if (lastDisable > lastEnable) {
                        enable.remove(f);
                    } else {
                        disable.remove(f);
                    }
                }
            }

            return new ChangePlan(ui, enable, disable);
        }

        private static Set<String> extractFeatureNames(String normalizedLower, String raw, String verb) {
            Set<String> out = new TreeSet<>();
            int idx = 0;
            while (idx >= 0) {
                idx = normalizedLower.indexOf(verb, idx);
                if (idx < 0) break;

                int end = Math.min(raw.length(), idx + 120);
                String windowRaw = raw.substring(idx, end);
                String windowLower = normalizedLower.substring(idx, end);

                // Prefer quoted feature name: enable feature "analytics"
                String quoted = extractQuotedValueAfter(windowLower, windowRaw, List.of("feature", "features"));
                if (quoted != null) {
                    out.add(requireSafeIdentifier(quoted));
                    idx = idx + verb.length();
                    continue;
                }

                // Non-quoted: enable feature analytics
                int featureIdx = windowLower.indexOf("feature");
                if (featureIdx >= 0) {
                    String after = windowRaw.substring(featureIdx + "feature".length()).trim();
                    String token = firstToken(after);
                    if (token != null) {
                        out.add(requireSafeIdentifier(token));
                    }
                }

                idx = idx + verb.length();
            }
            return out;
        }

        private static String firstToken(String s) {
            if (s == null) return null;
            String t = s.trim();
            if (t.isEmpty()) return null;
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < t.length(); i++) {
                char c = t.charAt(i);
                if (Character.isLetterOrDigit(c) || c == '_' || c == '-') {
                    sb.append(c);
                } else {
                    break;
                }
            }
            return sb.length() == 0 ? null : sb.toString();
        }

        private static String extractQuotedValueAfter(String normalizedLower, String raw, List<String> markers) {
            for (String marker : markers) {
                int m = normalizedLower.indexOf(marker);
                if (m < 0) continue;
                int startSearch = m + marker.length();
                String tailRaw = raw.substring(Math.min(startSearch, raw.length()));
                return extractFirstQuoted(tailRaw);
            }
            return null;
        }

        private static String extractFirstQuoted(String s) {
            if (s == null) return null;
            int single = s.indexOf('\'');
            int dbl = s.indexOf('"');
            int start;
            char quote;
            if (single < 0 && dbl < 0) return null;
            if (single >= 0 && (dbl < 0 || single < dbl)) {
                start = single;
                quote = '\'';
            } else {
                start = dbl;
                quote = '"';
            }
            int end = s.indexOf(quote, start + 1);
            if (end < 0) return null;
            String val = s.substring(start + 1, end).trim();
            return val.isEmpty() ? null : val;
        }

        private static String slug(String input) {
            String lower = input.toLowerCase(Locale.ROOT).trim();
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < lower.length(); i++) {
                char c = lower.charAt(i);
                if (Character.isLetterOrDigit(c)) {
                    sb.append(c);
                } else if (c == ' ' || c == '-' || c == '_') {
                    if (sb.length() > 0 && sb.charAt(sb.length() - 1) != '_') {
                        sb.append('_');
                    }
                }
            }
            String out = sb.toString();
            return out.isEmpty() ? "button" : out;
        }
    }

    /** Applies a change plan to the JSON document with safe defaults. */
    private static final class AppTransformer {
        static JsonObject applyPlan(JsonObject app, ChangePlan plan) {
            Objects.requireNonNull(app, "app");
            Objects.requireNonNull(plan, "plan");

            JsonObject out = app.deepCopy();

            ensureObject(out, "app");
            ensureObject(out, "ui");

            JsonObject ui = out.getObject("ui");
            JsonArray features = out.getOrCreateArray("features");

            // Apply UI changes
            for (UiChange c : plan.uiChanges) {
                if (c instanceof SetTitleChange st) {
                    ui.put("title", new JsonString(st.title));
                } else if (c instanceof AddButtonChange ab) {
                    JsonArray buttons = ui.getOrCreateArray("buttons");
                    // Avoid duplicates by id
                    if (!containsButtonId(buttons, ab.id)) {
                        JsonObject btn = new JsonObject();
                        btn.put("id", new JsonString(ab.id));
                        btn.put("label", new JsonString(ab.label));
                        btn.put("type", new JsonString("button"));
                        buttons.add(btn);
                    }
                }
            }

            // Apply feature toggles
            for (String f : plan.enableFeatures) {
                if (!containsString(features, f)) {
                    features.add(new JsonString(f));
                }
            }
            for (String f : plan.disableFeatures) {
                removeString(features, f);
            }

            // Keep a minimal metadata stamp without breaking existing properties.
            JsonObject meta = out.getOrCreateObject("meta");
            meta.put("lastEnhancedAt", new JsonString(Instant.now().toString()));
            meta.put("enhancer", new JsonString("AppEnhancerTool"));

            return out;
        }

        private static void ensureObject(JsonObject root, String key) {
            JsonValue v = root.get(key);
            if (v == null) {
                root.put(key, new JsonObject());
                return;
            }
            if (!(v instanceof JsonObject)) {
                throw new ValidationException("Field '" + key + "' must be an object.");
            }
        }

        private static boolean containsButtonId(JsonArray buttons, String id) {
            for (JsonValue v : buttons.values) {
                if (v instanceof JsonObject o) {
                    JsonValue vid = o.get("id");
                    if (vid instanceof JsonString s && id.equals(s.value)) {
                        return true;
                    }
                }
            }
            return false;
        }

        private static boolean containsString(JsonArray arr, String s) {
            for (JsonValue v : arr.values) {
                if (v instanceof JsonString js && js.value.equals(s)) return true;
            }
            return false;
        }

        private static void removeString(JsonArray arr, String s) {
            arr.values.removeIf(v -> (v instanceof JsonString js) && js.value.equals(s));
        }
    }

    /** Validates updated JSON document for basic schema integrity. */
    private static final class AppValidator {
        static void validate(JsonObject root) {
            Objects.requireNonNull(root, "root");

            JsonObject app = requireObject(root, "app");
            JsonObject ui = requireObject(root, "ui");

            // app.name required
            String name = optionalString(app, "name").orElse(null);
            if (name == null || name.trim().isEmpty()) {
                throw new ValidationException("app.name is required and must be non-empty.");
            }

            // ui.title optional but if present must be non-empty
            Optional<String> titleOpt = optionalString(ui, "title");
            if (titleOpt.isPresent() && titleOpt.get().trim().isEmpty()) {
                throw new ValidationException("ui.title must be non-empty if present.");
            }

            // ui.buttons if present must be array of objects with id/label
            JsonValue buttons = ui.get("buttons");
            if (buttons != null) {
                if (!(buttons instanceof JsonArray ba)) {
                    throw new ValidationException("ui.buttons must be an array if present.");
                }
                Set<String> ids = new TreeSet<>();
                for (JsonValue v : ba.values) {
                    if (!(v instanceof JsonObject o)) {
                        throw new ValidationException("ui.buttons elements must be objects.");
                    }
                    String id = requiredString(o, "id");
                    String label = requiredString(o, "label");
                    if (!ids.add(id)) {
                        throw new ValidationException("Duplicate button id: " + id);
                    }
                    if (label.trim().isEmpty()) {
                        throw new ValidationException("Button label must be non-empty.");
                    }
                }
            }

            // features if present must be array of strings
            JsonValue features = root.get("features");
            if (features != null) {
                if (!(features instanceof JsonArray fa)) {
                    throw new ValidationException("features must be an array if present.");
                }
                for (JsonValue v : fa.values) {
                    if (!(v instanceof JsonString)) {
                        throw new ValidationException("features elements must be strings.");
                    }
                }
            }
        }

        private static JsonObject requireObject(JsonObject root, String key) {
            JsonValue v = root.get(key);
            if (!(v instanceof JsonObject o)) {
                throw new ValidationException("Field '" + key + "' is required and must be an object.");
            }
            return o;
        }

        private static String requiredString(JsonObject obj, String key) {
            return optionalString(obj, key).orElseThrow(() -> new ValidationException("Field '" + key + "' is required and must be a string."));
        }

        private static Optional<String> optionalString(JsonObject obj, String key) {
            JsonValue v = obj.get(key);
            if (v == null) return Optional.empty();
            if (!(v instanceof JsonString s)) {
                throw new ValidationException("Field '" + key + "' must be a string.");
            }
            return Optional.ofNullable(s.value);
        }
    }

    /** Packages the updated JSON (and a manifest) into a ZIP for deployment. */
    private static final class Packager {
        static byte[] packageAsZip(Path updatedJsonPath, JsonObject updatedJson) throws IOException {
            Objects.requireNonNull(updatedJsonPath, "updatedJsonPath");
            Objects.requireNonNull(updatedJson, "updatedJson");

            String manifest = createManifest(updatedJsonPath, updatedJson);

            ByteArrayOutputStream baos = new ByteArrayOutputStream();
            try (ZipOutputStream zos = new ZipOutputStream(new BufferedOutputStream(baos), StandardCharsets.UTF_8)) {
                // JSON entry
                zipFile(zos, updatedJsonPath, "app.json");

                // Manifest entry
                ZipEntry me = new ZipEntry("manifest.txt");
                zos.putNextEntry(me);
                byte[] mb = manifest.getBytes(StandardCharsets.UTF_8);
                zos.write(mb);
                zos.closeEntry();
            }
            return baos.toByteArray();
        }

        private static String createManifest(Path updatedJsonPath, JsonObject updatedJson) throws IOException {
            byte[] jsonBytes = Files.readAllBytes(updatedJsonPath);
            String jsonSha = Crypto.sha256Hex(jsonBytes);
            String prettyName = Optional.ofNullable(updatedJson.getObject("app").get("name"))
                    .filter(v -> v instanceof JsonString)
                    .map(v -> ((JsonString) v).value)
                    .orElse("unknown");

            return "name=" + prettyName + "\n" +
                    "generatedAt=" + Instant.now().toString() + "\n" +
                    "appJsonSha256=" + jsonSha + "\n";
        }

        private static void zipFile(ZipOutputStream zos, Path path, String entryName) throws IOException {
            ZipEntry entry = new ZipEntry(entryName);
            zos.putNextEntry(entry);
            try (InputStream in = new BufferedInputStream(new FileInputStream(path.toFile()))) {
                byte[] buf = new byte[8192];
                int r;
                while ((r = in.read(buf)) >= 0) {
                    zos.write(buf, 0, r);
                }
            }
            zos.closeEntry();
        }
    }

    /** Deploys a ZIP package via HTTPS POST multipart/form-data. */
    private static final class Deployer {
        static DeployResult deploy(byte[] zipBytes, URI deployUrl) throws IOException, InterruptedException {
            Objects.requireNonNull(zipBytes, "zipBytes");
            Objects.requireNonNull(deployUrl, "deployUrl");

            String boundary = "----AppEnhancerBoundary" + System.currentTimeMillis();
            byte[] body = MultipartBodyBuilder.build(boundary, zipBytes);

            HttpClient client = HttpClient.newBuilder()
                    .followRedirects(HttpClient.Redirect.NEVER)
                    .build();

            HttpRequest.Builder rb = HttpRequest.newBuilder()
                    .uri(deployUrl)
                    .header("Content-Type", "multipart/form-data; boundary=" + boundary)
                    .header("Accept", "application/json, text/plain, */*")
                    .POST(HttpRequest.BodyPublishers.ofByteArray(body));

            String token = System.getenv("DATAPOWERAI_TOKEN");
            if (token != null && !token.trim().isEmpty()) {
                rb.header("Authorization", "Bearer " + token.trim());
            }

            HttpResponse<String> resp = client.send(rb.build(), HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8));
            return new DeployResult(resp.statusCode(), resp.body());
        }
    }

    /** Minimal multipart builder for single file field: "file". */
    private static final class MultipartBodyBuilder {
        static byte[] build(String boundary, byte[] zipBytes) throws IOException {
            Objects.requireNonNull(boundary, "boundary");
            Objects.requireNonNull(zipBytes, "zipBytes");

            ByteArrayOutputStream out = new ByteArrayOutputStream();

            writeAscii(out, "--" + boundary + "\r\n");
            writeAscii(out, "Content-Disposition: form-data; name=\"file\"; filename=\"app.zip\"\r\n");
            writeAscii(out, "Content-Type: application/zip\r\n\r\n");
            out.write(zipBytes);
            writeAscii(out, "\r\n--" + boundary + "--\r\n");

            return out.toByteArray();
        }

        private static void writeAscii(OutputStream out, String s) throws IOException {
            out.write(s.getBytes(StandardCharsets.US_ASCII));
        }
    }

    /** Crypto helpers. */
    private static final class Crypto {
        static String sha256Hex(byte[] data) {
            Objects.requireNonNull(data, "data");
            try {
                MessageDigest md = MessageDigest.getInstance("SHA-256");
                byte[] digest = md.digest(data);
                StringBuilder sb = new StringBuilder(digest.length * 2);
                for (byte b : digest) {
                    sb.append(Character.forDigit((b >>> 4) & 0xF, 16));
                    sb.append(Character.forDigit((b) & 0xF, 16));
                }
                return sb.toString();
            } catch (NoSuchAlgorithmException e) {
                throw new IllegalStateException("SHA-256 not available", e);
            }
        }
    }

    /** Input safety: allow identifiers used as ids/features. */
    private static String requireSafeIdentifier(String s) {
        if (s == null) {
            throw new ValidationException("Identifier must not be null.");
        }
        String t = s.trim();
        if (t.isEmpty()) {
            throw new ValidationException("Identifier must not be empty.");
        }
        // restrict to a safe subset
        for (int i = 0; i < t.length(); i++) {
            char c = t.charAt(i);
            boolean ok = Character.isLetterOrDigit(c) || c == '_' || c == '-';
            if (!ok) {
                throw new ValidationException("Invalid identifier: '" + s + "'. Allowed: [A-Za-z0-9_-]");
            }
        }
        return t;
    }

    /** Input safety: allow general UI text; block control chars. */
    private static String requireSafeText(String s) {
        if (s == null) {
            throw new ValidationException("Text must not be null.");
        }
        String t = s.trim();
        if (t.isEmpty()) {
            throw new ValidationException("Text must not be empty.");
        }
        for (int i = 0; i < t.length(); i++) {
            char c = t.charAt(i);
            if (c < 0x20 && c != '\t') {
                throw new ValidationException("Text contains control characters.");
            }
        }
        if (t.length() > 200) {
            throw new ValidationException("Text too long (max 200 chars).");
        }
        return t;
    }

    /** Base JSON value. */
    private sealed interface JsonValue permits JsonObject, JsonArray, JsonString, JsonNumber, JsonBoolean, JsonNull {
        JsonValue deepCopy();
    }

    private static final class JsonObject implements JsonValue {
        private final Map<String, JsonValue> map = new LinkedHashMap<>();

        JsonValue get(String key) { return map.get(key); }

        JsonObject getObject(String key) {
            JsonValue v = map.get(key);
            if (v == null) return null;
            if (!(v instanceof JsonObject o)) {
                throw new ValidationException("Expected object at '" + key + "'.");
            }
            return o;
        }

        JsonArray getOrCreateArray(String key) {
            JsonValue v = map.get(key);

                JsonArray a = new JsonArray();
                map.put(key, a);

            if (!(v instanceof JsonArray a)) {
                throw new ValidationException("Expected array at '" + key + "'.");

        JsonObject getOrCreateObject(String key) {
            JsonValue v = map.get(key);

            }
            if (!(v instanceof JsonObject o)) {
                throw new ValidationException("Expected object at '" + key + "'.");

            return o;
        }

        void put(String key, JsonValue value) {
            if (key == null || key.isBlank()) {
                throw new ValidationException("JSON object key must be non-empty.");
            }
            map.put(key, value == null ? JsonNull.INSTANCE : value);
        }

        @Override
        public JsonValue deepCopy() {

            for (Map.Entry<String, JsonValue> e : map.entrySet()) {
                copy.put(e.getKey(), e.getValue() == null ? JsonNull.INSTANCE : e.getValue().deepCopy());
            }
            return copy;
        }

    /** JSON parser and serializer. */
    private static final class Json {
        static JsonValue parse(String s) {
            if (s == null) throw new JsonParseException("Input is null");
            Parser p = new Parser(s);
            JsonValue v = p.parseValue();
            p.skipWs();
            if (!p.eof()) {
                throw new JsonParseException("Trailing content at pos " + p.pos);
            }
            return v;
        }

        static String stringify(JsonValue v) {
            StringBuilder sb = new StringBuilder();
            write(sb, v);
            return sb.toString();
        }

        private static void write(StringBuilder sb, JsonValue v) {
            if (v == null || v instanceof JsonNull) {
                sb.append("null");
            } else if (v instanceof JsonBoolean b) {
                sb.append(b.value ? "true" : "false");
            } else if (v instanceof JsonNumber n) {
                sb.append(n.value);
            } else if (v instanceof JsonString s) {
                sb.append('"').append(escape(s.value)).append('"');
            } else if (v instanceof JsonArray a) {
                sb.append('[');
                for (int i = 0; i < a.values.size(); i++) {
                    if (i > 0) sb.append(',');
                    write(sb, a.values.get(i));
                }
                sb.append(']');
            } else if (v instanceof JsonObject o) {
                sb.append('{');
                boolean first = true;
                for (Map.Entry<String, JsonValue> e : o.map.entrySet()) {
                    if (!first) sb.append(',');
                    first = false;
                    sb.append('"').append(escape(e.getKey())).append('"').append(':');
                    write(sb, e.getValue());
                }
                sb.append('}');
            } else {
                throw new IllegalStateException("Unknown JsonValue: " + v.getClass());
            }
        }

        private static String escape(String s) {
            if (s == null) return "";
            StringBuilder sb = new StringBuilder(s.length() + 16);
            for (int i = 0; i < s.length(); i++) {
                char c = s.charAt(i);
                switch (c) {
                    case '"' -> sb.append("\\\"");
                    case '\\' -> sb.append("\\\\");
                    case '\b' -> sb.append("\\b");
                    case '\f' -> sb.append("\\f");
                    case '\n' -> sb.append("\\n");
                    case '\r' -> sb.append("\\r");
                    case '\t' -> sb.append("\\t");
                    default -> {
                        if (c < 0x20) {
                            sb.append(String.format("\\u%04x", (int) c));
                        } else {
                            sb.append(c);
                        }
                    }
                }
            }
            return sb.toString();
        }

        private static final class Parser {
            private final String s;
            int pos;

            Parser(String s) {
                this.s = s;
            }

            boolean eof() { return pos >= s.length(); }

            void skipWs() {
                while (!eof()) {
                    char c = s.charAt(pos);
                    if (c == ' ' || c == '\n' || c == '\r' || c == '\t') {
                        pos++;
                    } else {
                        break;
                    }
                }
            }

            JsonValue parseValue() {
                skipWs();
                if (eof()) throw new JsonParseException("Unexpected end of input");
                char c = s.charAt(pos);
                return switch (c) {
                    case '{' -> parseObject();
                    case '[' -> parseArray();
                    case '"' -> new JsonString(parseString());
                    case 't', 'f' -> parseBoolean();
                    case 'n' -> parseNull();
                    default -> {
                        if (c == '-' || Character.isDigit(c)) {
                            yield parseNumber();
                        }
                        throw new JsonParseException("Unexpected character '" + c + "' at pos " + pos);
                    }
                };
            }

            JsonObject parseObject() {
                expect('{');
                JsonObject obj = new JsonObject();
                skipWs();
                if (peek('}')) {
                    pos++;
                    return obj;
                }
                while (true) {
                    skipWs();
                    if (!peek('"')) {
                        throw new JsonParseException("Expected string key at pos " + pos);
                    }
                    String key = parseString();
                    skipWs();
                    expect(':');
                    JsonValue val = parseValue();
                    obj.put(key, val);
                    skipWs();
                    if (peek('}')) {
                        pos++;
                        return obj;
                    }
                    expect(',');
                }
            }

            JsonArray parseArray() {
                expect('[');
                JsonArray arr = new JsonArray();
                skipWs();
                if (peek(']')) {
                    pos++;
                    return arr;
                }
                while (true) {
                    JsonValue v = parseValue();
                    arr.add(v);
                    skipWs();
                    if (peek(']')) {
                        pos++;
                        return arr;
                    }
                    expect(',');
                }
            }

            JsonValue parseBoolean() {
                if (s.startsWith("true", pos)) {
                    pos += 4;
                    return new JsonBoolean(true);
                }
                if (s.startsWith("false", pos)) {
                    pos += 5;
                    return new JsonBoolean(false);
                }
                throw new JsonParseException("Invalid boolean at pos " + pos);
            }

            JsonValue parseNull() {
                if (s.startsWith("null", pos)) {
                    pos += 4;
                    return JsonNull.INSTANCE;
                }
                throw new JsonParseException("Invalid null at pos " + pos);
            }

            JsonValue parseNumber() {
                int start = pos;
                if (peek('-')) pos++;
                if (eof()) throw new JsonParseException("Invalid number at end");

                if (peek('0')) {
                    pos++;
                } else if (Character.isDigit(s.charAt(pos))) {
                    while (!eof() && Character.isDigit(s.charAt(pos))) pos++;
                } else {
                    throw new JsonParseException("Invalid number at pos " + pos);
                }

                if (!eof() && peek('.')) {
                    pos++;
                    if (eof() || !Character.isDigit(s.charAt(pos))) {
                        throw new JsonParseException("Invalid fraction at pos " + pos);
                    }
                    while (!eof() && Character.isDigit(s.charAt(pos))) pos++;
                }

                if (!eof() && (peek('e') || peek('E'))) {
                    pos++;
                    if (!eof() && (peek('+') || peek('-'))) pos++;
                    if (eof() || !Character.isDigit(s.charAt(pos))) {
                        throw new JsonParseException("Invalid exponent at pos " + pos);
                    }
                    while (!eof() && Character.isDigit(s.charAt(pos))) pos++;
                }

                return new JsonNumber(s.substring(start, pos));
            }

            String parseString() {
                expect('"');
                StringBuilder sb = new StringBuilder();
                while (!eof()) {
                    char c = s.charAt(pos++);
                    if (c == '"') {
                        return sb.toString();
                    }
                    if (c == '\\') {
                        if (eof()) throw new JsonParseException("Unterminated escape at pos " + pos);
                        char e = s.charAt(pos++);
                        switch (e) {
                            case '"' -> sb.append('"');
                            case '\\' -> sb.append('\\');
                            case '/' -> sb.append('/');
                            case 'b' -> sb.append('\b');
                            case 'f' -> sb.append('\f');
                            case 'n' -> sb.append('\n');
                            case 'r' -> sb.append('\r');
                            case 't' -> sb.append('\t');
                            case 'u' -> {
                                if (pos + 4 > s.length()) throw new JsonParseException("Invalid unicode escape at pos " + pos);
                                String hex = s.substring(pos, pos + 4);
                                pos += 4;
                                try {
                                    sb.append((char) Integer.parseInt(hex, 16));
                                } catch (NumberFormatException ex) {
                                    throw new JsonParseException("Invalid unicode escape: \\u" + hex);
                                }
                            }
                            default -> throw new JsonParseException("Invalid escape \\" + e + " at pos " + (pos - 1));
                        }
                    } else {
                        sb.append(c);
                    }
                }
                throw new JsonParseException("Unterminated string");
            }

            boolean peek(char c) {
                return !eof() && s.charAt(pos) == c;
            }

            void expect(char c) {
                if (eof() || s.charAt(pos) != c) {
                    throw new JsonParseException("Expected '" + c + "' at pos " + pos);
                }
                pos++;
            }
        }
    }
