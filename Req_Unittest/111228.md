    // TC-01: AC 1 - identify AAVA Base URL Group by group type/category
    @Test
    void shouldUseGroupTypeToIdentifyAavaBaseUrlGroup() {
        ApiBaseUrlGroup group = TestData.aavaGroup("Shared APIs", "AAVA");

        when(tokenResolver.resolveForCurrentUser()).thenReturn("token-123");
        when(apiInvoker.invoke(any())).thenReturn(TestData.successResponse());

        AavaGroupTokenRequestExecutorTestSupport.ExecutionResult result = executor.executeForCurrentUser(api);

        assertTrue(result.aavaAuthenticationUsed());
        verify(tokenResolver).resolveForCurrentUser();
        verify(apiInvoker).invoke(any());

    // TC-02: AC 1 and AC 12 - do not rely only on group name; non-AAVA groups keep existing auth
    @Test
    void shouldNotTreatNonAavaGroupAsAavaWhenOnlyNameMatches() {
        ApiBaseUrlGroup group = TestData.aavaGroup("AAVA", "EXTERNAL");
        GenericAPI api = TestData.api(group, "/api/v1/projects", "https://partner.example.com");
        when(apiInvoker.invoke(any())).thenReturn(TestData.successResponse());

    void shouldResolveCurrentUsersAccessTokenAtExecutionTimeForAavaGroup() {

    void shouldRejectExecutionWhenCurrentUserTokenIsUnavailable() {

        GenericAPI api = TestData.api(group, "/api/v1/projects", "https://aava.example.com");

    void shouldRefreshExpiredTokenAndRetryOnceWhenPolicyAllows() {

    // TC-07: AC 10 - avoid infinite retry when unauthorized persists after refresh
    @Test
    void shouldAvoidInfiniteRetryWhenUnauthorizedPersistsAfterRefresh() {
        ApiBaseUrlGroup group = TestData.aavaGroup("Core", "AAVA");
        GenericAPI api = TestData.api(group, "/api/v1/projects", "https://aava.example.com");
        when(tokenResolver.resolveForCurrentUser()).thenReturn("expired-token");

    static class AavaGroupTokenRequestExecutorTestSupport {

        // ASSUMPTION: class under test, not found in repo content
        static class AavaGroupTokenRequestExecutor {
            private final AavaUserTokenResolver tokenResolver;
            private final AavaApiInvoker apiInvoker;
            private final TokenRefreshService tokenRefreshService;
            private final ExecutionAuditSanitizer executionAuditSanitizer;

            AavaGroupTokenRequestExecutor(
                    AavaUserTokenResolver tokenResolver,
                    AavaApiInvoker apiInvoker,
                    TokenRefreshService tokenRefreshService,
                    ExecutionAuditSanitizer executionAuditSanitizer
            ) {
                this.tokenResolver = tokenResolver;
                this.apiInvoker = apiInvoker;
                this.tokenRefreshService = tokenRefreshService;
                this.executionAuditSanitizer = executionAuditSanitizer;
            }

            ExecutionResult executeForCurrentUser(GenericAPI api) {
                return execute(api, null, true);
            }

            ExecutionResult executeForUser(GenericAPI api, String userReference) {
                return execute(api, userReference, false);
            }

            ExecutionResult executeAsync(AsyncJob job) {
                return executeForUser(job.getApi(), job.getCreatedBy());
            }

            private ExecutionResult execute(GenericAPI api, String userReference, boolean currentUser) {
                if (!"AAVA".equalsIgnoreCase(TestData.groupType(api))) {
                    apiInvoker.invoke(new OutboundRequest(TestData.fullUrl(api), Map.of()));
                    return new ExecutionResult(false);
                }

                String token = currentUser
                        ? tokenResolver.resolveForCurrentUser()
                        : tokenResolver.resolveForUser(userReference);

                if (token == null || token.isBlank()) {
                    throw new IllegalStateException("AAVA access token is not available for the current user.");
                }

                OutboundRequest request = new OutboundRequest(
                        TestData.fullUrl(api),
                        Map.of("Authorization", "Bearer " + token)
                );

                try {
                    apiInvoker.invoke(request);
                    return new ExecutionResult(true);
                } catch (UnauthorizedException unauthorized) {
                    String refreshed = currentUser
                            ? tokenRefreshService.refreshCurrentUserToken()
                            : tokenRefreshService.refreshTokenForUser(userReference);
                    apiInvoker.invoke(new OutboundRequest(
                            TestData.fullUrl(api),
                            Map.of("Authorization", "Bearer " + refreshed)
                    ));
                    return new ExecutionResult(true);
                } catch (RuntimeException runtimeException) {
                    throw new RuntimeException(executionAuditSanitizer.sanitize(runtimeException.getMessage()));
                }
            }
        }

        interface AavaUserTokenResolver {
            String resolveForCurrentUser();
            String resolveForUser(String userReference);
        }

        interface AavaApiInvoker {
            Response invoke(OutboundRequest request);
        }

        interface TokenRefreshService {
            String refreshCurrentUserToken();
            String refreshTokenForUser(String userReference);
        }

        interface ExecutionAuditSanitizer {
            String sanitize(String rawValue);
        }

        record OutboundRequest(String url, Map<String, String> headers) { }
        record Response(int statusCode, String body) { }
        record ExecutionResult(boolean aavaAuthenticationUsed) { }
    }

    static class TestData {
        static ApiBaseUrlGroup aavaGroup(String name, String type) {
            ApiBaseUrlGroup group = new ApiBaseUrlGroup();
            group.setName(name);
            group.setDescription(type);

        static GenericAPI api(ApiBaseUrlGroup group, String endpoint, String baseUrl) {
            GenericAPI api = new GenericAPI();
            api.setId(UUID.randomUUID());
            api.setEndpoint(endpoint); // ASSUMPTION: not found in repo content
            api.setBaseUrl(baseUrl); // ASSUMPTION: not found in repo content
            api.setApiBaseUrlGroup(group); // ASSUMPTION: not found in repo content
            return api;
        }

        static AsyncJob asyncJob(GenericAPI api, String createdBy) {
            AsyncJob job = AsyncJob.builder().api(api).build();
            job.setCreatedBy(createdBy); // ASSUMPTION: not found in repo content
            return job;
        }

        static AavaGroupTokenRequestExecutorTestSupport.Response successResponse() {
            return new AavaGroupTokenRequestExecutorTestSupport.Response(200, "ok");
        }

        static String groupType(GenericAPI api) {
            return api.getApiBaseUrlGroup().getDescription();
        }

        static String fullUrl(GenericAPI api) {
            return api.getBaseUrl() + api.getEndpoint();
        }
    }
}
