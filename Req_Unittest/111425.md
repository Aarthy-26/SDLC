        @TestConfiguration
        static class TestSecurityConfig {
            @Bean
            OncePerRequestFilter roleFilter() {
                return new OncePerRequestFilter() {
                    @Override
                    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain) throws ServletException, IOException {
                        var auth = SecurityContextHolder.getContext().getAuthentication();
                        if (auth != null) {
                            String method = request.getMethod();
                            String uri = request.getRequestURI();
                            boolean allowed = true;
                            if ("POST".equals(method) && "/api/apps".equals(uri)) {
                                allowed = auth.getAuthorities().stream().anyMatch(a -> a.getAuthority().equals("ROLE_TOOL_ADMIN"));
                            } else if ("DELETE".equals(method) && uri.startsWith("/api/apps/")) {
                                allowed = auth.getAuthorities().stream().anyMatch(a -> a.getAuthority().equals("ROLE_APP_EDITOR"));
                            } else if ("GET".equals(method) && uri.matches("/api/apps/[^/]+$")) {
                                allowed = auth.getAuthorities().stream().anyMatch(a -> a.getAuthority().equals("ROLE_APP_VIEWER"));
                            } else if ("GET".equals(method) && "/api/apps".equals(uri)) {
                                allowed = auth.getAuthorities().stream().anyMatch(a -> a.getAuthority().equals("ROLE_USER"));
                            }
                            if (!allowed) {
                                response.setStatus(HttpServletResponse.SC_FORBIDDEN);
                                return;
                            }
                        }
                        filterChain.doFilter(request, response);
                    }
                };
            }
        }
